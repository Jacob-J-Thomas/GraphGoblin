function contained(base, path) {
  const tail = relative(base, path);
  return tail !== '' && !isAbsolute(tail) && tail !== '..' && !tail.startsWith('..' + sep);
}
function parseJson(bytes, label) {
  try { return JSON.parse(bytes.toString('utf8').replace(/^\uFEFF/u, '')); }
  catch (error) { throw new Error(label + ': ' + error.message); }
}
function fileHashBeforeImport(path) {
  return readFile(path).then(sha256);
}

const captureProcessIdentity = (pid) => {
  const powershell = join(windowsRoot, 'System32', 'WindowsPowerShell', 'v1.0', 'powershell.exe');
  const command = [
    "$ErrorActionPreference = 'Stop'",
    '$qaPid = [int][Console]::In.ReadLine()',
    '$qaProcess = Get-CimInstance -ClassName Win32_Process -Filter ("ProcessId = $qaPid")',
    'if ($null -eq $qaProcess) { [ordered]@{ found = $false; processId = $qaPid } | ConvertTo-Json -Compress; exit 0 }',
    '[ordered]@{ found = $true; processId = $qaProcess.ProcessId; creationDate = $qaProcess.CreationDate; executablePath = $qaProcess.ExecutablePath; commandLine = $qaProcess.CommandLine } | ConvertTo-Json -Compress',
  ].join('\n');
  const encoded = Buffer.from(command, 'utf16le').toString('base64');
  const result = spawnSync(powershell, ['-NoLogo', '-NoProfile', '-NonInteractive', '-EncodedCommand', encoded], {
    input: String(pid) + '\n', encoding: 'utf8', windowsHide: true, timeout: 5000, maxBuffer: 32768,
    env: childEnv,
  });
  assert.equal(result.error, undefined, 'Process identity query failed');
  assert.equal(result.status, 0, 'Process identity query exit');
  return JSON.parse(result.stdout.trim());
};
