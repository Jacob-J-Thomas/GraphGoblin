async function raw(id,after=0,extra={}){const began=performance.now();const r=await fetch(`${base}/runs/${id}/events?after=${after}`,{headers:{...headers,accept:'text/event-stream',...extra},signal:AbortSignal.timeout(15000)});const text=await r.text();const frames=new SseParser().push(text);const data=frames.filter(f=>f.event!=='stream.error'&&f.event!=='stream.complete').flatMap(f=>{try{return [JSON.parse(f.data)];}catch{return [];}});return {status:r.status,ms:performance.now()-began,completions:frames.filter(f=>f.event==='stream.complete').map(f=>({id:f.id,data:JSON.parse(f.data)})),ids:data.map(e=>e.seq),types:data.map(e=>e.type),controls:frames.filter(f=>f.event==='stream.error').map(f=>({id:f.id,problem:JSON.parse(f.data)})),...(r.ok?{}:{body:text})};}
