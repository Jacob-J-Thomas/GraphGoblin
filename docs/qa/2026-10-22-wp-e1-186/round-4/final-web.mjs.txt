import { spawn } from 'node:child_process';
import { readFileSync, writeFileSync, openSync, closeSync, copyFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
const gates=JSON.parse(readFileSync('.tmp/r4-gates.json'));
if(!gates.results.some(item=>item.name==='corruption-after')) throw new Error('first gates not complete');
copyFileSync('.tmp/r4-source-frozen.json','.tmp/r4-pre-inspector-source-frozen.json');
const frozen=JSON.parse(readFileSync('.tmp/r4-source-frozen.json'));
const path='apps/web/src/runs/RunInspectorPage.test.tsx';
frozen[path]=createHash('sha256').update(readFileSync(path)).digest('hex');
writeFileSync('.tmp/r4-source-frozen.json',JSON.stringify(frozen,null,2));
gates.finalSourceFiles=Object.keys(frozen);
gates.finalSourceSnapshot='.tmp/r4-source-frozen.json';
gates.inspectorCorrection='The original web invocation tested obsolete no-reopen assertions; two updated cases now require cursor confirmation and durable audit catch-up. Production source is unchanged.';
for(const [name,command] of [
 ['typecheck-final','pnpm.cmd typecheck'],
 ['lint-final','pnpm.cmd lint'],
 ['prettier-final',`pnpm.cmd exec prettier --check ${Object.keys(frozen).join(' ')}`],
 ['coverage-web-final','pnpm.cmd --filter @graphgoblin/web test:coverage --maxWorkers=1 --coverage.reportOnFailure=true'],
]) {
 const log=`.tmp/r4-${name}.log.txt`,fd=openSync(log,'w');console.log(`START ${name}`);
 const started=new Date().toISOString();
 const code=await new Promise((resolve,reject)=>{
   const child=spawn(process.env.ComSpec??'C:\\Windows\\System32\\cmd.exe',['/d','/s','/c',command],{env:process.env,stdio:['ignore',fd,fd],windowsHide:true});child.on('error',reject);child.on('close',resolve);
 });
 closeSync(fd);
 const result={name,command,code,started,ended:new Date().toISOString(),log};
 if(name==='coverage-web-final') {
   const summary=readFileSync('apps/web/coverage/coverage-summary.json');
   writeFileSync('.tmp/r4-coverage-web-final-summary.json',summary);result.coverage=JSON.parse(summary).total;
 }
 gates.results.push(result);writeFileSync('.tmp/r4-gates.json',JSON.stringify(gates,null,2));console.log(`END ${name}: ${code}`);
 if(name!=='coverage-web-final' && code!==0) throw new Error('final static gate failed');
}
const drift=Object.entries(frozen).filter(([path,sha])=>createHash('sha256').update(readFileSync(path)).digest('hex')!==sha).map(([path])=>path);
writeFileSync('.tmp/r4-final-web-source-integrity.json',JSON.stringify({drift},null,2));
if(drift.length) throw new Error('source drift');
