function child(cmd, args, options = {}) {
  const p = spawn(cmd, args, { windowsHide: true, stdio: ['pipe','pipe','pipe'], ...options });
  const closed = once(p, 'close'); closed.catch(() => {}); owned.set(p, closed);

}
async function kill(p) {
  if (p.exitCode === null && p.signalCode === null) p.kill('SIGKILL');
  await owned.get(p).catch(() => {});
}
function send(p, type, extra = {}) {
  return new Promise((res, rej) => {
    const timer = setTimeout(() => { p.off('message', receive); rej(new Error('IPC timeout '+type)); }, 60000);
    function receive(m) { if (m.type === type) { clearTimeout(timer); p.off('message', receive); res(m); } }
    p.on('message', receive); p.send({ type, ...extra });
  });
}
async function waitFor(fn, label = 'state') {

  throw new Error('Deadline: '+label);
}

function compact(detail) {
  const snapshot=x=>x&&({runId:x.run?.id,status:x.run?.status,failure:x.run?.failure,projections:x.projections,turns:x.events?.filter(e=>e.type==='harness.turn').map(e=>({nodeId:e.nodeId,visitId:e.visitId,turnId:e.turnId,promptKind:e.promptKind,resumeSessionId:e.resumeSessionId,promptLength:e.prompt.length})),submissions:x.events?.filter(e=>e.type==='harness.submission').map(e=>({nodeId:e.nodeId,visitId:e.visitId,turnId:e.turnId,submission:e.submission,marker:e.marker})),calls:x.calls?.map(c=>({harness:c.harness,sessionId:c.sessionId,promptKind:c.prompt===continuation?'continuation':'authored',promptLength:c.prompt.length}))});
  const result={}; for(const [key,value] of Object.entries(detail||{})){if(['before','after','source','fork','child'].includes(key)){result[key]=Array.isArray(value)?value.map(snapshot):snapshot(value);}else if(['run','events','thread','calls'].includes(key)){result.observed=snapshot(detail);}else result[key]=value;}return result;
}
async function record(name, fn) {

  if(process.env.QA_ONLY && !name.startsWith('built main') && !new RegExp(process.env.QA_ONLY).test(name)) return;
  try { const detail = await fn(); report.checks.push({name, pass:true,...compact(detail)}); console.log('PASS '+name); }
  catch(e) { report.checks.push({name, pass:false,error:scrub(e.stack).slice(0,3000)}); console.log('FAIL '+name+': '+scrub(e.message).slice(0,600)); }
  await writeFile(join(evidence, process.env.QA_EVIDENCE || 'matrix-qualified.txt'), scrub(JSON.stringify(report,null,2))+'\n');
}
