export interface ScriptedTurn {
  /** Matched against the prompt; the first turn whose matcher accepts is used. Omit to match anything. */
  match?: (request: HarnessTurnRequest) => boolean;
  items?: HarnessItem[];
  finalText?: string;
  structured?: unknown;
  usage?: Partial<Usage>;
  /** Simulate a harness-level failure. */
  error?: { code: string; message: string; retriable?: boolean };
  /** Explicit transport evidence. Default failed scripts are uncertain. */
  submission?: PromptSubmission;
  /** Omit even initialization/session evidence. */
  announceSession?: boolean;
  /** Positive acceptance marker, independent of initialization. */
  accepted?: boolean;
  /** Delay before completing, in fake-clock-independent real milliseconds; lets tests cancel mid-turn. */
  delayMs?: number;
}

const ZERO_USAGE: Usage = {
  inputTokens: 0,
  outputTokens: 0,
  cachedInputTokens: 0,
  reasoningOutputTokens: 0,
};

export class FakeHarness implements HarnessPort {
  readonly started: HarnessStartRequest[] = [];
  readonly resumed: { sessionId: string; request: HarnessStartRequest }[] = [];
  readonly cancelled: string[] = [];
  preflightResult: HarnessPreflight = {
    state: 'ready',
    ok: true,
    version: 'fake',
    authenticated: true,
    problems: [],
  };
  private sessionCounter = 0;

  constructor(
    private turns: ScriptedTurn[] = [],
    readonly id: HarnessId = 'codex',
  ) {}

  /** Replace the script. Turns are consumed in order unless they carry a matcher. */
  script(turns: ScriptedTurn[]): void {
    this.turns = turns;
  }

  preflight(): Promise<HarnessPreflight> {
    return Promise.resolve(this.preflightResult);
  }

  start(request: HarnessStartRequest, signal: AbortSignal): HarnessSession {
    this.started.push(request);
    this.sessionCounter += 1;
    const sessionId =
      this.id === 'codex'
        ? `fake-session-${this.sessionCounter}`
        : `fake-${this.id}-session-${this.sessionCounter}`;
    return this.session(sessionId, 'fresh', request.turn, signal);
  }

  resume(sessionId: string, request: HarnessStartRequest, signal: AbortSignal): HarnessSession {
    this.resumed.push({ sessionId, request });
    return this.session(sessionId, 'resumed', request.turn, signal);
  }

  private pick(request: HarnessTurnRequest): ScriptedTurn {
    const index = this.turns.findIndex((t) => (t.match ? t.match(request) : true));
    if (index < 0) return { finalText: 'OK' };
    const [turn] = this.turns.splice(index, 1);
    return turn as ScriptedTurn;
  }

  private session(
    sessionId: string,
    mode: 'fresh' | 'resumed',
    request: HarnessTurnRequest,
    signal: AbortSignal,
  ): HarnessSession {
    const turn = this.pick(request);
    const usage: Usage = { ...ZERO_USAGE, ...turn.usage };
    const items = turn.items ?? [{ id: 'i1', type: 'message', summary: turn.finalText ?? 'OK' }];
    const events: HarnessEvent[] = [
      ...(turn.announceSession === false ? [] : [{ type: 'session' as const, sessionId, mode }]),
      ...((turn.accepted ?? !turn.error)
        ? [{ type: 'submitted' as const, marker: 'fake.accepted' as const }]
        : []),
      ...items.map((item) => ({ type: 'item', item }) as const),
    ];
    if (turn.error) {
      events.push({
        type: 'error',
        code: turn.error.code,
        message: turn.error.message,
        retriable: turn.error.retriable ?? false,
        submission:
          turn.submission ??
          (turn.error.code === 'HARNESS_SESSION_MISSING'
            ? 'session-missing'
            : 'delivery-uncertain'),
      });
    } else {
      events.push({ type: 'usage', usage }, { type: 'turn-complete' });
    }
    const cancelled = { value: false };
    const resultPromise: Promise<HarnessResult> = new Promise((resolve, reject) => {
      const finish = (): void => {
        if (signal.aborted || cancelled.value) {
          reject(new DOMException('aborted', 'AbortError'));
          return;
        }
        if (turn.error) {
          reject(
            Object.assign(new Error(turn.error.message), {
              code: turn.error.code,
              retriable: turn.error.retriable ?? false,
            }),
          );
          return;
        }
        resolve({
          finalText: turn.finalText ?? 'OK',
          ...(turn.structured !== undefined ? { structured: turn.structured } : {}),
          usage,
          items,
        });
      };
      if (turn.delayMs) {
        const timer = setTimeout(finish, turn.delayMs);
        signal.addEventListener(
          'abort',
          () => {
            clearTimeout(timer);
            finish();
          },
          { once: true },
        );
      } else {
        finish();
      }
    });
    resultPromise.catch(() => undefined);
    const cancelledLog = this.cancelled;
    return {
      sessionId: Promise.resolve(sessionId),
      events: asyncIterableOf(events),
      result: resultPromise,
      cancel(): Promise<void> {
        cancelled.value = true;
        cancelledLog.push(sessionId);
        return Promise.resolve();
      },
    };
  }
}

      workingDirectory,

      effort,
      options: config.harnessOptions,
      ...(config.capabilities ? { capabilities: config.capabilities } : {}),

    let afterResultFailure: RunFailureError | undefined;
    let result: HarnessResult;
    let session: HarnessSession | undefined;
    let invoking = false;
    let intentUnconfirmed = true;
    pending.intentUnconfirmed = true;
    const failures: TurnFailures = { rejected: false, recording: [] };
    let usage: Usage = ZERO_USAGE;
    try {
      // Initial tasks, exact retries and repairs share the same guarded intent boundary.
      await recordTurn(pending);
      intentUnconfirmed = false;
      pending.intentUnconfirmed = false;
      await ctx.ports.sessions.upsert({ ...row });
      if (!harness) {

        runId,
        new InternalRunFailureError(`loop version ${run.versionId} not found`, {
          resumable: false,
        }),
      );
      return;
    }
    // A run admitted before ADR-0044 can be pinned to a definition whose exit publishes an Event:
    // fail it explicitly before any node runs, instead of running to an exit it cannot deliver.
    const refused = definitionUpgradeRequired(version);
    if (refused) {
      await this.failRun(runId, refused.toFailure());
      return;
    }
    const def = version.definition;
    // Inherited defaults (and classifier bindings, read by the decision handler) are the ones
    // pinned when the run was created (ADR-0041): a Settings or process change reaches new runs
    // only, never a run that waits, resumes or recovers.
    const { owner: ownerDefaults, process: processDefaults } = run.pinned.defaults;
    // Read before any recovery marker is appended; a marker does not consume a wake either.
    const pendingWake = findPendingWake(events);
    let thread = await this.loadThread(run, events);
    const freshEntries = new FreshEntries(events);
    /** Append drafts atomically and return the seq of the last one. */
    const appendAll = async (drafts: EventDraft[]): Promise<number> => {
      for (const draft of drafts)
        if (draft.type === 'decision.made') {
          const { type: _type, nodeId: _nodeId, ...evidence } = draft;
          DecisionEmissionSchema.parse(evidence);
        }
      const stored = await this.ports.events.append(runId, drafts);
      freshEntries.record(stored);
      events.push(...stored);
      return (stored.at(-1) as RunEvent).seq;
    };

      await appendAll([draft]);
    };

      } else if (this.recovering.delete(runId)) {

        await append({ type: 'run.started', attempt: previousStarts + 1 });
      }
    }

    // the cursor write must not run the finished node again.
    const caughtUp = await this.catchUpCursor(run, def, events);
