  private async startAdmittedRun(
    input: StartRunInput,
    beforeAdmission?: (plan: RunStartPlan) => Promise<void>,
  ): Promise<RunRecord> {
    const prepared = await this.prepareStart(input);
    if (beforeAdmission) {
      const versions = new Map([[prepared.version.id, prepared.version]]);
      const intent = await this.withPinLock(async () => {
        await this.assertVersionPresent(prepared.sourceId, input.loopId);
        const intent = await this.prepareAdmission(prepared, input.subloopVersions ?? {}, versions);
        return input.parentNodeId ? { ...intent, parentNodeId: input.parentNodeId } : intent;
      });
      // Slow admission policy (such as no-turn CLI probes) never holds the shared pin lock.
      // Commit this same intent afterwards: a publish or draft edit cannot change what was checked.
      await beforeAdmission({
        root: prepared.version,
        versions,
        subloopVersions: intent.queued.subloopVersions ?? {},
        defaults: prepared.run.pinned.defaults,
      });
      const admitted = await this.withPinLock(async () => {
        await this.assertVersionPresent(prepared.sourceId, input.loopId);
        for (const version of versions.values())
          if (version.id !== prepared.version.id)
            await this.assertVersionPresent(version.id, version.loopId);

        return this.ports.admission.create(intent);
      });
      if (input.parentRunId && this.draining) this.drainRuns.add(admitted.id);
      if (admitted.id === prepared.run.id) this.enqueue(admitted.id);
      return admitted;
    }
    const admitted = await this.withPinLock(async () => {
      await this.assertVersionPresent(prepared.sourceId, input.loopId);
      const intent = await this.prepareAdmission(prepared, input.subloopVersions ?? {});
      return this.ports.admission.create(
        input.parentNodeId ? { ...intent, parentNodeId: input.parentNodeId } : intent,
      );
    });
    if (input.parentRunId && this.draining) this.drainRuns.add(admitted.id);
    if (admitted.id === prepared.run.id) this.enqueue(admitted.id);
    return admitted;
  }

  transition(
    runId: string,
    from: readonly RunStatus[],
    changes: RunRecordChanges,
    guard?: { waitId?: string; observedSeq?: number },
  ): Promise<RunRecord | undefined> {
    const current = this.runs.get(runId);
    if (!current) return Promise.reject(new Error(`run ${runId} not found`));
    if (
      !from.includes(current.status) ||
      (guard?.waitId !== undefined && current.waiting?.waitId !== guard.waitId) ||
      (guard?.observedSeq !== undefined && current.lastEventSeq !== guard.observedSeq)
    )
      return Promise.resolve(undefined);
    return Promise.resolve(this.apply(current, changes));
  }
  claimCancel(
    runId: string,
    from: readonly RunStatus[],
    at: string,
  ): Promise<RunRecord | undefined> {
    const current = this.runs.get(runId);
    if (!current) return Promise.reject(new Error(`run ${runId} not found`));
    if (current.cancelRequestedAt || !from.includes(current.status))
      return Promise.resolve(undefined);
    return Promise.resolve(this.apply(current, { cancelRequestedAt: at }));
  }
  private apply(current: RunRecord, changes: RunRecordChanges): RunRecord {
    const next: Record<string, unknown> = { ...current };
    for (const [key, value] of Object.entries(changes)) {
      if (value === undefined) delete next[key];
      else next[key] = value;
    }
    const record = next as unknown as RunRecord;
    this.runs.set(current.id, record);
    return record;
  }

/** Services the executor hands to handlers. Everything that touches the outside world goes through here. */
export interface HandlerServices {
  /** Render a Liquid template against the thread view plus extras. */
  render(template: string, extras?: Record<string, unknown>): Promise<string>;
  /** Append events to the run log right away (progress and usage). Decision completion is appended atomically by the executor. */
  record(draft: EventDraft): Promise<void>;
  /** Resolve the model and effort for an inferencing or decision step. */
  resolveModel(
    harness: HarnessId,
    model?: string,
    effort?: Effort,
  ): Promise<{ model: string; effort: Effort }>;
  /**
   * Start a child run and return its id. The child and this run's `child_run.started` commit
   * together (#320), so the visit's child is in the log before this returns. The parent then parks
   * on it.
   */
  startChild(request: ChildStartRequest): Promise<string>;
  /** Read a finished child's outcome and final thread. */
  childOutcome(childRunId: string): Promise<ChildOutcome>;
  /**
   * Resume a child whose runtime failure is resumable, on behalf of this run (an explicit resume
   * of this run re-executes its subloop node, which resumes the child as an inline node would be
   * re-executed), then read the child again.
   */
  resumeChild(childRunId: string): Promise<ChildOutcome>;
  /** Events of this run so far. */
  events(): Promise<readonly RunEvent[]>;
  /** Current depth of this run in the subloop tree (0 for a top-level run). */
  depth(): Promise<number>;
  /** Working directory for this run, resolved once per run. */
  workingDirectory(): Promise<WorkingDirectoryResult>;
  /** Generate ids and read the clock. */
  newId(): string;
  now(): string;
}
