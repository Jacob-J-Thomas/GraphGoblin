const RECORDING_OPERATIONS = [
  'harness.submission',
  'turn.intent.reconcile',
  'harness.status',
] as const;
export type InferenceRecordingOperation = (typeof RECORDING_OPERATIONS)[number];

type JsonObject = Record<string, JsonValue>;

function isObject(value: JsonValue | undefined): value is JsonObject {
  return (
    value !== undefined && value !== null && typeof value === 'object' && !Array.isArray(value)
  );
}

function isChildWrapper(details: JsonObject): details is JsonObject & { childRunId: string } {
  return (
    UlidSchema.safeParse(details.childRunId).success &&
    (details.childNodeId === undefined || SlugSchema.safeParse(details.childNodeId).success) &&
    Object.keys(details).every(
      (key) => key === 'childRunId' || key === 'childNodeId' || key === 'childDetails',
    )
  );
}

function isLegacyRecording(
  entry: JsonValue,
): entry is JsonObject & { operation: InferenceRecordingOperation } {
  return (
    isObject(entry) &&
    RECORDING_OPERATIONS.some((operation) => operation === entry.operation) &&
    typeof entry.message === 'string' &&
    (entry.name === undefined || typeof entry.name === 'string') &&
    (entry.stack === undefined || typeof entry.stack === 'string') &&
    Object.keys(entry).every(
      (key) => key === 'operation' || key === 'message' || key === 'name' || key === 'stack',
    )
  );
}

function internalMessage(details: JsonObject): string {
  if (
    typeof details.code === 'string' &&
    (details.strategy === undefined ||
      details.strategy === 'codex' ||
      details.strategy === 'jev') &&
    Object.keys(details).every((key) => key === 'code' || key === 'strategy')
  ) {
    const provider = deciderFailure(details);
    // Reconstruct fixed copy from the known code. Historical message text is untrusted.
    if (provider.details.code === details.code) return provider.message;
  }
  return INTERNAL_FAILURE_MESSAGE;
}

function childDetails(
  details: JsonValue | undefined,
  internal: boolean,
  diagnostics: JsonValue[],
): { details: JsonValue | undefined; internalMessage: string } {
  if (!isObject(details)) return { details, internalMessage: INTERNAL_FAILURE_MESSAGE };
  if (isChildWrapper(details)) {
    const child = childDetails(details.childDetails, internal, diagnostics);
    return {
      details: child.details === undefined ? details : { ...details, childDetails: child.details },
      internalMessage: `subloop child ${details.childRunId} failed: ${child.internalMessage}`,
    };
  }

  const message = internalMessage(details);
  if (
    internal &&
    typeof details.name === 'string' &&
    (typeof details.stack === 'string' || details.stack === null)
  ) {
    const { name, stack, ...publicDetails } = details;
    diagnostics.push({ name, stack });
    details = publicDetails;
  }
  if (Array.isArray(details.recording)) {
    const recording = details.recording.map((entry) => {
      if (!isLegacyRecording(entry)) return entry;
      diagnostics.push(entry);
      return { operation: entry.operation };
    });
    details = { ...details, recording };
  }
  return { details, internalMessage: message };
}

/** Redact only the old engine slots on the child-failure chain being copied into a new run. */
export function childFailureForPublication(failure: RunFailure) {
  const diagnostics: JsonValue[] = [];
  const internal = failure.code === 'INTERNAL_ERROR';
  const projected = childDetails(failure.details, internal, diagnostics);
  const message = internal ? projected.internalMessage : failure.message;
  if (message !== failure.message) diagnostics.unshift({ message: failure.message });
  return { message, details: projected.details, diagnostics };
}

function renderCanonical(template: string, thread: ContextThread): Promise<string> {
  const view = { ...threadView(thread) };
  return renderTemplate(template, view).catch(() => {

/** Trim only a known history slot, never an arbitrary rendered prompt or the stored thread. */
export async function admitInferencePrompt(options: {
  prompt: string;
  template: string;
  thread: ContextThread;
  allowHistoryTrim: boolean;
  window: Window;
}) {
  const { window } = options;
  const parsed = ChatLoopInputSchema.safeParse(options.thread.invocation.trigger.payload);
  const history = parsed.success ? parsed.data.history : undefined;
  const paired =
    history !== undefined &&
    history.length % 2 === 0 &&
    history.every((entry, index) => entry.role === (index % 2 === 0 ? 'user' : 'assistant'));
  const canTrim = options.allowHistoryTrim && canonicalPrompts.has(options.template) && paired;
  const pairs = paired ? history.length / 2 : 0;
  const report = (retainedHistoryPairs: number) => ({
    profile: window.profile,
    contextTokens: window.contextTokens,
    outputReserveTokens: window.outputReserveTokens,
    retainedHistoryPairs,
    omittedHistoryPairs: pairs - retainedHistoryPairs,
  });
  if (window.fits(options.prompt)) return { prompt: options.prompt, context: report(pairs) };
  if (!canTrim || !parsed.success) overflow();
  const candidateThread = structuredClone(options.thread);
  candidateThread.invocation.trigger.payload = JsonValueSchema.parse({
    ...parsed.data,
    history: [],
  });
  let prompt = await renderCanonical(options.template, candidateThread);
  // Authored instructions and required current/normalized input must fit on their own.
  if (!window.fits(prompt)) overflow();
  let retained = 0;
  for (let count = 1; count <= pairs; count += 1) {
    candidateThread.invocation.trigger.payload = JsonValueSchema.parse({
      ...parsed.data,
      history: parsed.data.history.slice(-count * 2),
    });
    const candidate = await renderCanonical(options.template, candidateThread);
    if (!window.fits(candidate)) break;
    prompt = candidate;
    retained = count;
  }
  return { prompt, context: report(retained) };
}

## M3 - Persistence and API (M)

M3 notes: the intermittent 0xC0000005 crash in the `apps/api` suite comes from libsql's native addon, which on Windows can crash or hang the process when it is loaded only inside a worker thread and that thread exits, as Vitest's `threads` and `vmThreads` pools do. `apps/api` and `adapter-sqlite` (now `infrastructure`) therefore pin Vitest to `pool: 'forks'`, where the crash never reproduced; production loads libsql on the main thread and is unaffected. Fixing the suite also surfaced a real bug: revoking an already revoked API key returned 204 instead of 404, so `SqliteApiKeys.revoke` now matches only active keys. Details are in `research/sqlite-libsql.md`.

Goal: the engine runs behind a real API with SQLite, usable from curl before any UI exists.

Packages: `infrastructure` (`src/sqlite`, `src/scheduler` for timers only, `src/fs`, `src/process`, `src/http`), `apps/api`. Built as five `adapter-*` packages and merged by ADR-0014.

Tasks:
