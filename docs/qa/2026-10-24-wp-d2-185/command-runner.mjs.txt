import fs from 'node:fs';
import { spawn } from 'node:child_process';
const root = process.cwd(), dir = root + '/docs/qa/2026-10-24-wp-d2-185';
const jobs = process.argv.includes('--build') ? [
  ['build-final', 'pnpm.cmd build'],
  ['browser-final', 'pnpm.cmd --filter @graphgoblin/web exec playwright test e2e/webhook-delivery.spec.ts'],
  ['native-after', 'node "%TEMP%/gg-wp-d2-185-reproduce.mjs" --verify'],
] : ['engine','infrastructure','api-client','api','mcp','web'].map(p => ['coverage-'+p, 'pnpm.cmd --filter @graphgoblin/'+p+' test:coverage --maxWorkers=2 --coverage.reportOnFailure']);
(async () => {
  for (const [name, command] of jobs) {
    const fd = fs.openSync(dir+'/'+name+'.log.txt', 'w');
    console.log('START '+command);
    const child = spawn(process.env.ComSpec || 'cmd.exe', ['/d','/s','/c',command], { cwd: root, windowsHide:true, env: process.env, stdio:['ignore',fd,fd] });
    const code = await new Promise((ok,no) => { child.on('error',no); child.on('exit',ok); });
    fs.closeSync(fd);
    fs.appendFileSync(dir+'/command-results.jsonl',JSON.stringify({ command, code })+'\n');
    console.log('END '+name+' '+code);
    if (process.argv.includes('--build') && name === 'build-final' && code !== 0) break;
  }
})().catch(e => { console.error(e); process.exitCode=1; });
