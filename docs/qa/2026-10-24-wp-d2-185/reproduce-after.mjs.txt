const receiver = createServer((req, res) => {

    const n = (counts.get(req.url) ?? 0) + 1; counts.set(req.url, n);

let child, apiLog = '';
async function start() {
  child = spawn(process.execPath, ['apps/api/dist/main.js'], { cwd: process.cwd(), windowsHide: true,
    env: { ...process.env, GG_DATA_DIR: dataDir, GG_DB_URL: 'file:' + join(dataDir, 'graphgoblin.db').replaceAll(String.fromCharCode(92), '/'), GG_WEB_DIST: resolve('apps/web/dist'), GG_MASTER_KEY: Buffer.alloc(32, 9).toString('base64'), GG_PORT: '5800', GG_REQUIRE_API_KEY: 'false', GG_LOG_LEVEL: 'warn' }, stdio: ['ignore', 'pipe', 'pipe'] });
  child.stdout.on('data', b => { apiLog += b; }); child.stderr.on('data', b => { apiLog += b; });
  for (let n = 0; n < 300; n++) {
    if (child.exitCode !== null) throw Error('API exited: ' + apiLog);
    try { if ((await fetch(baseUrl + '/healthz')).ok) return; } catch {}
    await delay(50);
  }
  throw Error('API did not start: ' + apiLog);
}

}

async function json(path, body) {
  const response = await fetch(baseUrl + path, body === undefined ? {} : { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) });
  const result = await response.json(); if (!response.ok) throw Error(JSON.stringify(result)); return result;
}
async function make(kind) {
  const { loop } = await json('/loops', { definition: { schemaVersion: 4, name: '185 ' + kind,
    nodes: [{ id: 'start', kind: 'trigger', label: 'Start', config: { type: 'manual' } },
      { id: 'done', kind: 'exit', label: 'Done', config: { returns: 'trigger.payload', destinations: kind === 'none' ? [] : [{ kind: 'webhook', url: 'http://127.0.0.1:5801/' + kind }] } }],
    edges: [{ id: 'a', from: { node: 'start', port: 'out' }, to: { node: 'done' } }] } });
  await json('/loops/' + loop.id + '/publish', {}); return (await runs.start(client, loop.id, { input: 42 })).id;
}
async function until(check) { for (let n = 0; n < 400; n++) { if (await check()) return; await delay(50); } throw Error('condition timed out'); }
async function observe(label, id) {
  const snapshot = await runs.get(client, id), events = await runs.events(client, id);
  const terminal = events.items.findLast(e => ['run.finished', 'run.failed', 'run.cancelled'].includes(e.type));
  const observation = { case: label, execution: snapshot.status, delivery: snapshot.deliveries?.map(d => ({ status: d.status, attempts: d.attempts, id: d.id })), finalized: snapshot.finalizedAt !== undefined };
  for (const after of [0, terminal.seq, snapshot.lastEventSeq + 100]) {
    let settled = false; const seen = [];
    const sub = subscribeRunEvents({ client, runId: id, after, onEvent: e => { seen.push(e.type); }, backoff: { initialMs: 20 } });
    subscriptions.push(sub); void sub.done.then(() => { settled = true; });
    await delay(250); observation['SSE after ' + after] = { settled, seen }; sub.close();
  }
  const wait = await waitForRun(client, id, { timeoutMs: 300, pollMs: 50 });
  observation.clientWait = { finished: wait.finished, status: wait.run?.status, finalized: wait.run?.finalizedAt !== undefined };
  const tool = await mcp.callTool({ name: 'wait_for_run', arguments: { runId: id, timeoutSeconds: 1 } });
  const result = JSON.parse(tool.content[0].text);
  observation.mcpWait = { finished: result.finished, status: result.status ?? result.run?.status, deliveries: result.deliveries ?? result.run?.deliveries };
  await page.goto(baseUrl + '/app/runs/' + id); await page.getByRole('heading', { name: /^Timeline/ }).waitFor(); await delay(300);
  observation.inspector = { timeline: await page.getByRole('heading', { name: /^Timeline/ }).innerText(), text: (await page.locator('body').innerText()).match(/Execution:.*|Pending \(.*|Delivered after.*|Failed after.*/g) };
  await page.screenshot({ path: resolve('docs/qa/2026-10-24-wp-d2-185/' + label.replaceAll(' ', '-') + '.png'), fullPage: true });
  console.log(JSON.stringify(observation));
  if (process.argv.includes('--verify')) {
    const isPending = label.includes('pending') && !label.startsWith('none');
    for (const [key, item] of Object.entries(observation)) {
      let actual, expected;
      if (key.startsWith('SSE after')) { actual = item.settled; expected = !isPending; }
      else if (key === 'clientWait' || key === 'mcpWait') { actual = item.finished; expected = !isPending; }
      else if (key === 'inspector') { actual = item.timeline.includes(', complete)'); expected = !isPending; }
      else continue;
      try { assert.equal(actual, expected, label + ': ' + key); }
      catch (error) { failures.push(error.message); }
    }
  }
}
function release(kind) { const key = '/' + kind; released.add(key); const response = pending.get(key); pending.delete(key); if (response && !response.destroyed && !response.writableEnded) response.writeHead(kind === 'failure' ? 400 : 200).end(); }

    if (kind === 'none') await until(async () => (await runs.get(client, id)).finalizedAt !== undefined);
    await observe(kind + ' pending', id);
    if (kind === 'restart') {

      const seen = [];

        onEvent: event => { seen.push({ seq: event.seq, type: event.type }); },
        onOpen: () => { opened++; }, onError: (_error, attempt) => { if (attempt) errors++; },
        backoff: { initialMs: 20, maxMs: 100 },
      });
      subscriptions.push(live);
      void live.done.then(() => { settled = true; });
      await until(() => opened > 0);
      assert.equal(settled, false, 'pending subscription before real restart');

      assert.equal(settled, false, 'restart/EOF cannot complete pending delivery');

      assert.equal(new Set(seen.map(e => e.seq)).size, seen.length);
      console.log(JSON.stringify({ case: 'continuous restart subscription', opened, errors, settled, seen }));
    }
    release(kind); await until(async () => ((await runs.get(client, id)).finalizedAt !== undefined));
