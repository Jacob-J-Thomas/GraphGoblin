async function check(name, fn) {

async function api(base, path, method = 'GET', body, token) {
  const r = await fetch(base + path, { method, headers: { ...(body === undefined ? {} : { 'content-type': 'application/json' }), ...(token ? { authorization: 'Bearer ' + token } : {}) }, ...(body === undefined ? {} : { body: JSON.stringify(body) }) });
  const t = await r.text(); let data; try { data = JSON.parse(t); } catch { data = t; }
  return { status: r.status, data };
}

}

async function terminal(base, id) {
  let r;
  for (let i = 0; i < 100; i++) { r = await api(base, '/runs/' + id); if (['succeeded', 'failed', 'cancelled'].includes(r.data.status)) return r.data; await delay(100); }
  throw new Error('run did not terminate: ' + JSON.stringify(r));
}

async function stopChild(child) {
  if (!child || child.exitCode !== null || child.signalCode !== null) return;
  const exited = once(child, 'exit'); child.kill();
  await Promise.race([exited, delay(10000).then(() => { if(child.exitCode === null) throw new Error('child did not exit'); })]);
}
