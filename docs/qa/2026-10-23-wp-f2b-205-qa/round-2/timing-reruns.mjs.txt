import {readFile,writeFile,open} from 'node:fs/promises';
import {execFileSync,spawn} from 'node:child_process';
import path from 'node:path';
const out=path.resolve('docs/qa/2026-10-23-wp-f2b-205-qa/round-2');
const changed=new Set(execFileSync('git',['diff','--name-only','0a1ff340dc8e0eb0ad6621b4020d100dcdc3be71','HEAD'],{encoding:'utf8'}).trim().split(/\r?\n/));
const gates=JSON.parse(await readFile(path.join(out,'continued-gates.json'),'utf8'));
const results=[];
for(const gate of gates){
 if(gate.code===0||!gate.command.includes('test:coverage'))continue;
 const log=await readFile(path.join(out,gate.log),'utf8');
 if(!/timed out|timeout/i.test(log))continue;
 const pkg=gate.command.match(/@graphgoblin\/(\S+)/)?.[1];
 const base=pkg==='api'||pkg==='web'?`apps/${pkg}`:`packages/${pkg}`;
 const files=[...new Set([...log.matchAll(/FAIL\s+(src\/[^\s:]+\.test\.tsx?)/g)].map(m=>m[1]))];
 for(const file of files){
  if(changed.has(base+'/'+file))continue;
  const name=`isolated-${pkg}-${file.replaceAll('/','-')}.log.txt`;
  const command=`pnpm.cmd --filter @graphgoblin/${pkg} exec vitest run ${file} --reporter=verbose`;
  const fd=await open(path.join(out,name),'w');const started=Date.now();
  const child=spawn(process.env.ComSpec??'cmd.exe',['/d','/s','/c',command],{stdio:['ignore',fd.fd,fd.fd],windowsHide:true});
  const code=await new Promise(r=>child.on('exit',r));await fd.close();
  results.push({command,code,seconds:(Date.now()-started)/1000,log:name,initial:gate.log});
  await writeFile(path.join(out,'timing-reruns.json'),JSON.stringify(results,null,2));console.log(JSON.stringify(results.at(-1)));
 }
}
await writeFile(path.join(out,'timing-reruns.json'),JSON.stringify(results,null,2));
