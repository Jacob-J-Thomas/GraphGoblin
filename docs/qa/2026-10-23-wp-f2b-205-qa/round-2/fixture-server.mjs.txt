import {pathToFileURL} from 'node:url';
import path from 'node:path';
import {mkdir} from 'node:fs/promises';
import {createServer} from 'node:http';
const root=process.cwd(); const imp=p=>import(pathToFileURL(path.join(root,p)).href);
const {buildApp,createContainer,loadConfig}=await imp('apps/api/dist/index.js');
const {FakeHarness,FakeClassifier,FakeClassifierRegistry,FakeStructured,CapturingLogger}=await imp('packages/engine/dist/testing/index.js');
const data=process.env.QA_DATA; await mkdir(data,{recursive:true});
const defaults=process.env.QA_DEFAULTS?JSON.parse(process.env.QA_DEFAULTS):{byHarness:{codex:{model:'gpt-6-luna',effort:'low'},claude:{model:'claude-opus-5-5',effort:'low'}}};
class CountingHarness extends FakeHarness { probes=0; async preflight(){this.probes++;return super.preflight();} }
const harnesses={codex:new CountingHarness([],'codex'),claude:new CountingHarness([],'claude')};
const recorded={codex:[],claude:[]};
const deciders=['codex','claude'].map(id=>({id,available:()=>true,route:async request=>{recorded[id].push(request);return {route:request.routes[0].id,confidence:1,reasoning:'QA fake'};}}));
const logger=new CapturingLogger();
const container=await createContainer(loadConfig({GG_DATA_DIR:data,GG_DB_URL:pathToFileURL(path.join(data,'qa.sqlite')).href,GG_SWAGGER_UI:'false',GG_REQUIRE_API_KEY:'true',GG_MASTER_KEY:Buffer.alloc(32,12).toString('base64'),GG_DEFAULTS:JSON.stringify(defaults),GG_WEB_DIST:path.join(root,'apps/web/dist')}),{harnesses,deciders,classifiers:new FakeClassifierRegistry(new FakeClassifier('jev')),structured:new FakeStructured(),logger,startTimers:false});
await container.start();
let entries=await container.repos.catalog.list();
function setCapabilities(only){harnesses.claude.preflightResult={state:'ready',ok:true,version:'2.1.285 (independent QA fake)',authenticated:true,problems:[],authMethod:'claude.ai',supportedPolicies:[{sandbox:'read-only',approval:'never',permissionMode:'dontAsk',tools:['Read','Glob','Grep'],authMethod:'claude.ai',boundary:'builtin-tools',network:'unconfined'}],models:entries.filter(e=>e.harness==='claude').map(e=>({model:e.model,efforts:only??e.efforts}))};}
setCapabilities();
const app=await buildApp(container,{logger:false});
await app.listen({host:'127.0.0.1',port:5840});
const {token}=await container.repos.apiKeys.create('local','QA all',['*']);
const {token:scoped}=await container.repos.apiKeys.create('local','QA no system',['loops:read','loops:write','runs:read','runs:write','settings:read','settings:write']);
const server=createServer(async(req,res)=>{try{let raw='';for await(const b of req)raw+=b;const body=raw?JSON.parse(raw):{};let result;
 if(req.url==='/state')result={started:Object.fromEntries(Object.entries(harnesses).map(([id,h])=>[id,h.started])),deciders:recorded,probes:Object.fromEntries(Object.entries(harnesses).map(([id,h])=>[id,h.probes])),logs:logger.lines};
 else if(req.url==='/reset'){for(const h of Object.values(harnesses)){h.started.length=0;h.probes=0;}for(const a of Object.values(recorded))a.length=0;result={ok:true};}
 else if(req.url==='/direct'){result=await container.manager.startRun(body);}
 else if(req.url==='/catalog'){await container.repos.catalog.upsert(body);entries=await container.repos.catalog.list();result={ok:true};}
 else if(req.url==='/capabilities'){setCapabilities(body.efforts);result={ok:true};}
 else if(req.url==='/script'){harnesses[body.harness].script(body.turns);result={ok:true};}
 else if(req.url==='/stop'){res.end('{"ok":true}');setImmediate(async()=>{await app.close();await container.stop();server.close();process.exit(0);});return;}
 else {res.statusCode=404;result={error:'unknown'};}
 res.setHeader('content-type','application/json');res.end(JSON.stringify(result));
 }catch(e){res.statusCode=500;res.end(JSON.stringify({error:String(e)}));}});
await new Promise(r=>server.listen(5841,'127.0.0.1',r));
console.log(JSON.stringify({ready:true,token,scoped,data:data.replace(process.env.USERPROFILE,'%USERPROFILE%'),defaults}));
