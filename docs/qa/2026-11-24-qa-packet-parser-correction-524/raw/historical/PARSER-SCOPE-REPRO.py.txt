"""Run the #524 parser/sparse public-command control in a disposable Git repository."""

import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile


CONFIG_FILES = [".gitattributes", ".prettierignore", ".prettierrc", ".gitignore"]
HTML_PATH = "docs/qa/visible/bad.html"
OMITTED_PATH = "docs/qa/omitted/README.md"
CONTENTS = {
    "docs/qa/visible/A.md": "# Formatted page\n",
    HTML_PATH: "<main><section>Hello</main>\n",
    OMITTED_PATH: "# Omitted page\n",
}
VALID_HTML = "<p>OK</p>\n"


def sha256(content):
    return hashlib.sha256(content).hexdigest()


def main(checkout):
    checker = checkout.resolve() / "tooling/scripts/check-qa-packet.mjs"
    environment = {k: v for k, v in os.environ.items() if not k.upper().startswith("GIT_")}
    environment.update(
        GIT_CONFIG_NOSYSTEM="1",
        GIT_CONFIG_GLOBAL=os.devnull,
        GIT_AUTHOR_NAME="Fixture",
        GIT_COMMITTER_NAME="Fixture",
        GIT_AUTHOR_EMAIL="qa@example.invalid",
        GIT_COMMITTER_EMAIL="qa@example.invalid",
        GIT_AUTHOR_DATE="2026-11-24T00:00:00Z",
        GIT_COMMITTER_DATE="2026-11-24T00:00:00Z",
    )
    with tempfile.TemporaryDirectory(prefix="gg-524-final-pair-") as temporary_directory:
        fixture = Path(temporary_directory).resolve()
        assert fixture.parent == Path(tempfile.gettempdir()).resolve()

        def git(*arguments):
            return subprocess.check_output(
                ["git", "-C", str(fixture), *arguments],
                env=environment,
                stderr=subprocess.PIPE,
            ).decode("utf-8")

        def run_checker():
            index = fixture / ".git/index"
            before = sha256(index.read_bytes())
            result = subprocess.run(
                ["node", str(checker), "--base", "main"],
                cwd=fixture,
                env=environment,
                capture_output=True,
                timeout=30,
            )
            after = sha256(index.read_bytes())
            return {
                "exit": result.returncode,
                "stdout": result.stdout.decode("utf-8").splitlines(),
                "stderr": result.stderr.decode("utf-8").splitlines(),
                "indexBefore": before,
                "indexAfter": after,
            }

        git("init", "--quiet", "--initial-branch=main")
        for name in CONFIG_FILES:
            shutil.copyfile(checkout / name, fixture / name)
        git("add", "--", *CONFIG_FILES)
        git("commit", "-qm", "base")
        git("switch", "-qc", "feature")
        for name, content in CONTENTS.items():
            destination = fixture / name
            destination.parent.mkdir(parents=True, exist_ok=True)
            destination.write_bytes(content.encode())
        git("add", "--", "docs/qa")
        git("commit", "-qm", "packet")
        selected = git("diff", "--name-only", "main", "HEAD").splitlines()
        assert set(selected) == set(CONTENTS)
        full = run_checker()

        git("sparse-checkout", "set", "--cone", "docs/qa/visible")
        flag = git("ls-files", "-t", "--", OMITTED_PATH).strip()
        assert flag == "S " + OMITTED_PATH and not (fixture / OMITTED_PATH).exists()
        sparse = run_checker()
        (fixture / HTML_PATH).write_bytes(VALID_HTML.encode())
        repaired_html = run_checker()
        return {
            "signature": "qa-packet/parser-error-discards-selected-scope",
            "checkerSha256": sha256(checker.read_bytes()),
            "fixtureBase": git("rev-parse", "main").strip(),
            "fixtureHead": git("rev-parse", "HEAD").strip(),
            "fixtureConfigFilesFromCandidate": CONFIG_FILES,
            "inputFiles": CONTENTS,
            "confirmedSelectedByGit": selected,
            "skipWorktreeEntry": flag,
            "missingOnDisk": not (fixture / OMITTED_PATH).exists(),
            "fullCheckoutMalformed": full,
            "sparseCheckoutMalformed": sparse,
            "onlyControlEdit": {"path": HTML_PATH, "bytes": VALID_HTML},
            "sparseCheckoutRepairedHtml": repaired_html,
        }


if __name__ == "__main__":
    print(json.dumps(main(Path(sys.argv[1])), indent=2))
