async function startServer(){

 const ready=await new Promise((resolve,reject)=>{const t=setTimeout(()=>reject(Error('startup deadline '+log)),60000);proc.on('message',m=>{if(m.ready){clearTimeout(t);resolve(m);}else if(m.id){const p=pending.get(m.id);if(p){pending.delete(m.id);m.error?p.reject(Error(m.error)):p.resolve(m.value);}}});proc.once('exit',c=>{clearTimeout(t);reject(Error('server exit '+c+' '+log));});});

}
