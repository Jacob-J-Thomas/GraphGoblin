// Audit final report references, source boundary, path normalization and evidence sizes/hashes.
import assert from 'node:assert/strict';
import {readFile,writeFile,readdir} from 'node:fs/promises';
import {join} from 'node:path';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
const root=process.env.QA_ROOT,out=join(root,'docs/qa/2026-10-24-wp-g1-116-qa'),profile=process.env.USERPROFILE;
const forms=[JSON.stringify(profile).slice(1,-1),profile,profile.replaceAll('\\','/')];
for(const name of await readdir(out)){
  if(!name.endsWith('.log.txt'))continue;
  const b=await readFile(join(out,name));let t=b[0]===255&&b[1]===254?b.subarray(2).toString('utf16le'):b.toString('utf8');
  for(const form of forms)t=t.split(form).join('%USERPROFILE%');t=t.replaceAll('\r\n','\n');await writeFile(join(out,name),t);
}
const names=(await readdir(out)).sort(),readme=await readFile(join(out,'README.md'),'utf8');
const metadata=[];
for(const name of ['failed','restart-failed','resumed-unfinished','midway-reload','succeeded']){
  const capture=JSON.parse(await readFile(join(out,name+'.json'),'utf8'));
  for(const view of capture.views){const prefix=view.body.split('Timeline')[0],statusAgrees=prefix.includes(capture.status),cursorAgrees=view.body.includes('Thread at event '+capture.run.lastEventSeq),explicitlyOffline=prefix.includes('Cannot reach the GraphGoblin API')&&prefix.includes('Offline');metadata.push({stage:name,tab:view.name,width:view.width,expectedStatus:capture.status,expectedCursor:capture.run.lastEventSeq,statusAgrees,cursorAgrees,explicitlyOffline,availability:statusAgrees?'status available and agrees':explicitlyOffline?'explicit offline during reconnect':'unexpected missing status',header:prefix.slice(0,650)});assert.ok(statusAgrees||(name==='restart-failed'&&explicitlyOffline),'Inspector status '+name+' '+view.name);assert.ok(cursorAgrees,'Inspector cursor '+name+' '+view.name);}
}
await writeFile(join(out,'ui-metadata-check.json'),JSON.stringify(metadata,null,2));
const links=[...readme.matchAll(/\]\(([^)]+)\)/g)].map(m=>decodeURIComponent(m[1]));
const unlinked=names.filter(n=>n!=='README.md'&&!links.includes(n)),missing=links.filter(n=>!names.includes(n));
const head=execFileSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8',windowsHide:true}).trim();
const changed=execFileSync('git',['diff','--name-only'],{cwd:root,encoding:'utf8',windowsHide:true}).trim();
const index=execFileSync('git',['diff','--cached','--name-only'],{cwd:root,encoding:'utf8',windowsHide:true}).trim();
const sizes=[];for(const n of names){const b=await readFile(join(out,n));sizes.push({name:n,bytes:b.length});}
const oversized=sizes.filter(x=>x.bytes>1000000);
const findings={head,trackedChanges:changed,indexChanges:index,files:names.length,links:links.length,unlinked,missing,oversized,uiMetadataChecks:metadata.length,statusAgreements:metadata.filter(x=>x.statusAgrees).length,explicitOfflineCaptures:metadata.filter(x=>x.explicitlyOffline).length,largest:sizes.sort((a,b)=>b.bytes-a.bytes).slice(0,4)};
assert.equal(head,'d7371f25b87ee4d474d1bb75258a851e3cde648f');assert.equal(changed,'');assert.equal(index,'');assert.deepEqual(unlinked,[]);assert.deepEqual(missing,[]);assert.deepEqual(oversized,[]);
await writeFile(join(out,'audit.log.txt'),JSON.stringify(findings,null,2)+'\n');
const lines=['Final artifact bytes and SHA-256; this manifest excludes itself.'];
for(const n of names){if(n==='manifest.log.txt')continue;const b=await readFile(join(out,n));lines.push(b.length+'\t'+createHash('sha256').update(b).digest('hex')+'\t'+n);}
await writeFile(join(out,'manifest.log.txt'),lines.join('\n')+'\n');console.log(JSON.stringify(findings));
