async function api(path,body){const r=await fetch(base+path,{headers:{authorization:'Bearer '+token,'content-type':'application/json'},...(body===undefined?{}:{method:'POST',body:JSON.stringify(body)}),signal:AbortSignal.timeout(120000)});if(!r.ok)throw new Error(path+' HTTP '+r.status);return r.json();}
function readClient(id,after){let last=after,count=0,gaps=0;const errors=[],opens=[];const sub=subscribeRunEvents({client:createGraphGoblinClient({baseUrl:base,apiKey:token}),runId:id,after,onEvent:e=>{if(e.seq!==last+1)gaps++;last=e.seq;count++;},onError:(e,a)=>errors.push({code:e.code,message:e.message,attempt:a}),onOpen:()=>opens.push(performance.now()),backoff:{initialMs:100,maxMs:500,maxAttempts:20}});return {close:()=>sub.close(),done:sub.done.then(()=>({last,count,gaps,errors,opens}),e=>({last,count,gaps,errors,rejected:String(e),opens}))};}
async function readFetch(id,after){const start=performance.now();const r=await fetch(base+'/runs/'+id+'/events?after='+after,{headers:{authorization:'Bearer '+token,accept:'text/event-stream'},signal:AbortSignal.timeout(180000)});const p=new SseParser(),d=new TextDecoder(),reader=r.body.getReader();let last=after,count=0,gaps=0;const controls=[];for(;;){const {value,done}=await reader.read();if(done)break;for(const f of p.push(d.decode(value,{stream:true}))){if(f.event==='stream.error'){controls.push(JSON.parse(f.data));continue;}const e=JSON.parse(f.data);if(e.seq!==last+1)gaps++;last=e.seq;count++;}}return {after,last,count,gaps,controls,totalMs:performance.now()-start};}

async function setup(width){const context=await browser.newContext({viewport:{width,height:900}});await context.addInitScript(({token})=>{localStorage.setItem('graphgoblin-api-key',token);window.__qa={longTasks:[],samples:[]};new PerformanceObserver(list=>{for(const t of list.getEntries())window.__qa.longTasks.push({start:t.startTime,duration:t.duration});}).observe({type:'longtask',buffered:true});},{token});const page=await context.newPage();page.setDefaultTimeout(180000);page.on('pageerror',e=>result.errors.push({width,error:e.message}));const cdp=await context.newCDPSession(page);await cdp.send('Performance.enable');return {context,page,cdp,width};}
async function inspect(tab,label,expected){const {page,cdp,width}=tab;const start=performance.now();let waitError;try{await page.waitForFunction(n=>document.querySelector('ol[aria-label="Timeline"]')?.children.length>=n,expected,{timeout:180000});}catch(e){waitError=e.message;}const actual=await page.evaluate(()=>{const list=document.querySelector('ol[aria-label="Timeline"]');const ids=[...(list?.children??[])].map(li=>Number(li.querySelector('button span')?.textContent?.replace('#','')));let gaps=0,duplicates=0;ids.forEach((id,i)=>{if(id!==(i?ids[i-1]+1:1))gaps++;if(i&&id<=ids[i-1])duplicates++;});const tasks=window.__qa.longTasks;return {count:ids.length,first:ids[0],last:ids.at(-1),gaps,duplicates,title:[...document.querySelectorAll('h1,h2,h3')].map(x=>x.textContent).filter(x=>x.includes('Timeline')),nodes:document.querySelectorAll('*').length,heap:performance.memory?{used:performance.memory.usedJSHeapSize,total:performance.memory.totalJSHeapSize}:null,longTasks:{count:tasks.length,totalMs:tasks.reduce((n,x)=>n+x.duration,0),maxMs:Math.max(0,...tasks.map(x=>x.duration)),entries:tasks},documentWidth:document.documentElement.scrollWidth,viewportWidth:innerWidth,alerts:[...document.querySelectorAll('[role="alert"]')].map(x=>x.textContent)};});const metrics=await cdp.send('Performance.getMetrics');const shot=label+'-'+width+'.png';await page.screenshot({path:join(out,shot)});const snapshot=await page.locator('nav').first().ariaSnapshot();await writeFile(join(out,label+'-'+width+'.snapshot.log.txt'),snapshot.length>100000?snapshot.slice(0,50000)+'\n[truncated intentionally]\n'+snapshot.slice(-50000):snapshot);console.log('QA_BROWSER_CHECK '+label+' '+width+' '+actual.count); const observation={label,width,expected,waitMs:performance.now()-start,waitError,...actual,cdp:Object.fromEntries(metrics.metrics.map(x=>[x.name,x.value])),screenshot:shot,pass:!waitError&&(label==='live-loading'?actual.count>=expected:actual.count===expected)&&actual.gaps===0&&actual.duplicates===0};result.observations.push(observation);await save();return observation;}
