function collector(after=0){return {after,count:0,last:after,gaps:0,duplicates:0,controls:[],hash:createHash('sha256'),firstMs:null,consume(frame,at){if(frame.event==='stream.error'){this.controls.push({problem:JSON.parse(frame.data),id:frame.id,atMs:at});return;}let e;try{e=JSON.parse(frame.data);}catch{this.partialFrames=(this.partialFrames??0)+1;return;}if(!Number.isInteger(e.seq))return;if(e.seq<=this.last)this.duplicates++;if(e.seq!==this.last+1)this.gaps++;if(this.firstMs===null)this.firstMs=at;this.count++;this.last=e.seq;this.hash.update(JSON.stringify(e)+'\n');},summary(ms){return {after:this.after,count:this.count,last:this.last,gaps:this.gaps,duplicates:this.duplicates,controls:this.controls,partialFrames:this.partialFrames??0,firstMs:this.firstMs,totalMs:ms,sha256:this.hash.digest('hex')};}};}
async function raw(runId,{after=0,pauseMs=0,paceMs=0,headers={},onProgress}={}){
 const start=performance.now(),c=collector(after),parser=new SseParser();
 return new Promise((resolve,reject)=>{let pauseTimer;const request=get(base+'/runs/'+runId+'/events?after='+after,{headers:{authorization:'Bearer '+token,accept:'text/event-stream',...headers}},res=>{let bytes=0;res.setEncoding('utf8');res.pause();pauseTimer=setTimeout(()=>res.resume(),pauseMs);res.on('data',chunk=>{bytes+=Buffer.byteLength(chunk);for(const frame of parser.push(chunk))c.consume(frame,performance.now()-start);onProgress?.(c);if(paceMs){res.pause();pauseTimer=setTimeout(()=>res.resume(),paceMs);}});const finish=(error)=>{clearTimeout(pauseTimer);resolve({...c.summary(performance.now()-start),bytes,status:res.statusCode,transportError:error});};res.once('end',()=>finish());res.once('error',e=>finish(String(e)));});request.on('error',reject);request.setTimeout(120000,()=>request.destroy(new Error('raw timeout')));});
}
async function fetched(runId,after=0){const start=performance.now(),c=collector(after),p=new SseParser();const res=await fetch(base+'/runs/'+runId+'/events?after='+after,{headers:{authorization:'Bearer '+token,accept:'text/event-stream'},signal:AbortSignal.timeout(120000)});const reader=res.body.getReader(),d=new TextDecoder();for(;;){const {value,done}=await reader.read();if(done)break;for(const f of p.push(d.decode(value,{stream:true})))c.consume(f,performance.now()-start);}return {...c.summary(performance.now()-start),status:res.status};}
function clientRead(runId,after=0){const start=performance.now(),c=collector(after),errors=[],opens=[];const client=createGraphGoblinClient({baseUrl:base,apiKey:token});const sub=subscribeRunEvents({client,runId,after,onEvent:e=>c.consume({data:JSON.stringify(e)},performance.now()-start),onOpen:()=>opens.push(performance.now()-start),onError:(e,attempt)=>errors.push({code:e.code??e.problem?.code,message:e.message,attempt}),backoff:{initialMs:50,maxMs:100,maxAttempts:5}});return sub.done.then(()=>({...c.summary(performance.now()-start),lastSeq:sub.lastSeq,errors,opens}),e=>({...c.summary(performance.now()-start),lastSeq:sub.lastSeq,errors,opens,rejected:{code:e.code??e.problem?.code,message:e.message}}));}
