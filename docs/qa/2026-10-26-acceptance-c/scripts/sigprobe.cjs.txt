const { spawn } = require('node:child_process');
const fs = require('node:fs');
const log = (m) => fs.appendFileSync(process.argv[2], `${new Date().toISOString()} ${m}\n`);
log(`probe pid ${process.pid}`);
process.on('SIGINT', () => log('probe got SIGINT'));
// A hidden, fully piped child (like ProcessScripts) and a piped child without windowsHide (like the Codex SDK).
const hidden = spawn(process.execPath, ['-e', "setTimeout(()=>{},30000)"], { stdio: ['pipe','pipe','pipe'], windowsHide: true });
const plain = spawn(process.execPath, ['-e', "setTimeout(()=>{},30000)"], { stdio: ['pipe','pipe','pipe'] });
log(`hidden child ${hidden.pid}, plain child ${plain.pid}`);
hidden.on('exit', (c, s) => log(`hidden child exit code=${c} signal=${s}`));
plain.on('exit', (c, s) => log(`plain child exit code=${c} signal=${s}`));
setTimeout(() => { log('probe done'); process.exit(0); }, 12000);
