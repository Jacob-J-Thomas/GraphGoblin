import { pathToFileURL } from 'node:url';
import path from 'node:path';
const root = process.argv[2];
const { createContainer, buildApp, loadConfig } = await import(pathToFileURL(path.join(root, 'apps/api/dist/index.js')));
const { FakeHarness } = await import(pathToFileURL(path.join(root, 'packages/engine/dist/testing/index.js')));
const config = loadConfig(process.env);
const owner = await createContainer(config, { harnesses: { codex: new FakeHarness([], 'codex'), claude: new FakeHarness([], 'claude') }, startTimers: false });
await owner.start();
const app = await buildApp(owner, { logger: false });
await app.listen({ host: config.host, port: config.port });
console.log(JSON.stringify({ ready: true, pid: process.pid, port: config.port }));
process.on('message', async message => {
 if(message === 'stop'){ await app.close(); await owner.stop(); console.log('STOPPED'); process.exit(0); }
});
