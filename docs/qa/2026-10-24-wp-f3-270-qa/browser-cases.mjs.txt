import {createRequire} from 'node:module';
import {resolve} from 'node:path';
import {writeFile} from 'node:fs/promises';
import assert from 'node:assert/strict';
const root=process.cwd(),out=resolve(root,'docs/qa/2026-10-24-wp-f3-270-qa'),base='http://127.0.0.1:5880';
const {chromium}=createRequire(resolve(root,'apps/web/package.json'))('@playwright/test');
const browser=await chromium.launch({channel:'msedge',headless:true});
const results=[];
async function api(path,data,method=data===undefined?'GET':'POST',headers={}){const r=await fetch(base+path,{method,headers:{'content-type':'application/json',...headers},...(data===undefined?{}:{body:JSON.stringify(data)})});const body=await r.json();assert(r.ok,JSON.stringify(body));return body;}
async function poll(fn,pred){for(let i=0;i<150;i++){let x=await fn();if(pred(x))return x;await new Promise(r=>setTimeout(r,100));}throw Error('poll expired');}
async function record(name,fn){try{results.push({name,status:'PASS',...await fn()})}catch(e){results.push({name,status:'FAIL',error:String(e),stack:e.stack})}await writeFile(resolve(out,'browser-results.json'),JSON.stringify(results,null,2));console.log(JSON.stringify(results.at(-1)));}
function def(name){return {schemaVersion:4,name,nodes:[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'},ui:{x:0,y:100}},{id:'wait',kind:'wait',label:'Wait',config:{mode:'input',prompt:'QA wait'},ui:{x:260,y:100}},{id:'done',kind:'exit',label:'Done',config:{},ui:{x:520,y:100}}],edges:[{id:'a',from:{node:'start',port:'out'},to:{node:'wait'}},{id:'b',from:{node:'wait',port:'out'},to:{node:'done'}}],settings:{maxNodeVisits:10,runDeadlineSeconds:null}};}
try{
for(const width of [1440,390]){
 const context=await browser.newContext({viewport:{width,height:1000}}),page=await context.newPage();page.setDefaultTimeout(15000);
 const errors=[];page.on('pageerror',e=>errors.push(String(e)));
 const created=await api('/loops',{definition:def('QA Edge v2 '+width)}),id=created.loop.id;
 const mode=()=>page.getByRole('combobox',{name:'Structured completion milliseconds mode'});
 const field=()=>page.getByRole('spinbutton',{name:'Structured completion milliseconds',exact:true});
 async function editor(){await page.goto(base+'/app/loops/'+id+'/edit');await page.getByRole('button',{name:/^(Show|Hide) loop settings$/}).waitFor();if(await page.getByRole('button',{name:'Show loop settings'}).isVisible())await page.getByRole('button',{name:'Show loop settings'}).click();await mode().waitFor();}
 await record('Edge-'+width+'-omitted-default-help-order',async()=>{
   await editor();assert.equal(await mode().inputValue(),'unlimited');await mode().scrollIntoViewIfNeeded();
   await page.screenshot({path:resolve(out,'loop-unlimited-'+width+'.png')});
   const body=await page.locator('body').innerText();await writeFile(resolve(out,'loop-'+width+'.log.txt'),body);
   const names=await page.locator('[role="form"],form').evaluateAll(xs=>xs.map(x=>x.innerText));
   const positions=await page.getByRole('combobox').evaluateAll(xs=>xs.map(x=>x.getAttribute('aria-label')));
   assert(positions.indexOf('Structured completion milliseconds mode')>positions.indexOf('Run deadline seconds mode'));
   assert(body.includes('No completion deadline; LLM decisions and repairs remain cancellable.'));
   return {loopId:id,positions,screenshot:'loop-unlimited-'+width+'.png',pageErrors:errors};
 });
 await record('Edge-'+width+'-keyboard-boundaries-save-reload',async()=>{
   await mode().focus();await page.keyboard.press('Home');await page.keyboard.press('Enter');assert.equal(await mode().inputValue(),'limited');assert.equal(await field().inputValue(),'120000');
   const rejections=[];
   for(const v of ['0','-1','0.5','31536000001']){await field().fill(v);await page.keyboard.press('Tab');await page.getByRole('alert').filter({hasText:/Too small|Too big|Invalid input/}).first().waitFor();const alerts=await page.getByRole('alert').allTextContents();assert.equal(await field().getAttribute('aria-invalid'),'true');rejections.push({value:v,alerts});}
   await field().scrollIntoViewIfNeeded();await page.screenshot({path:resolve(out,'invalid-'+width+'.png')});
   await field().fill('500');await page.keyboard.press('Tab');await poll(()=>api('/loops/'+id),x=>x.draft.definition.settings.structuredTimeoutMs===500);
   await page.reload();if(await page.getByRole('button',{name:'Show loop settings'}).isVisible())await page.getByRole('button',{name:'Show loop settings'}).click();assert.equal(await field().inputValue(),'500');
   await field().scrollIntoViewIfNeeded();await page.screenshot({path:resolve(out,'loop-limited-'+width+'.png')});
   return {rejections,saved:await api('/loops/'+id),screenshot:'loop-limited-'+width+'.png'};
 });
 await record('Edge-'+width+'-export-import-inspector-settings',async()=>{
   await page.goto(base+'/app/loops');const downloaded=page.waitForEvent('download');await page.getByRole('button',{name:'Export QA Edge v2 '+width,exact:true}).click();const download=await downloaded;await download.saveAs(resolve(out,'ui-export-'+width+'.json'));
   const exported=await api('/loops/'+id+'/export?draft=true');assert.equal(exported.loop.settings.structuredTimeoutMs,500);exported.loop.name='QA Edge v2 import '+width;const imported=await api('/loops/import',exported);assert.equal(imported.draft.definition.settings.structuredTimeoutMs,500);
   await api('/loops/'+id+'/publish',{});const {run}=await api('/loops/'+id+'/runs',{});await poll(()=>api('/runs/'+run.id),x=>x.status==='waiting');
   const edited=structuredClone(exported.loop);edited.name='QA Edge v2 '+width;edited.settings.structuredTimeoutMs=null;await api('/loops/'+id+'/draft',{definition:edited},'PUT');
   await page.goto(base+'/app/runs/'+run.id);await page.getByText('Structured completion: 500 ms per call.',{exact:true}).waitFor();await page.screenshot({path:resolve(out,'inspector-'+width+'.png'),fullPage:true});
   const inspected=await page.locator('body').innerText();await writeFile(resolve(out,'inspector-'+width+'.log.txt'),inspected);await api('/runs/'+run.id+'/cancel',{});
   await page.goto(base+'/app/settings');await page.getByText('No structured completion deadline unless the loop sets one.',{exact:true}).waitFor();await page.getByText('No structured completion deadline unless the loop sets one.',{exact:true}).scrollIntoViewIfNeeded();await page.screenshot({path:resolve(out,'settings-'+width+'.png')});
   const settings=await page.locator('body').innerText();await writeFile(resolve(out,'settings-'+width+'.log.txt'),settings);
   return {importedId:imported.loop.id,runId:run.id,export:exported,pin:(await api('/runs/'+run.id)).pinned.limits,pageErrors:errors};
 });
 await context.close();
}
}finally{await browser.close()}
