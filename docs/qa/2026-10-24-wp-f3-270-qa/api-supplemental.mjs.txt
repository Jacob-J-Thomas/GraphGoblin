async function api(path,body,method=body===undefined?'GET':'POST',headers={}) {
  const res=await fetch(base+path,{method,headers:{'content-type':'application/json',...headers},...(body===undefined?{}:{body:JSON.stringify(body)})});
  const raw=await res.text(); let data; try{data=JSON.parse(raw)}catch{data=raw}
  return {status:res.status,data};
}
async function ok(path,body,method,headers) { const r=await api(path,body,method,headers); assert(r.status<300,JSON.stringify(r));return r.data; }
async function poll(fn,predicate,ms=5000) { const start=Date.now();let v;do{v=await fn();if(predicate(v))return v;await sleep(10)}while(Date.now()-start<ms);throw new Error('poll timed out: '+JSON.stringify(v)); }
const terminal=r=>['succeeded','failed','cancelled'].includes(r.status);
const read=id=>ok('/runs/'+id);
async function snapshot(id) {return {run:await read(id),thread:await ok('/runs/'+id+'/thread'),events:await ok('/runs/'+id+'/events?limit=1000')};}
const terminalEvents=s=>s.events.items.filter(e=>['run.failed','run.finished','run.cancelled'].includes(e.type));
