import { createRequire } from 'node:module';
import { resolve } from 'node:path';
import { writeFile } from 'node:fs/promises';
const root=process.cwd(), out=resolve(root,'docs/qa/2026-10-24-wp-f3-270-qa');
const require=createRequire(resolve(root,'apps/web/package.json'));
const {chromium}=require('@playwright/test');
const browser=await chromium.launch({channel:'msedge',headless:true});
const context=await browser.newContext({viewport:{width:1440,height:1000}});
const page=await context.newPage();
const rows=await (await fetch('http://127.0.0.1:5880/loops?limit=100')).json();
console.log(JSON.stringify(rows.items?.map(x=>({id:x.id,name:x.name}))));
const item=rows.items.find(x=>x.name==='QA bound null');
await page.goto('http://127.0.0.1:5880/app/loops/'+item.id+'/edit');
await page.getByRole('button',{name:/^(Show|Hide) loop settings$/}).waitFor();
if(await page.getByRole('button',{name:'Show loop settings'}).isVisible())await page.getByRole('button',{name:'Show loop settings'}).click();
await writeFile(resolve(out,'browser-discovery.log.txt'),await page.locator('body').innerText());
await page.screenshot({path:resolve(out,'initial-1440.png'),fullPage:true});
console.log(await page.locator('body').innerText());
await browser.close();
