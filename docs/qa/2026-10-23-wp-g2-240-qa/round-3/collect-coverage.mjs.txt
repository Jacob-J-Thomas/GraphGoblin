import fs from 'node:fs';import path from 'node:path';
const root=process.cwd(),out=path.join(root,'docs/qa/2026-10-23-wp-g2-240-qa/round-3');
const gates=JSON.parse(fs.readFileSync(path.join(out,'gates.json'),'utf8')),packages={web:'apps/web',api:'apps/api',infrastructure:'packages/infrastructure','api-client':'packages/api-client'},results=[];
for(const [name,folder]of Object.entries(packages)){
 const gate=gates.find(g=>g.name===name+'-coverage');if(!gate)continue;
 const raw=fs.readFileSync(path.join(out,gate.log),'utf8'),log=raw.replace(/\u001b\[[0-9;]*m/g,'');
 const file=path.join(root,folder,'coverage/coverage-summary.json');let coverage=null,source=null;
 if(fs.existsSync(file)&&fs.statSync(file).mtimeMs>=fs.statSync(path.join(out,gate.log)).birthtimeMs){coverage=JSON.parse(fs.readFileSync(file,'utf8')).total;source='coverage-'+name+'-summary.json';fs.copyFileSync(file,path.join(out,source));}
 results.push({package:name,command:gate.command,exit:gate.code,ms:gate.ms,files:log.match(/^\s*Test Files\s+(.+)$/m)?.[1]??null,tests:log.match(/^\s*Tests\s+(.+)$/m)?.[1]??null,coverage,coverageIsComplete:gate.code===0,coverageSource:source,log:gate.log,failureHeadings:log.split('\n').filter(l=>/FAIL |\[vitest-pool\]|Unhandled Error/.test(l))});
}
fs.writeFileSync(path.join(out,'coverage-totals.json'),JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(results));
