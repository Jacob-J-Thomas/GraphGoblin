import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import {execFileSync} from 'node:child_process';

const head='987b70f72b71b132a19918978d03d8ea09bef040';
const clean=s=>String(s).replaceAll(process.env.USERPROFILE,'%USERPROFILE%').replaceAll(process.env.USERPROFILE.replaceAll('\\','/'),'%USERPROFILE%');
const git=(...args)=>execFileSync('git',args,{encoding:'utf8'});

fs.writeFileSync(path.join(out,'admission.json'),clean(JSON.stringify({head,branch:git('branch','--show-current').trim(),base:'74e78bb212a58af15d5c1792fa599ec8a6f4bbad',node:process.version,modules:fs.readdirSync('node_modules').length,initialGitStatus:git('status','--short'),at:new Date().toISOString()},null,2)));

fs.writeFileSync(path.join(out,'audit-original.mjs.txt'),clean(fs.readFileSync(path.join(os.tmpdir(),'gg-audit-g2-240-round1/repro-blockers.mjs'),'utf8')));

s=s.replaceAll('2026-10-23-wp-g2-240-qa/round-2','2026-10-23-wp-g2-240-qa/round-3').replaceAll('cc1c3fceca4f9ed914915fd1e2a2e2fbde6cecd1',head).replaceAll('gg-qa-g2-240-','gg-qa-g2-240-r3-').replaceAll('p=5820;p<=5829','p=5920;p<=5929');
