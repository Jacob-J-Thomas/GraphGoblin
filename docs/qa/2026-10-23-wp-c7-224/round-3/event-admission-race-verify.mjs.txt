function launch(args = []) {

  p.exit = new Promise((res, rej) => { child.once('error',rej); child.once('close',(code,signal) => { p.ended=true; p.result={code,signal}; res(p.result); }); });
  processes.push(p); return p;
}

  const nodes = [{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},...work,{id:'done',kind:'exit',label:'Done',config:exit}];
