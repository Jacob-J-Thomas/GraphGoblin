let g;
try {
  g=new Gateway(await mkdtemp(join(tmpdir(),'gg-qa-f4-265-extra-')),6044);await g.start();

    await g.control('plans',{plans:[{error:'HARNESS_TRANSPORT_FAILED',submission:'submitted',retriable:true,afterAccept:true}]});

    const failureSeq=(await g.events(run.id)).findLast(x=>x.type==='run.failed').seq;
    await g.api('/runs/'+run.id+'/recover-session',{nodeId:'worker',failureSeq});await g.status(run.id,'succeeded');
    const snap=await g.snapshot(run.id,'current-visit-'+family);
    assert.equal(snap.events.filter(x=>x.type==='run.resumed'&&x.freshSession).length,1);
    assert.deepEqual(g.calls.slice(before).map(x=>x.mode),['fresh','resume','fresh']);
    assert.equal(g.calls.at(-1).prompt,'Authored task: preserve exactly [QA-265]');
    return {runId:run.id,missing:failed.failure.missingSession,calls:g.calls.slice(before)};
  });

    const before=g.calls.length,def=graph('codex','resume-own');def.nodes.find(x=>x.id==='again').config.maxPasses=3;
    def.limits={...def.limits,maxNodeVisits:100};
    const run=await g.launch(def);let waiting=await g.status(run.id,'waiting');
    const old=g.calls[before].id;await unlink(join(g.data,'fake-provider',old));

    await g.api('/runs/'+run.id+'/recover-session',{nodeId:'worker',failureSeq});waiting=await g.status(run.id,'waiting');
    const fresh=g.calls.at(-1).id;await g.api('/runs/'+run.id+'/input',{waitId:waiting.waiting.waitId,input:'go'});await g.status(run.id,'succeeded');
    assert.deepEqual(g.calls.slice(before).map(x=>x.mode),['fresh','resume','fresh','resume']);assert.equal(g.calls.at(-1).id,fresh);
    return {runId:run.id,calls:g.calls.slice(before),snapshot:await g.snapshot(run.id,'own-policy-after-choice')};
  });
  await check('repair-missing-restarts-authored-task','Successful authored turn returns invalid schema; repair refuses missing handle; choose fresh','New conversation receives original authored task, then validates successfully',async()=>{
    const before=g.calls.length;
    await g.control('plans',{plans:[{finalText:'invalid'},{error:'HARNESS_SESSION_MISSING',submission:'session-missing'}]});
