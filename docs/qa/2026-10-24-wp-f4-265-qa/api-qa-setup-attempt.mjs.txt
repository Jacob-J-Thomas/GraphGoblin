async function until(fn, description) { const start = Date.now(); for (;;) { const value = await fn(); if (value) return value; assert.ok(Date.now() - start < 30000, description); await delay(35); } }

class Gateway {
  constructor(data, port) { this.data = data; this.port = port; this.calls = []; this.messages = []; }
  async start() {
    const env = Object.fromEntries(Object.entries(process.env).filter(([k]) => !k.toUpperCase().startsWith('GG_') && !['JEV_API_KEY', 'NODE_V8_COVERAGE'].includes(k)));
    Object.assign(env, { GG_DATA_DIR: this.data, GG_DB_URL: 'file:' + join(this.data, 'graphgoblin.db').replaceAll('\\', '/'), GG_PORT: String(this.port), GG_REQUIRE_API_KEY: 'true', GG_SWAGGER_UI: 'false', GG_MASTER_KEY: Buffer.alloc(32, 26).toString('base64'), GG_WEB_DIST: resolve(root, 'apps/web/dist'), GG_LOG_LEVEL: 'silent' });
    const fixture = join(out, 'gateway.mjs.txt');
    this.child = spawn(process.execPath, ['--input-type=module', '-e', "import fs from 'node:fs'; await import('data:text/javascript;base64,'+fs.readFileSync(process.argv[1]).toString('base64'))", fixture], { cwd: root, env, windowsHide: true, stdio: ['ignore', 'pipe', 'pipe', 'ipc'] });
    this.child.stdout.on('data', x => { this.log = (this.log ?? '') + x; });
    this.child.stderr.on('data', x => { this.log = (this.log ?? '') + x; });
    this.child.on('message', m => { this.messages.push(m); if (m.type === 'call') this.calls.push(m.call); });
    const ready = await until(() => { const m = this.messages.findLast(x => x.type === 'ready'); if (this.child.exitCode !== null) throw Error('Fixture exited: ' + this.log); return m; }, 'gateway ready');
    this.token = ready.token; this.reader = ready.reader;
    return this;
  }
  async control(type, rest = {}) { const requestId = Math.random().toString(); this.child.send({ type, requestId, ...rest }); return until(() => this.messages.find(x => x.requestId === requestId), type); }
  async stop() { if (this.child.exitCode !== null) return; const closed = once(this.child, 'close'); this.child.send({ type: 'stop' }); await closed; }
  async exit() { const closed = once(this.child, 'close'); this.child.send({ type: 'exit' }); await closed; }
  async restart() { this.messages = []; return this.start(); }
  async api(path, body, expected = 200, token = this.token) {
    const method = body === undefined ? 'GET' : 'POST';
    const r = await fetch('http://127.0.0.1:' + this.port + path, { method, headers: { authorization: 'Bearer ' + token, ...(body === undefined ? {} : { 'content-type': 'application/json' }) }, ...(body === undefined ? {} : { body: JSON.stringify(body) }) });
    const data = await r.json(); traffic.push({ port: this.port, method, path, body, status: r.status, response: data });
    if (expected !== null) assert.equal(r.status, expected, path + ' ' + JSON.stringify(data));
    return { status: r.status, data };
  }
  async run(id) { return (await this.api('/runs/' + id)).data; }
  async events(id) { return (await this.api('/runs/' + id + '/events?limit=1000')).data.items; }
  async status(id, status) { return until(async () => { const r = await this.run(id); return r.status === status && r; }, 'run ' + id + ' status=' + status); }
  async publish(definition) { const { data } = await this.api('/loops', { definition }, 201); await this.api('/loops/' + data.loop.id + '/publish', {}); return data.loop.id; }
  async launch(def) { const loop = await this.publish(def); return (await this.api('/loops/' + loop + '/runs', {}, 202)).data.run; }
  async snapshot(id, label) { const snapshot = { run: await this.run(id), events: await this.events(id), sessions: (await this.api('/runs/' + id + '/sessions')).data, thread: (await this.api('/runs/' + id + '/thread')).data }; await save(label + '.json', snapshot); return snapshot; }
}
const config = (family, policy, extra = {}) => ({ harness: family, model: family === 'claude' ? 'claude-opus-5-5' : 'gpt-6-luna', effort: 'low', session: { policy, ...extra }, harnessOptions: { sandbox: 'read-only', approval: 'never' }, prompt: { template: 'Authored task: preserve exactly [QA-265]' } });
function graph(family, policy, reset = false) {
  const def = singleNodeLoop('QA ' + family + ' ' + policy + (reset ? ' reset' : '') + ' ' + Math.random(), { id: 'worker', kind: 'inference', label: 'Worker', config: config(family, policy, reset ? { resetAfterVisits: 1 } : policy === 'resume-named' ? { key: 'qa-conversation-' + Math.floor(Math.random()*1e9) } : {}) });
  const edge = (id, node, port, to) => ({ id, from: { node, port }, to: { node: to } });
  def.nodes.push({ id: 'hold', kind: 'wait', label: 'Hold', config: { mode: 'input', prompt: 'QA remove fake provider file', exposeTo: ['api'] } });
  if (policy === 'resume-own') {
    def.nodes.push({ id: 'again', kind: 'decision', label: 'Again', config: { kind: 'expression', jsonata: '"repeat"', maxPasses: 2, routes: [{ id: 'repeat', label: 'Repeat', repeat: true }, { id: 'finish', label: 'Finish' }] } });
    def.edges = [edge('e1','start','out','worker'), edge('e2','worker','out','again'), edge('e3','again','repeat','hold'), edge('e4','hold','out','worker'), edge('e5','again','limit-reached','done'), edge('e6','again','finish','done')];
  } else {
    const warmConfig = config(family, policy === 'resume-named' ? 'resume-named' : 'fresh', policy === 'resume-named' ? { key: def.nodes[1].config.session.key } : {});
    warmConfig.prompt.template = 'Earlier unrelated provider context';
    def.nodes.push({ id: 'warm', kind: 'inference', label: 'Warm', config: warmConfig });
    def.edges = [edge('e1','start','out','warm'), edge('e2','warm','out','hold'), edge('e3','hold','out','worker'), edge('e4','worker','out','done')];
  }
  return def;
}
async function removed(g, family, policy, reset = false, plans = []) {
  await g.control('plans', { plans });
  const before = g.calls.length;
  const run = await g.launch(graph(family, policy, reset));
  const waiting = await g.status(run.id, 'waiting');
  const old = g.calls[before].id;
  await unlink(join(g.data, 'fake-provider', old));
  await g.api('/runs/' + run.id + '/input', { waitId: waiting.waiting.waitId, input: 'continue' });
  const failed = await g.status(run.id, policy === 'fresh' || reset ? 'succeeded' : 'failed');
  const history = await g.events(run.id);
  const failureSeq = history.findLast(x => x.type === 'run.failed')?.seq;
  return { run, old, failed, failureSeq, before, payload: { nodeId: 'worker', failureSeq } };
}
async function recover(g, c, concurrent = false) {
  const path = '/runs/' + c.run.id;
  await g.api(path + '/resume', {}, 409);
  const snapshot = await g.snapshot(c.run.id, 'before-' + c.run.id);
  assert.equal(snapshot.run.failure.code, 'HARNESS_SESSION_MISSING');
  assert.ok(snapshot.run.failure.message.includes(c.old));
  const announced = snapshot.events.filter(x => x.type === 'harness.session');
  assert.equal(announced.length, 1); assert.equal(announced[0].mode, 'fresh');
  if (concurrent) {
    const choices = await Promise.all([g.api(path + '/recover-session', c.payload, null), g.api(path + '/recover-session', c.payload, null)]);
    assert.deepEqual(choices.map(x => x.status).sort(), [200, 409]);
  } else await g.api(path + '/recover-session', c.payload);
  await g.status(c.run.id, 'succeeded');
  await g.api(path + '/recover-session', c.payload, 409);
  const after = await g.snapshot(c.run.id, 'after-' + c.run.id);
  const choices = after.events.filter(x => x.type === 'run.resumed');
  assert.equal(choices.length, 1);
  const choice = choices[0];
  assert.equal(choice.actor.kind, 'api-key'); assert.ok(choice.actor.id); assert.ok(choice.ts);
  assert.deepEqual(choice.freshSession, { ...snapshot.run.failure.missingSession, ...c.payload, priorContextCarried: false, prompt: 'Authored task: preserve exactly [QA-265]' });
  const calls = g.calls.slice(c.before);
  assert.deepEqual(calls.map(x => x.mode), ['fresh', 'resume', 'fresh']);
  assert.equal(calls[2].prompt, choice.freshSession.prompt);
  assert.equal(calls[2].model, choice.freshSession.model); assert.equal(calls[2].effort, choice.freshSession.effort);
  assert.notEqual(calls[2].id, c.old);
  assert.equal(after.events.filter(x => x.type === 'run.succeeded').length, 1);
  return { runId: c.run.id, failure: snapshot.run.failure, choice, calls, terminal: after.run.status, modes: after.events.filter(x => x.type === 'harness.session').map(x => x.mode) };
}
