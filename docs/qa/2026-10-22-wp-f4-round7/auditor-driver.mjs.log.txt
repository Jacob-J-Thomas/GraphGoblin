async function api(path, body) {

  const response = await fetch('http://127.0.0.1:5664'+path, body === undefined ? {} : {method:'POST', headers:{'content-type':'application/json'}, body:JSON.stringify(body)});

  return {status:response.status, value:await response.json()};

}

async function stopped(id) {

  const until = Date.now()+60000;

  while (Date.now()<until) { const {value:r}=await api('/runs/'+id); if(['failed','succeeded','cancelled'].includes(r.status))return r; await delay(20); }

  throw new Error('Run state deadline '+id);

}

function faultHarness(spec) {

  const calls=[], cancelled=[];

  const seed=spec.policy==='fresh'?0:1;

  const refuseAt=seed+2+(spec.laterRepair?1:0);

  const refuseCount=spec.repeated?2:1;

  let n=0;

  const shouldRefuse=()=> n>=refuseAt && n<refuseAt+refuseCount;

  const text=()=>n<=seed?'42':n<refuseAt?(n===refuseAt-1 && spec.laterRepair?'invalid second':'invalid first'):'42';

  const fake = new FakeHarness([], spec.family);

  const adapter = new CodexHarness({clientFactory:()=>{

    const answer=text();

    const thread=id=>({runStreamed:async()=>({events:(async function*(){

      yield {type:'thread.started',thread_id:id??('audit-thread-'+spec.name)};

      yield {type:'turn.started'};

      yield {type:'item.completed',item:{id:'answer',type:'agent_message',text:answer}};

      yield {type:'turn.completed',usage:{}};

    })()})});

    return {startThread:()=>thread(),resumeThread:id=>{if(shouldRefuse())throw Object.assign(new Error('Recorded session refused at resumeThread'),{code:spec.code,retriable:spec.retriable});return thread(id);}};

  }});

  function invoke(r,s,id) {

    n++; calls.push({id,prompt:r.turn.prompt});

    const own=n;

    let session;

    if(spec.mode==='sync' && spec.family==='codex') session=id?adapter.resume(id,r,s):adapter.start(r,s);

    else {

      if(spec.mode==='sync' && shouldRefuse())throw Object.assign(new Error('Controlled pre-launch refusal'),{code:spec.code,retriable:spec.retriable??true});

      fake.script([shouldRefuse()?{accepted:spec.accepted??false,announceSession:false,items:[],submission:spec.accepted?'delivery-uncertain':'definitely-not-submitted',error:{code:spec.code,message:'Controlled asynchronous refusal',retriable:spec.retriable??true}}:{finalText:text()}]);

      session=id?fake.resume(id,r,s):fake.start(r,s);

    }

    return {...session,cancel:async()=>{cancelled.push(own);return session.cancel();}};

  }

  return {calls,cancelled,refuseAt,start:(r,s)=>invoke(r,s),resume:(id,r,s)=>invoke(r,s,id)};

}

function definition(spec) {

  const session=spec.policy==='resume-named'?{policy:spec.policy,key:'audit-'+spec.name.split('-')[1]}:{policy:spec.policy};

  const config={harness:spec.family, model:spec.family==='claude'?'claude-opus-5-5':'gpt-6-luna',effort:'low',harnessOptions:{sandbox:'read-only',approval:'never'},session,

    prompt:{template:'Original task'},output:{schema:{jsonSchema:{type:'number'},repair:{enabled:true,maxAttempts:3,prompt:spec.large?'Repair exactly: {{ invocation.trigger.payload.blob }}{{ invocation.trigger.payload.blob }}{{ invocation.trigger.payload.blob }} / {{ value }}':'Repair exactly: {{ value }}\nquote " slash \\ NUL \u0000 LS \u2028 PS \u2029 emoji \u{1F642}'}}}};

  const d=singleNodeLoop(spec.name,{id:'infer',kind:'inference',label:'Infer',config});

  if(spec.policy!=='fresh'){

    const incoming=d.edges.find(e=>e.to.node==='infer');

    const source=incoming.from;

    incoming.from={node:'seed',port:'out'};

    d.nodes.push({id:'seed',kind:'inference',label:'Seed',config:{harness:spec.family,model:config.model,effort:'low',harnessOptions:{sandbox:'read-only',approval:'never'},session,prompt:{template:'Seed task'}}});

    d.edges.push({id:'seed-entry',from:source,to:{node:'seed',port:'in'}});

  }

  return d;

}
