import {readFile,writeFile,readdir,stat} from 'node:fs/promises';import {join,relative,resolve,sep} from 'node:path';import {createHash} from 'node:crypto';
const out=join(process.cwd(),'docs/qa/2026-10-24-wp-d4-343-qa/round-2');const items=JSON.parse(await readFile(join(out,'coverage-results.json'),'utf8'));const generated=[];
async function scan(root,path,pkg){const resolved=resolve(path);if(resolved!==root&&!resolved.startsWith(root+sep))throw Error('coverage boundary');for(const d of await readdir(path,{withFileTypes:true})){const file=join(path,d.name);if(d.isDirectory())await scan(root,file,pkg);else if(d.isFile()){const s=await stat(file);if(s.size>1000000){const bytes=await readFile(file);generated.push({name:'generated-'+pkg+'/'+relative(root,file).replaceAll('\\','/'),bytes:s.size,sha256:createHash('sha256').update(bytes).digest('hex'),destination:file.replaceAll(process.env.USERPROFILE,'%USERPROFILE%'),generated:true})}}}}
for(const i of items){const path=i.reportDir.replace('%USERPROFILE%',process.env.USERPROFILE);await scan(resolve(path),path,i.pkg)}
await writeFile(join(out,'generated-oversized.json'),JSON.stringify({completedPackages:items.map(i=>i.pkg),files:generated},null,2));console.log(JSON.stringify({completedPackages:items.length,oversizedGeneratedFiles:generated.length}));
