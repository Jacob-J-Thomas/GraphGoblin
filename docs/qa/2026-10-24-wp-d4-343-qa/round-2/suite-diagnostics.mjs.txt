import {readFile,writeFile} from 'node:fs/promises';import {join} from 'node:path';import {spawn,execFileSync} from 'node:child_process';
const root=process.cwd(),out=join(root,'docs/qa/2026-10-24-wp-d4-343-qa/round-2');
for(;;){try{if(JSON.parse(await readFile(join(out,'coverage-results.json'),'utf8')).length===6)break}catch{}await new Promise(r=>setTimeout(r,1000))}
const touched=new Set(execFileSync('git',['diff','--name-only','0a1ff340dc8e0eb0ad6621b4020d100dcdc3be71','HEAD'],{encoding:'utf8'}).trim().split(/\r?\n/));
const packages=JSON.parse(await readFile(join(out,'coverage-results.json'),'utf8'));const failures=[],jobs=[];const seen=new Set();
for(const item of packages){if(item.code===0)continue;const dir=['api','web'].includes(item.pkg)?'apps/'+item.pkg:'packages/'+item.pkg;const raw=await readFile(join(out,'coverage-'+item.pkg+'.log.txt'),'utf8');const log=raw.replace(/\x1b\[[0-9;]*m/g,'').replaceAll('\\','/');const re=/FAIL\s+(src\/[A-Za-z0-9_./-]+\.test\.tsx?)/g;const matches=[...log.matchAll(re)];
 for(let i=0;i<matches.length;i++){const file=matches[i][1],block=log.slice(matches[i].index,matches[i+1]?.index??log.length);const timing=/Test timed out|Hook timed out|timed out in|Timeout waiting for worker|timeout of \d+/i.test(block);const changed=touched.has(dir+'/'+file);const entry={pkg:item.pkg,file,changed,timing,permissionHint:/EPERM|EACCES|access.*denied/i.test(block),excerpt:block.slice(0,1600)};failures.push(entry);const key=item.pkg+':'+file;if(timing&&!changed&&!seen.has(key)){seen.add(key);jobs.push({pkg:item.pkg,dir,file})}}
}
await writeFile(join(out,'suite-observations.json'),JSON.stringify({failures,isolatedJobs:jobs},null,2));const results=[];
for(const job of jobs){const label='isolated-'+job.pkg+'-'+job.file.replaceAll('/','-').replace(/\.tsx?$/,'');const command='pnpm.cmd --filter @graphgoblin/'+job.pkg+' exec vitest run '+job.file+' --maxWorkers=1';console.log('START',label);let log='';const start=Date.now();const p=spawn('cmd.exe',['/d','/c',command],{cwd:root,env:process.env,windowsHide:true});p.stdout.on('data',d=>log+=d);p.stderr.on('data',d=>log+=d);const code=await new Promise(r=>{p.on('exit',r);p.on('error',e=>{log+=e.stack;r(-1)})});await writeFile(join(out,label+'.log.txt'),log.replaceAll(process.env.USERPROFILE,'%USERPROFILE%'));results.push({...job,command,code,ms:Date.now()-start,evidence:label+'.log.txt'});await writeFile(join(out,'isolated-results.json'),JSON.stringify(results,null,2));console.log('DONE',label,code)}
await writeFile(join(out,'isolated-results.json'),JSON.stringify(results,null,2));await writeFile(join(out,'diagnostics-complete.json'),JSON.stringify({completed:true,isolatedCount:jobs.length,allSequential:true},null,2));console.log('DIAGNOSTICS COMPLETE',jobs.length);
