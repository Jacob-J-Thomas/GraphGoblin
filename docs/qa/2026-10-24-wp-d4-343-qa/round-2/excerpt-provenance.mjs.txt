import {readFile,writeFile} from 'node:fs/promises';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {join} from 'node:path';
const out=join(process.cwd(),'docs/qa/2026-10-24-wp-d4-343-qa/round-2');const manifest=JSON.parse(await readFile(join(out,'oversized-captures.json'),'utf8'));const entry=manifest.find(e=>e.name==='diagnostics-results.json');assert.ok(entry);const bytes=await readFile(entry.destination.replace('%USERPROFILE%',process.env.USERPROFILE));const hash=createHash('sha256').update(bytes).digest('hex');assert.equal(bytes.length,entry.bytes);assert.equal(hash,entry.sha256);const full=JSON.parse(bytes.toString('utf8')),excerpt=JSON.parse(await readFile(join(out,'diagnostics-summary.json'),'utf8'));assert.deepEqual(excerpt.checks,full.checks);assert.deepEqual(excerpt.sweepFacts,full.sweepFacts);assert.deepEqual(excerpt.http,full.http.filter(e=>!e.path.startsWith('/events?')));const result={sourceCaptureBytes:bytes.length,sourceCaptureSha256:hash,allChecksPreserved:full.checks.length,allSweepFactsPreserved:true,retainedHttpResponses:excerpt.http.length,omittedRepeatedHistoryResponses:full.http.length-excerpt.http.length,status:'PASS'};await writeFile(join(out,'excerpt-provenance.json'),JSON.stringify(result,null,2));console.log(JSON.stringify(result));
