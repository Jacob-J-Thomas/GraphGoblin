import {readFile,writeFile} from 'node:fs/promises';
import {join} from 'node:path';
import assert from 'node:assert/strict';
const out=join(process.cwd(),'docs/qa/2026-10-24-wp-d4-343-qa/round-3');
const ready=JSON.parse(await readFile(join(out,'browser-ready.json'),'utf8'));
const auth=JSON.parse(await readFile(ready.authFile.replace('%USERPROFILE%',process.env.USERPROFILE),'utf8'));
const results={checks:[],http:[]};
const call=async(method,path,body)=>{const r=await fetch(auth.url+path,{method,headers:{authorization:'Bearer '+auth.key,...(body!==undefined?{'content-type':'application/json'}:{})},...(body!==undefined?{body:JSON.stringify(body)}:{})});const b=await r.json();if(path!='/__qa')results.http.push({method,path,status:r.status,body:b});return {status:r.status,body:b}};
const sql=async(sql,args=[])=>{const r=await call('POST','/__qa',{op:'sql',sql,args});assert.equal(r.status,200);return r.body};
const qa=async(op,more={})=>{const r=await call('POST','/__qa',{op,...more});assert.equal(r.status,200);return r.body};
const test=async(id,fn)=>{const c={id};try{c.actual=await fn();c.status='PASS'}catch(e){c.status='FAIL';c.error=e.stack}results.checks.push(c);await writeFile(join(out,'sweep-reproduction.json'),JSON.stringify(results,null,2));console.log(id,c.status,c.error??'')};
// Controlled local SQLite fault proves same-ledger behavior separately from webhook isolation.
await test('same-ledger-sweep-metadata-fault',async()=>{
 const def=name=>({schemaVersion:4,name,nodes:[{id:'start',kind:'trigger',label:'Start',config:{type:'event',eventType:'diag'}},{id:'done',kind:'exit',label:'Done',config:{}}],edges:[{id:'e',from:{node:'start',port:'out'},to:{node:'done'}}]});
 const loops=[];for(const name of ['Sweep repeated bad','Sweep repeated healthy']){const r=await call('POST','/loops',{definition:def(name)});assert.equal(r.status,201);loops.push(r.body.loop.id);assert.equal((await call('POST','/loops/'+r.body.loop.id+'/publish')).status,200)}
 await sql(`CREATE TRIGGER qa_diag_all BEFORE INSERT ON runs WHEN NEW.loop_id IN ('${loops[0]}','${loops[1]}') BEGIN SELECT RAISE(ABORT,'QA fault'); END`);
 const receipts=[];for(let i=0;i<2;i++){receipts.push((await call('POST','/events',{loopId:loops[i],type:'diag',dedupeKey:'diagnostic-'+i})).body);await qa('settle')}
 await sql('DROP TRIGGER qa_diag_all');
 await sql(`CREATE TRIGGER qa_diag_bad BEFORE INSERT ON runs WHEN NEW.loop_id='${loops[0]}' BEGIN SELECT RAISE(ABORT,'QA fault'); END`);
 await sql(`CREATE TRIGGER qa_diag_meta BEFORE UPDATE ON event_receipts WHEN OLD.inbound_id='${receipts[0].id}' BEGIN SELECT RAISE(ABORT,'QA metadata fault'); END`);
 const before=await sql('SELECT inbound_id,status,attempts,next_attempt_at FROM event_receipts WHERE loop_id IN (?,?)',loops);
 const after=[];for(let i=0;i<3;i++){await qa('sweep',{ms:300001});await qa('settle');after.push(await sql('SELECT inbound_id,status,attempts,next_attempt_at FROM event_receipts WHERE loop_id IN (?,?)',loops));}assert.ok(after.every(rows=>rows.find(r=>r.inbound_id===receipts[0].id).status==='pending'&&rows.find(r=>r.inbound_id===receipts[0].id).attempts===1));assert.equal(after[0].find(r=>r.inbound_id===receipts[1].id).status,'admitted');
 await sql('DROP TRIGGER qa_diag_meta');
 // Ordinary per-receipt failure must allow the healthy recipient through.
 await qa('sweep',{ms:300001});await qa('settle');const ordinary=await sql('SELECT inbound_id,status,attempts FROM event_receipts WHERE loop_id IN (?,?)',loops);
 await sql('DROP TRIGGER qa_diag_bad');await qa('sweep',{ms:300001});await qa('settle');const recovered=await sql('SELECT inbound_id,status,attempts FROM event_receipts WHERE loop_id IN (?,?)',loops);assert.ok(recovered.every(r=>r.status==='admitted'));
 results.sweepFacts={loops,receipts,before,after,ordinary,recovered};
 assert.equal(ordinary.find(r=>r.inbound_id===receipts[1].id).status,'admitted');
 return results.sweepFacts;
});
await writeFile(join(out,'sweep-reproduction.json'),JSON.stringify(results,null,2));
