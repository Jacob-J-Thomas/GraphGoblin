import {readFile,writeFile,mkdtemp} from 'node:fs/promises';import {join} from 'node:path';import {tmpdir} from 'node:os';import {spawn} from 'node:child_process';
const root=process.cwd(),out=join(root,'docs/qa/2026-10-24-wp-d4-343-qa/round-4');
const clean=s=>s.replaceAll(process.env.USERPROFILE,'%USERPROFILE%');
const results=[];const coverage=await mkdtemp(join(tmpdir(),'gg-qa-d4-343-4-coverage-'));
async function run(name,command,cwd=root){console.log('START '+name);const start=Date.now();let output='';const p=spawn('cmd.exe',['/d','/c',command],{cwd,env:process.env,windowsHide:true});p.stdout.on('data',d=>output+=d);p.stderr.on('data',d=>output+=d);const code=await new Promise(r=>{p.on('exit',r);p.on('error',e=>{output+=e.stack;r(-1)})});await writeFile(join(out,name+'.log.txt'),clean(output));const r={name,command:clean(command),cwd:clean(cwd),code,ms:Date.now()-start,lock:output.includes('ERR_PNPM_STORE_DIR_OPEN_OPERATION_LOCK')};results.push(r);await writeFile(join(out,'gates-results.json'),JSON.stringify(results,null,2));console.log('DONE '+name+' '+code);return r;}
for(const [name,command,fallback] of [['typecheck','pnpm.cmd typecheck','node_modules\\.bin\\turbo.cmd run typecheck'],['lint','pnpm.cmd lint','node_modules\\.bin\\turbo.cmd run lint'],['check-layers','pnpm.cmd check:layers','node tooling/scripts/check-layers.mjs'],['check-deps','pnpm.cmd check:deps','node_modules\\.bin\\depcruise.cmd --config .dependency-cruiser.cjs packages apps'],['check-docs','pnpm.cmd check:docs','node tooling/scripts/gen-docs.mjs --check'],['check-tokens','pnpm.cmd check:tokens','node tooling/scripts/check-design-tokens.mjs'],['check-migrations','pnpm.cmd check:migrations','node tooling/scripts/check-migrations.mjs']]){const r=await run(name,command);if(r.lock)await run(name+'-fallback',fallback)}
for(const pkg of ['domain','engine','api-client','web','infrastructure','api']){
 const directory=(pkg==='web'||pkg==='api'?'apps/':'packages/')+pkg;const reportDir=join(coverage,pkg);const args='--maxWorkers=1 --coverage.reportOnFailure --coverage.reportsDirectory="'+reportDir+'"';
 const r=await run('coverage-'+pkg,'pnpm.cmd --filter @graphgoblin/'+pkg+' test:coverage '+args);
 if(r.lock)await run('coverage-'+pkg+'-fallback','..\\..\\node_modules\\.bin\\vitest.cmd run --coverage '+args,join(root,directory));
 try{const summary=JSON.parse(await readFile(join(reportDir,'coverage-summary.json'),'utf8'));await writeFile(join(out,'coverage-'+pkg+'-summary.json'),JSON.stringify(summary.total,null,2));r.coverage=summary.total}catch(e){r.summaryUnavailable=clean(String(e))}await writeFile(join(out,'gates-results.json'),JSON.stringify(results,null,2));
}
await writeFile(join(out,'gates-finished.json'),JSON.stringify({complete:true,coverage:clean(coverage)},null,2));
