import {readFile,writeFile,readdir,stat,copyFile} from 'node:fs/promises';import {join} from 'node:path';import {spawn,execFileSync} from 'node:child_process';import {createHash} from 'node:crypto';import assert from 'node:assert/strict';
const root=process.cwd(),relative='docs/qa/2026-10-24-wp-d4-343-qa/round-4',out=join(root,relative);const loc=JSON.parse(await readFile(join(out,'script-location.json'),'utf8'));const scripts=loc.runtime.replace('%USERPROFILE%',process.env.USERPROFILE);const clean=s=>s.replaceAll(process.env.USERPROFILE,'%USERPROFILE%').replaceAll(process.env.USERPROFILE.replaceAll('\\','/'),'%USERPROFILE%');
async function command(name,cmd){let output='';const p=spawn('cmd.exe',['/d','/c',cmd],{cwd:root,env:process.env,windowsHide:true});p.stdout.on('data',d=>output+=d);p.stderr.on('data',d=>output+=d);const code=await new Promise(r=>{p.once('exit',r);p.once('error',e=>{output+=e.stack;r(-1)})});await writeFile(join(out,name+'.log.txt'),clean(output));return {command:cmd,code,lock:output.includes('ERR_PNPM_STORE_DIR_OPEN_OPERATION_LOCK')};}
async function script(name){await copyFile(join(out,name+'.mjs.txt'),join(scripts,name+'.mjs'));const p=spawn(process.execPath,[join(scripts,name+'.mjs')],{cwd:root,env:process.env,windowsHide:true,stdio:'inherit'});const code=await new Promise(r=>p.once('exit',r));assert.equal(code,0,name);}
while(true){try{await readFile(join(out,'isolated-finished.json'));break}catch{await new Promise(r=>setTimeout(r,1000))}}
for(const name of ['MANIFEST.json','final-verification.json','format-result.json','format-final-result.json'])await writeFile(join(out,name),'{}\n');
for(const name of ['format-check.log.txt','format-check-final.log.txt','report-format.log.txt','report-format-final.log.txt'])await writeFile(join(out,name),'pending\n');
await script('normalize');await script('report-generator');
let formatted=await command('report-format','pnpm.cmd exec prettier --write '+relative+'/README.md '+relative+'/NOT-KEPT.md');if(formatted.lock)formatted=await command('report-format-fallback','node_modules\\.bin\\prettier.cmd --write '+relative+'/README.md '+relative+'/NOT-KEPT.md');
let format=await command('format-check','pnpm.cmd format:check');if(format.lock)format=await command('format-check-fallback','node_modules\\.bin\\prettier.cmd --check .');await writeFile(join(out,'format-result.json'),JSON.stringify(format,null,2));
await script('report-generator');
formatted=await command('report-format-final','pnpm.cmd exec prettier --write '+relative+'/README.md '+relative+'/NOT-KEPT.md');if(formatted.lock)await command('report-format-final-fallback','node_modules\\.bin\\prettier.cmd --write '+relative+'/README.md '+relative+'/NOT-KEPT.md');
const finalFormat=await command('format-check-final','pnpm.cmd format:check');await writeFile(join(out,'format-final-result.json'),JSON.stringify(finalFormat,null,2));
await script('normalize');
const readme=await readFile(join(out,'README.md'),'utf8');const names=(await readdir(out)).sort();const unlinked=[];const overLimit=[];const pathLeaks=[];
for(const name of names){const p=join(out,name);const info=await stat(p);assert.ok(info.isFile(),'unexpected nested evidence directory');if(info.size>1000000)overLimit.push(name);if(name!=='README.md'&&!readme.includes(']('+name+')'))unlinked.push(name);if(/\.(md|txt|json)$/.test(name)){const text=await readFile(p,'utf8');if(text.includes(process.env.USERPROFILE)||text.includes(process.env.USERPROFILE.replaceAll('\\','/'))||text.includes(process.env.USERPROFILE.replaceAll('\\','\\\\')))pathLeaks.push(name)}}
const git=args=>execFileSync('git',args,{cwd:root,encoding:'utf8',windowsHide:true}).trim();const head=git(['rev-parse','HEAD']);const productionChanges=git(['diff','--name-only','HEAD','--','.',':!docs/qa']);const staged=git(['diff','--cached','--name-only']);
const verification={head,expectedHead:head==='ad9d9f15f0ab65f19504863f39c759fdbe1e6527',branch:git(['branch','--show-current']),productionChanges,staged,unlinked,overLimit,pathLeaks,finalFormat,node:process.version,retainedFiles:names.length,gitStatus:git(['status','--short']),completed:new Date().toISOString()};
await writeFile(join(out,'final-verification.json'),JSON.stringify(verification,null,2));
const files=[];for(const name of names){if(name==='MANIFEST.json')continue;const bytes=await readFile(join(out,name));files.push({file:name,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')})}await writeFile(join(out,'MANIFEST.json'),JSON.stringify({head,files},null,2));
assert.equal(verification.expectedHead,true);assert.equal(productionChanges,'');assert.equal(staged,'');assert.deepEqual(unlinked,[]);assert.deepEqual(overLimit,[]);assert.deepEqual(pathLeaks,[]);console.log(JSON.stringify(verification,null,2));
