def main():
 repo=Path(sys.argv[1]);out=Path(sys.argv[2]);tier=sys.argv[3];edge=Path(sys.argv[4]);assert edge.is_file();out.mkdir(exist_ok=False)

 counter=0;ready={};opened=False;browser_closed=False;success=False;checks={}
 def pw(*args,private=False):
  nonlocal counter
  counter+=1
  command=shlex.join([PW,'-s='+session,*map(str,args)])
  result=subprocess.run([BASH,'--noprofile','--norc','-c',command],cwd=browserout,env=env,capture_output=True,timeout=70,creationflags=subprocess.CREATE_NO_WINDOW)
  data=result.stdout+result.stderr;name=f'{counter:02d}-{args[0]}'+('.private' if private else '')+'.log';(browserout/name).write_bytes(data)
  text=data.decode('utf-8',errors='replace')
  if result.returncode or '### Error' in text:raise RuntimeError(name+': '+text[-1600:])
  return text
 def snap():
  text=pw('snapshot',private=True)
  if '```yaml\n' in text:return text.split('```yaml\n',1)[1].split('\n```',1)[0]
  match=re.search(r'\[Snapshot\]\((.*?)\)',text)
  if not match:raise RuntimeError('Missing snapshot output')
  return (browserout/match[1]).read_text(encoding='utf-8')
 def ref(text,label,role='button'):
  for line in text.splitlines():
   if role+' '+json.dumps(label,ensure_ascii=False) in line:
    match=re.search(r'\[ref=([^\]]+)\]',line)
    if match:return match[1]
  raise RuntimeError('No fresh ref for '+label)
 def value(text):
  raw=text.split('### Result\n',1)[-1].lstrip()
  return json.JSONDecoder().raw_decode(raw)[0]
 stdout=(out/'fixture.stdout.log').open('wb');stderr=(out/'fixture.stderr.log').open('wb')
 worker=subprocess.Popen([str(NODE),str(FIXTURE),str(repo),str(out),'serve',tier],cwd=ROOT,env=env,stdout=stdout,stderr=stderr,creationflags=subprocess.CREATE_NO_WINDOW)
 try:

  ready=json.loads((out/'READY.private.json').read_bytes())

  pw('resize','1440','1100');pw('tracing-start')

   text=snap()
   # Bounded condition observation, no timing-based claim or test retry.
   for attempt in range(3):
    if ready['pinnedLabel'] in text and 'Pinned 525 Work' in text:break
    text=snap()
   assert ready['pinnedLabel'] in text and version in text and 'Pinned 525 Work' in text,text[-3000:]
   assert 'Pinned definition unavailable' not in text and 'Could not load run' not in text
   (browserout/(label+'-loaded.yml')).write_text(text,encoding='utf-8')

   pw('click',ref(text,'Pinned definition JSON'))
   expanded=snap();(browserout/(label+'-expanded.yml')).write_text(expanded,encoding='utf-8')
   definition=value(pw('eval',"() => JSON.parse(document.querySelector('[aria-label=\"Pinned definition JSON\"]').textContent)"))
   assert definition['nodes'][1]['config']['harness']=='codex'
   assert 'model' not in definition['nodes'][1]['config'] and 'effort' not in definition['nodes'][1]['config']
   assert definition['nodes'][1]['label']=='Pinned 525 Work'
   rendered[label]=definition;(browserout/(label+'-definition.json')).write_text(json.dumps(definition,indent=2,ensure_ascii=False)+'\n',encoding='utf-8')
   pw('screenshot','--filename='+str(browserout/(label+'-definition.png')),'--full-page')
   requests=pw('requests');(browserout/(label+'-requests.log')).write_text(requests,encoding='utf-8')
   path=f"{ready['address']}/loops/{ready['loopId']}/versions/{version}"
   matches=re.findall(r'(?m)^(\d+)\. \[GET\] '+re.escape(path)+r' => \[200\]',requests)
   assert matches,'Pinned GET 200 not in actual browser request list'
   number=matches[-1];headers=pw('response-headers',number);body=pw('response-body',number)
   response=json.loads(body);assert response['id']==version and response['definition']==definition
   requestid=re.search(r'(?im)^x-qa-request-id:\s*(\S+)',headers);assert requestid
   network[label]={'requestIndex':int(number),'url':path,'status':200,'observerRequestId':requestid[1]}
   (browserout/(label+'-version-response.json')).write_text(json.dumps(response,indent=2,ensure_ascii=False)+'\n',encoding='utf-8')
  assert rendered['omitted']==rendered['materialized']
  console=pw('console','error');(browserout/'console-errors.log').write_text(console,encoding='utf-8')
  # The fresh profile intentionally exercises authentication; only its pre-auth HTTP401 resource messages are expected.
  errors=[line for line in console.splitlines() if '[ERROR]' in line]
  assert all('401' in line and 'Failed to load resource' in line for line in errors),errors
  pw('tracing-stop');checks={'bothInspectorPagesRendered':True,'pinnedLabel':ready['pinnedLabel'],'threeNodeLabelsRendered':True,'pinnedDefinitionJsonCodex':True,'materializedUiEquivalent':True,'absentModelEffortPreserved':True,'network':network,'returnedConsoleErrorEntries':len(errors),'consoleScope':'current materialized page; prior authentication/navigation logs retained separately','profileOwned':True,'session':session,'browserChannel':'msedge','evidenceTier':tier};success=True
 finally:
  if opened:

  (out/'FINALIZE.request').write_text(tier+' browser wrapper finished\n',encoding='utf-8')

 if not checks['success']:raise RuntimeError('Browser/fixture check or cleanup failed; retain receipts')
 print(json.dumps({k:v for k,v in checks.items() if k!='fixtureFinal'}));return 0
if __name__=='__main__':sys.exit(main())
