// Cross-check the previous keyboard hand-off guarantee when the new message window removes
// an earlier expanded value. Public API fixtures and normal BoundedValue controls only.
import {createRequire} from 'node:module';import {resolve} from 'node:path';import {readFile,writeFile} from 'node:fs/promises';
const {chromium}=createRequire(resolve('apps/web/package.json'))('@playwright/test'),dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-4'),api='http://127.0.0.1:5720';
const messages=JSON.parse(await readFile(resolve(dir,'messages-api.json.txt'),'utf8'));messages[0].content='F'.repeat(20000);
async function call(path,body){const r=await fetch(api+path,{method:body===undefined?'GET':'POST',headers:{'content-type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const v=await r.json();if(!r.ok)throw Error(r.status+' '+path);return v;}
async function wait(id,node){for(let i=0;i<1200;i++){const r=await call('/runs/'+id);if(r.waiting?.nodeId===node)return r;await new Promise(ok=>setTimeout(ok,50));}throw Error('wait timeout');}
const browser=await chromium.launch({channel:'msedge',headless:true}),page=await browser.newPage();const results=[];
page.setDefaultTimeout(20000);const errors=[];page.on('pageerror',e=>errors.push(e.message));
for(const width of [1440,390]){
 const nodes=[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},{id:'seed',kind:'mutate',label:'Seed',config:{operations:[{op:'set',path:'/messages',value:{kind:'literal',value:messages}}]}},{id:'ask',kind:'wait',label:'Ask',config:{mode:'input',prompt:'Continue?'}},{id:'append',kind:'mutate',label:'Append',config:{operations:[{op:'append-message',role:'note',content:'live appended message'}]}},{id:'next',kind:'wait',label:'Next',config:{mode:'input',prompt:'Continue again?'}},{id:'done',kind:'exit',label:'Done',config:{}}].map((n,i)=>({...n,ui:{x:i*260,y:80}}));
 const definition={schemaVersion:4,name:'QA R4 focused old message '+width+' '+Date.now(),nodes,edges:nodes.slice(1).map((n,i)=>({id:'e'+i,from:{node:nodes[i].id,port:'out'},to:{node:n.id}}))};
 const {loop}=await call('/loops',{definition});await call('/loops/'+loop.id+'/publish',{});const {run}=await call('/loops/'+loop.id+'/runs',{input:{}});const current=await wait(run.id,'ask');
 await page.setViewportSize({width,height:width===390?844:900});await page.goto(api+'/app/runs/'+run.id);await page.getByRole('list',{name:'Timeline',exact:true}).getByRole('button',{name:/run\.waiting ask/}).waitFor();
 await page.getByRole('button',{name:'Show all messages',exact:true}).click();await page.getByRole('button',{name:'Show all of Message m0 content',exact:true}).click();await page.keyboard.press('Shift+Tab');
 const before=await page.evaluate(()=>({tag:document.activeElement.tagName,label:document.activeElement.getAttribute('aria-label'),role:document.activeElement.getAttribute('role'),messages:document.querySelectorAll('ol[aria-label="Messages"] > li').length}));if(before.role!=='region'||before.label!=='Message m0 content')throw Error('old message region not focused');
 await page.evaluate(()=>{window.__focus=[];document.addEventListener('focusin',e=>window.__focus.push({tag:e.target.tagName,label:e.target.getAttribute('aria-label')}),true);});
 await call('/runs/'+run.id+'/input',{waitId:current.waiting.waitId,input:{yes:true}});await wait(run.id,'next');await page.getByText('Showing the last 50 of 6,002 messages.',{exact:true}).waitFor();await page.evaluate(()=>new Promise(ok=>requestAnimationFrame(()=>requestAnimationFrame(ok))));
 const after=await page.evaluate(()=>({tag:document.activeElement.tagName,label:document.activeElement.getAttribute('aria-label'),messages:document.querySelectorAll('ol[aria-label="Messages"] > li').length,oldRegionPresent:!!document.querySelector('pre[aria-label="Message m0 content"]'),width:document.documentElement.scrollWidth,viewport:innerWidth,trace:window.__focus}));
 await page.keyboard.press('Tab');const afterTab=await page.evaluate(()=>({tag:document.activeElement.tagName,label:document.activeElement.getAttribute('aria-label'),text:document.activeElement.tagName==='BUTTON'?document.activeElement.textContent:undefined}));
 const thread=await call('/runs/'+run.id+'/thread');const screenshot='message-focus-after-tab-'+width+'.png';await page.getByRole('button',{name:'Show all messages',exact:true}).scrollIntoViewIfNeeded();await page.screenshot({path:resolve(dir,screenshot)});
 const row={width,runId:run.id,system:'message window / BoundedValue keyboard focus',steps:'Expand all 6000 messages; expand Message m0; Shift+Tab into its named region; external wait input appends two messages and resets window',expected:'Removed focused old region hands keyboard to a surviving meaningful control; BODY does not retain it',before,after,afterTab,storedMessages:thread.messages.length,storedOldValueCharacters:thread.messages[0].content.length,screenshot,pass:after.tag!=='BODY'&&after.messages===50&&thread.messages.length===6002};results.push(row);await writeFile(resolve(dir,'message-focus-results.json'),JSON.stringify({results,errors},null,2));console.log(JSON.stringify(row));
}
await browser.close();process.exitCode=results.some(x=>!x.pass)?1:0;
