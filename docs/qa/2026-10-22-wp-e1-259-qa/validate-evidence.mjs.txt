import {readFile,writeFile,readdir} from 'node:fs/promises';import {resolve} from 'node:path';import {homedir} from 'node:os';import {createHash} from 'node:crypto';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa'),home=homedir();const files=await readdir(dir),redacted=[];
for(const name of files.filter(x=>/\.(txt|md|json)$/.test(x))){const file=resolve(dir,name),raw=await readFile(file),utf16=raw[0]===255&&raw[1]===254,before=utf16?raw.subarray(2).toString('utf16le'):raw.toString('utf8');let after=before;for(const p of [home,home.replaceAll('\\','/'),JSON.stringify(home).slice(1,-1),home.slice(2),home.replaceAll('\\','/').slice(2)])after=after.replaceAll(p,'<USER>');after=after.replace(/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/gi,'<USER>');if(after!==before||utf16){await writeFile(file,after);redacted.push(name);}}
const report=await readFile(resolve(dir,'README.md'),'utf8'),links=[...report.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(x=>decodeURIComponent(x[1]));const missing=[];for(const target of links){if(target.startsWith('http'))continue;try{await readFile(resolve(dir,target));}catch{missing.push(target);}}
const unlinked=files.filter(x=>x!=='README.md'&&!links.includes(x));
const apiThread=JSON.parse(await readFile(resolve(dir,'mutate-thread-before.json'),'utf8')),apiRun=JSON.parse(await readFile(resolve(dir,'inference-run-before.json'),'utf8')),fixture=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8'));
const sort=x=>x&&typeof x==='object'?Array.isArray(x)?x.map(sort):Object.fromEntries(Object.keys(x).sort().map(k=>[k,sort(x[k])])):x,hash=x=>createHash('sha256').update(x).digest('hex');const downloads=[];
for(const width of [1440,390])for(const [stem,value]of [['result',apiRun.result],['variables',apiThread.vars],['event',fixture.event.payload],['thread',apiThread]]){const raw=await readFile(resolve(dir,stem+'-'+width+'.json')),expected=Buffer.from(JSON.stringify(value,null,2));downloads.push({file:stem+'-'+width+'.json',bytes:raw.length,sha256:hash(raw),apiSha256:hash(expected),byteEqual:raw.equals(expected),semanticEqual:JSON.stringify(sort(JSON.parse(raw)))===JSON.stringify(sort(value))});}
