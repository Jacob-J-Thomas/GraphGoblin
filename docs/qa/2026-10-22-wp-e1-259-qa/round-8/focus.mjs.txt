import {resolve} from 'node:path';import {dir,call,createRun,op,mut,waitNode,advance,assert,save} from './gg-qa-e1-r8-support.mjs';import {launch,variables,timeline,frame,settled,go,focus,previewSettled,metrics} from './gg-qa-e1-r8-browser-support.mjs';
const s=await launch(),{browser,page}=s,results=[];
async function check(id,expected,fn){try{const actual=await fn();results.push({id,expected,actual,pass:true});console.log('PASS '+id);}catch(e){results.push({id,expected,pass:false,error:e.message,focus:await focus(page)});console.log('FAIL '+id+' '+e.message.slice(0,180));}await save('focus-results.json',results);}
async function tabTo(locator){const path=[];for(let i=0;i<120;i++){if(await locator.evaluate(e=>document.activeElement===e))return path;await page.keyboard.press('Tab');path.push(await focus(page));}throw Error('Target not reached in Tab order');}
for(const width of [1440,390]){
 await page.setViewportSize({width,height:width===390?844:900});
 await check(width+': stale hand-off exact keyboard steps','Live B hand-off once; keyboard selections A and cached B remain on the timeline',async()=>{
  const id=await createRun('focus cached '+width,[mut('seed',[op('/vars',{marker:'A'.repeat(20000)}),{op:'append-message',role:'note',content:'M'.repeat(20000)}]),waitNode('ask'),mut('replace',[op('/vars',{marker:'B'.repeat(21000)})]),waitNode('notouch')]);await go(page,id);await previewSettled(page);
  await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();await page.keyboard.press('Shift+Tab');const before=await focus(page);assert(before.label==='Variables','not in expanded region');await advance(id,'notouch');await settled(page,id);await previewSettled(page);const handed=await focus(page);assert(handed.label==='Show all of Variables','initial hand-off failed');
  const a=timeline(page,/node\.finished seed\b/),b=timeline(page,/run\.waiting notouch\b/),tabsToA=await tabTo(a);await page.keyboard.press('Enter');await previewSettled(page);const onA=await focus(page);assert(await a.evaluate(e=>document.activeElement===e),'A moved keyboard');assert(await page.getByRole('button',{name:'Show all of Variables',exact:true}).count()===1,'A did not start bounded');const aChars=(await variables(page).textContent()).length;
  const tabsToB=await tabTo(b);await page.keyboard.press('Enter');await previewSettled(page);const onB=await focus(page);assert(await b.evaluate(e=>document.activeElement===e),'cached B stole focus');assert((await variables(page).textContent()).includes('BBB'),'B contents stale');
  const msg=page.getByRole('button',{name:/^Show all of Message .+ content$/});await msg.click();await page.keyboard.press('Shift+Tab');const otherBefore=await focus(page);assert(otherBefore.label?.startsWith('Message '),'other region not focused');await a.evaluate(e=>e.click());await previewSettled(page);const otherOnA=await focus(page);await b.evaluate(e=>e.click());await previewSettled(page);const otherOnB=await focus(page);assert(otherOnA.label===otherBefore.label&&otherOnB.label===otherBefore.label,'another value component lost focus');
  const screenshot='focus-cached-'+width+'.png';await page.screenshot({path:resolve(dir,screenshot)});return {id,before,handed,tabsToA,onA,aChars,tabsToB,onB,otherBefore,otherOnA,otherOnB,otherValueSelection:'native DOM button.click preserves existing region focus',screenshot,...await metrics(page)};
 });
 await check(width+': large-to-small value hand-off','Text holds focus at tabindex -1; Tab moves on and removes tabindex',async()=>{const id=await createRun('focus small '+width,[mut('seed',[op('/vars',{marker:'A'.repeat(20000)})]),waitNode('ask'),mut('replace',[op('/vars',{marker:'small'})]),waitNode('notouch')]);await go(page,id);await previewSettled(page);await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();await page.keyboard.press('Shift+Tab');await advance(id,'notouch');await settled(page,id);await previewSettled(page);const held=await focus(page);assert(held.tag==='PRE'&&held.label==='Variables'&&held.tabindex==='-1','text did not hold focus');await page.keyboard.press('Tab');const afterTab=await focus(page),tabindex=await variables(page).getAttribute('tabindex');assert(afterTab.text==='Full thread JSON'&&tabindex===null,'Tab/cleanup wrong');return {id,held,afterTab,tabindex};});
}
await save('focus-errors.json',s.errors);await browser.close();console.log(JSON.stringify({checks:results.length,failed:results.filter(r=>!r.pass).length,errors:s.errors}));process.exitCode=results.some(r=>!r.pass)?1:0;
