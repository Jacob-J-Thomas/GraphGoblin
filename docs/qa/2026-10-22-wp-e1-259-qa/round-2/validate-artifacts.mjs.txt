import {readFile,writeFile,readdir} from 'node:fs/promises';import {resolve} from 'node:path';import {createHash} from 'node:crypto';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-2'),files=await readdir(dir),normalized=[];const local=/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/gi;
for(const f of files.filter(f=>/\.(txt|md|json)$/.test(f))){const p=resolve(dir,f),bytes=await readFile(p),utf16=bytes[0]===255&&bytes[1]===254,text=utf16?bytes.subarray(2).toString('utf16le'):bytes.toString('utf8'),clean=text.replace(local,'<USER>');if(utf16||text!==clean){await writeFile(p,clean);normalized.push(f);}}
const md=await readFile(resolve(dir,'README.md'),'utf8'),targets=[...new Set([...md.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(x=>decodeURIComponent(x[1])))],missing=[];
for(const t of targets.filter(t=>!t.startsWith('http'))){try{await readFile(resolve(dir,t));}catch{missing.push(t);}}
const unlinked=files.filter(f=>f!=='README.md'&&!targets.includes(f)),paths=[];for(const f of files.filter(f=>/\.(txt|md|json)$/.test(f))){const t=await readFile(resolve(dir,f),'utf8');if(new RegExp(local.source,'i').test(t))paths.push(f);}
const integrity=JSON.parse(await readFile(resolve(dir,'download-integrity.json'),'utf8')),raw=[];for(const r of integrity){const b=await readFile(resolve(dir,r.protectedFile));raw.push({file:r.protectedFile,hashMatches:createHash('sha256').update(b).digest('hex')===r.sha256});}
const result={files:files.length,uniqueLinks:targets.length,missing,unlinked,remainingLocalUserPaths:paths,normalized,raw};await writeFile(resolve(dir,'artifact-validation.json'),JSON.stringify(result,null,2));console.log(JSON.stringify({files:files.length,missing,unlinked,remainingLocalUserPaths:paths,normalized,allProtectedHashesMatch:raw.every(r=>r.hashMatches)}));
