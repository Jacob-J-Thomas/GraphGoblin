// Preserve downloads under .json.txt so later JSON formatting cannot alter their raw evidence.
import {readFile,writeFile,readdir} from 'node:fs/promises';import {resolve} from 'node:path';import {createHash} from 'node:crypto';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-2'),load=async n=>JSON.parse(await readFile(resolve(dir,n),'utf8')),fixture=await load('fixtures.json'),run=await load('inference-run-before.json'),thread=await load('mutate-thread-before.json'),events=await load('exact-events.json');
const hash=b=>createHash('sha256').update(b).digest('hex'),sort=v=>v&&typeof v==='object'?(Array.isArray(v)?v.map(sort):Object.fromEntries(Object.keys(v).sort().map(k=>[k,sort(v[k])]))):v,results=[];
for(const file of (await readdir(dir)).filter(n=>/^(result|variables|event|thread|exact-.+)-(1440|390)\.json$/.test(n))){const bytes=await readFile(resolve(dir,file)),protectedFile='raw-'+file+'.txt';await writeFile(resolve(dir,protectedFile),bytes);let value;
 if(file.startsWith('result-'))value=run.result;else if(file.startsWith('variables-'))value=thread.vars;else if(file.startsWith('thread-'))value=thread;else if(file.startsWith('event-'))value=fixture.event.payload;else{const name=file.replace(/^exact-/,'').replace(/-(1440|390)\.json$/,'');value=events.find(x=>x.name===name).event.payload;}
 const expected=Buffer.from(JSON.stringify(value,null,2));results.push({file,protectedFile,bytes:bytes.length,sha256:hash(bytes),apiSha256:hash(expected),byteExact:bytes.equals(expected),valueExact:JSON.stringify(sort(JSON.parse(bytes)))===JSON.stringify(sort(value)),disposition:file.startsWith('thread-')?'Reconstruction property order accepted by round-2 brief':'Exact API-formatted bytes required'});
}
await writeFile(resolve(dir,'download-integrity.json'),JSON.stringify(results,null,2));console.log(JSON.stringify({downloads:results.length,valueDifferences:results.filter(x=>!x.valueExact).length,unexpectedByteDifferences:results.filter(x=>!x.byteExact&&!x.file.startsWith('thread-')).length,knownThreadOrderDifferences:results.filter(x=>!x.byteExact&&x.file.startsWith('thread-')).length}));
