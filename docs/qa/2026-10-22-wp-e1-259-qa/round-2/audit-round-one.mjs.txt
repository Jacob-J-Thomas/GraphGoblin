import {readFile,writeFile,readdir} from 'node:fs/promises';import {resolve} from 'node:path';import {createHash} from 'node:crypto';import {execFileSync} from 'node:child_process';
const base=resolve('docs/qa/2026-10-22-wp-e1-259-qa'),dir=resolve(base,'round-2');
const md=await readFile(resolve(base,'README.md'),'utf8'),targets=[...new Set([...md.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(x=>decodeURIComponent(x[1])).filter(x=>!x.startsWith('http')))];
const tracked=new Set(execFileSync('git',['ls-files','docs/qa/2026-10-22-wp-e1-259-qa'],{encoding:'utf8'}).trim().split(/\r?\n/));
const links=[];for(const target of targets){try{const bytes=await readFile(resolve(base,target));links.push({target,exists:true,committed:tracked.has('docs/qa/2026-10-22-wp-e1-259-qa/'+target.replaceAll('\\','/')),bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex')});}catch{links.push({target,exists:false});}}
const manifest=JSON.parse(await readFile(resolve(base,'artifact-validation.json'),'utf8'));const downloads=[];for(const d of manifest.downloads){const raw=await readFile(resolve(base,d.file));downloads.push({file:d.file,recordedBytes:d.bytes,currentBytes:raw.length,sameHash:createHash('sha256').update(raw).digest('hex')===d.sha256,lfNormalizedHashSame:createHash('sha256').update(raw.toString('utf8').replaceAll('\r\n','\n')).digest('hex')===d.sha256,canonicalValueHashSame:createHash('sha256').update(JSON.stringify(JSON.parse(raw),null,2)).digest('hex')===d.sha256});}
const out={head:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),uniqueReferences:links.length,missing:links.filter(x=>!x.exists),uncommittedReferences:links.filter(x=>!x.committed),downloadHashes:downloads,links};await writeFile(resolve(dir,'round-one-audit.json'),JSON.stringify(out,null,2));console.log(JSON.stringify({references:links.length,missing:out.missing,uncommitted:out.uncommittedReferences,downloadHashesMatch:downloads.every(x=>x.sameHash)}));
