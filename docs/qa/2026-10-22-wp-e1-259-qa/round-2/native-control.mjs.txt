import {readFile,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-2');
if(process.argv[2]==='key'){const r=await fetch('http://127.0.0.1:5720/api-keys',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({label:'Independent E1 round 2',scopes:['*']})}),v=await r.json();if(!r.ok)throw new Error('QA key creation refused');await writeFile(resolve(dir,'api-key-metadata.json'),JSON.stringify({created:typeof v.token==='string',id:v.key?.id,label:v.key?.label},null,2));console.log('QA API key created; token excluded from artifacts');}
else{const info=await (await fetch('http://127.0.0.1:5721/info')).json(),r=await fetch('http://127.0.0.1:5721/stop',{method:'POST',headers:{'content-type':'application/json'},body:'{}'}),stop=await r.json();await writeFile(resolve(dir,process.argv[2]==='final'?'final-stop.json':'before-restart.json'),JSON.stringify({info,stop},null,2));console.log(JSON.stringify({info,stop}));}
