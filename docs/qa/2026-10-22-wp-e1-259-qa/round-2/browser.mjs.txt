async function download(label,value,name){const [d]=await Promise.all([page.waitForEvent('download'),page.getByRole('button',{name:'Download '+label+' as JSON',exact:true}).click()]);const actual=await readFile(await d.path());const expected=Buffer.from(JSON.stringify(value,null,2));await writeFile(resolve(dir,name+'.json'),actual);if(label!=='Full thread JSON')assert(actual.equals(expected),'download bytes differ for '+label);return {file:name+'.json',bytes:actual.length,sha256:hash(actual),apiSha256:hash(expected),byteExact:actual.equals(expected)};}
async function value(label,value,{text=false,bound=4096,downloadName,pre}={}){
 const expected=text?value:JSON.stringify(value,null,2);
 const show=page.getByRole('button',{name:'Show all of '+label,exact:true});
 const p=pre??show.locator('..').locator('..').locator('pre').first();
 const preview=await p.textContent();assert(preview===expected.slice(0,bound)+'…',`prefix mismatch ${label}: length ${preview.length}, expected ${bound+1}`);
 const notice=await show.locator('..').textContent();assert(notice.includes('Showing the first '+bound.toLocaleString('en-US')),'notice missing');
 const collapsedWidth=await widths();assert(collapsedWidth.document<=collapsedWidth.window,'collapsed horizontal overflow '+JSON.stringify(collapsedWidth));
 const region=page.locator('pre[role="region"]').and(page.getByLabel(label,{exact:true}));
 const expanded=await timing(async()=>{await show.click();await region.waitFor();});
 const fullActual=await region.textContent();const stable=x=>x&&typeof x==='object'?(Array.isArray(x)?x.map(stable):Object.fromEntries(Object.keys(x).sort().map(k=>[k,stable(x[k])]))):x;const comparable=label==='Full thread JSON'?JSON.stringify(stable(JSON.parse(fullActual)))===JSON.stringify(stable(JSON.parse(expected))):fullActual===expected; if(!comparable){await writeFile(resolve(dir,'mismatch-'+label.replaceAll(/[^a-zA-Z0-9]/g,'-')+'-'+page.viewportSize().width+'.json'),fullActual);throw new Error('full text mismatch '+label+'; actual '+fullActual.length+', expected '+expected.length);}
 await page.keyboard.press('Shift+Tab');const focused=await region.evaluate(e=>document.activeElement===e);assert(focused,'expanded region is not keyboard focusable');
 const scroll=await region.evaluate(e=>({height:e.clientHeight,scroll:e.scrollHeight,overflow:getComputedStyle(e).overflowY,tabIndex:e.tabIndex}));
 if(scroll.scroll>scroll.height){await page.keyboard.press('PageDown');await page.waitForTimeout(150);assert(await region.evaluate(e=>e.scrollTop)>0,'keyboard scrolling failed');}
 const expandedWidth=await widths();assert(expandedWidth.document<=expandedWidth.window,'expanded horizontal overflow '+JSON.stringify(expandedWidth));
 const dl=downloadName?await download(label,value,downloadName):null;
 const collapsed=await timing(async()=>{await page.getByRole('button',{name:'Show less of '+label,exact:true}).click();});
 assert(await p.textContent()===preview,'Show less did not restore prefix');
 return {previewChars:preview.length-1,prettyChars:expected.length,notice,collapsedWidth,expandedWidth,expanded,collapsed,focused,scroll,download:dl};
}
