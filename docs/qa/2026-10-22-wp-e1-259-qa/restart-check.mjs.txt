import { createRequire } from 'node:module';
import { resolve } from 'node:path';
import { readFile,writeFile } from 'node:fs/promises';
import {createHash} from 'node:crypto';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa'),api='http://127.0.0.1:5720';
const fixture=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8'));
const results=[],hash=v=>createHash('sha256').update(JSON.stringify(v)).digest('hex');
for(const [name,id]of Object.entries(fixture.runs))for(const suffix of ['','/thread','/events']){const r=await fetch(api+'/runs/'+id+suffix),actual=await r.json();const expected=JSON.parse(await readFile(resolve(dir,`${name}${suffix.replaceAll('/','-')||'-run'}-before.json`),'utf8'));results.push({name,suffix,pass:r.ok&&hash(actual)===hash(expected),beforeSha256:hash(expected),afterSha256:hash(actual)});}
const info=await (await fetch('http://127.0.0.1:5721/info')).json();
const req=createRequire(resolve('apps/web/package.json'));const {chromium}=req('@playwright/test');const browser=await chromium.launch({channel:'msedge',headless:true});const page=await browser.newPage({viewport:{width:390,height:844}});await page.goto(api+'/app/runs/'+fixture.runs.mutate);await page.getByRole('list',{name:'Timeline',exact:true}).getByRole('button',{name:/run\.waiting/}).click();const text=await page.getByLabel('Variables',{exact:true}).textContent();results.push({name:'UI restart latest marker',pass:text.includes('large')&&text.length===4097,previewLength:text.length});await page.getByRole('button',{name:'Full thread JSON',exact:true}).click();await page.getByRole('button',{name:'Show all of Full thread JSON',exact:true}).click();const pre=page.getByTestId('thread-json');await page.keyboard.press('Shift+Tab');const focused=await pre.evaluate(e=>document.activeElement===e);await page.keyboard.press('PageDown');await page.waitForTimeout(150);results.push({name:'full thread keyboard region',pass:focused&&(await pre.evaluate(e=>e.scrollTop))>0,focused,tabindex:await pre.getAttribute('tabindex'),role:await pre.getAttribute('role'),scrollTop:await pre.evaluate(e=>e.scrollTop)});await browser.close();await writeFile(resolve(dir,'restart-results.json'),JSON.stringify({info,results},null,2));console.log(JSON.stringify({checks:results.length,failed:results.filter(x=>!x.pass).length,info}));
