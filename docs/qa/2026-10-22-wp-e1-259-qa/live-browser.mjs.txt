async function request(path,body){const r=await fetch(api+path,{method:body===undefined?'GET':'POST',headers:{'content-type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const v=await r.json();assert(r.ok(),`${r.status} ${path} ${JSON.stringify(v).slice(0,300)}`);return v;}
// Fetch's Response.ok is a boolean, not a function.
async function call(path,body){const r=await fetch(api+path,{method:body===undefined?'GET':'POST',headers:{'content-type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const v=await r.json();assert(r.ok,`${r.status} ${path} ${JSON.stringify(v).slice(0,300)}`);return v;}
async function check(id,fn){const t=performance.now();try{results.push({id,pass:true,actual:await fn(),ms:performance.now()-t});console.log('PASS '+id);}catch(e){results.push({id,pass:false,error:e.message.slice(0,1200),ms:performance.now()-t});console.log('FAIL '+id+' '+e.message.slice(0,200));}await writeFile(resolve(dir,'live-results.json'),JSON.stringify(results,null,2));}
