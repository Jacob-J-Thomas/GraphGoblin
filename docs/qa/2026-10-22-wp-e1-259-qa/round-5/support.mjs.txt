export async function call(path,body){const r=await fetch(api+path,{method:body===undefined?'GET':'POST',headers:{'content-type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const value=await r.json();assert(r.ok,r.status+' '+path+' '+JSON.stringify(value).slice(0,250));return value;}

export async function waitForRun(id,node){for(let i=0;i<1200;i++){const r=await call('/runs/'+id);if(r.waiting?.nodeId===node)return r;assert(r.status!=='failed','run failed '+JSON.stringify(r.failure));await new Promise(ok=>setTimeout(ok,50));}throw Error('wait timeout '+node);}
export async function createRun(name,middle){const nodes=[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},...middle,{id:'done',kind:'exit',label:'Done',config:{}}].map((n,i)=>({...n,ui:{x:i*260,y:80}}));const definition={schemaVersion:4,name:'QA R5 '+name+' '+Date.now(),nodes,edges:nodes.slice(1).map((n,i)=>({id:'e'+i,from:{node:nodes[i].id,port:'out'},to:{node:n.id}}))};const {loop}=await call('/loops',{definition});await call('/loops/'+loop.id+'/publish',{});const {run}=await call('/loops/'+loop.id+'/runs',{input:{}});await waitForRun(run.id,'ask');return run.id;}
export async function advance(id,node){const r=await call('/runs/'+id);await call('/runs/'+id+'/input',{waitId:r.waiting.waitId,input:{yes:true}});return waitForRun(id,node);}
