import {readFile,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';import {isDeepStrictEqual} from 'node:util';
import {dir,call,advance,assert,save,sha} from './gg-qa-e1-r5-support.mjs';
import {launch,variables,timeline,frame,settled,go,focus,metrics,measure,prepareClick,trace} from './gg-qa-e1-r5-browser-support.mjs';
const fixtures=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8')),s=await launch(),{browser,page,cdp}=s,results=[];
async function check(id,expected,fn){try{const actual=await fn();results.push({id,expected,actual,pass:true});console.log('PASS '+id);}catch(e){results.push({id,expected,pass:false,error:e.message,focus:await focus(page)});console.log('FAIL '+id+' '+e.message.slice(0,200));}await save('performance-results.json',results);}
async function select(re,name){const target=timeline(page,re),click=await prepareClick(page,cdp,target);const result=await measure(page,name,async()=>{await click();await page.waitForFunction(()=>!document.querySelector('pre[aria-label="Variables"]')?.textContent.startsWith('Preparing the preview'));});assert(await target.getAttribute('aria-current')==='true','wrong selected event');return result;}
async function fullThread(id,kind,width){const expected=await call('/runs/'+id+'/thread');await page.getByRole('button',{name:'Full thread JSON',exact:true}).click();const promise=page.waitForEvent('download');await page.getByRole('button',{name:'Download Full thread JSON as JSON',exact:true}).click();const raw=await readFile(await (await promise).path()),name=kind+'-thread-download-'+width+'.json.txt';await writeFile(resolve(dir,name),raw);await writeFile(resolve(dir,kind+'-thread-api-'+width+'.json.txt'),JSON.stringify(expected,null,2));assert(isDeepStrictEqual(JSON.parse(raw),expected),'incremental reconstruction differs from API values');await page.getByRole('button',{name:'Full thread JSON',exact:true}).click();return {download:name,bytes:raw.length,semanticEqual:true,sha256:sha(raw)};}
await save('browser-environment.json',{browser:browser.version(),node:process.version,server:await (await fetch('http://127.0.0.1:5721/info')).json(),traceViewportHeight:3000,normalViewports:[{width:1440,height:900},{width:390,height:844}],samplingIntervalUs:1000});
for(const width of [1440,390]){
 const ids=fixtures.runs[width];await page.setViewportSize({width,height:3000});
 await check(`${width}: wide first seed selection, cached selections, and paint trace`,'Exact prefix and bounded width; capture actual DOM, paints, filmstrip and sampled serializer work',async()=>{await go(page,ids.wide);const expected=(await readFile(resolve(dir,'wide-api-before.json.txt'),'utf8')).slice(0,4096)+'…';await select(/run\.queued/,'queued before first seed');const target=timeline(page,/node\.finished seed\b/),click=await prepareClick(page,cdp,target);const cold=await trace(page,cdp,'wide-cold-seed-'+width,click);assert(await variables(page).textContent()===expected,'cold prefix differs');const cached=[];for(let i=0;i<2;i++){cached.push(await select(/run\.queued/,'queued '+i));cached.push(await select(/node\.finished seed\b/,'cached seed '+i));assert(await variables(page).textContent()===expected,'cached prefix differs');}assert(cold.width<=width&&cached.every(x=>x.maxPre<=4097&&x.width<=width),'wide layout escaped');return {runId:ids.wide,cold,cached};});
 await check(`${width}: live append leaves vars untouched`,'Retained preview, no repeated serializer enumeration; full stored value unchanged',async()=>{await go(page,ids.wide);const before=await call('/runs/'+ids.wide+'/thread');const probe=await trace(page,cdp,'wide-live-no-vars-'+width,async()=>{await advance(ids.wide,'notouch');await settled(page,ids.wide);});const after=await call('/runs/'+ids.wide+'/thread');assert(isDeepStrictEqual(before.vars,after.vars),'untouched vars changed');assert(await variables(page).textContent()===JSON.stringify(after.vars,null,2).slice(0,4096)+'…','no-vars prefix wrong');assert(probe.serializerSamples===0,'sampled preview serializer recurred after untouched-vars append');return {probe,keys:Object.keys(after.vars).length,varsUnchanged:true};});
 await check(`${width}: live append changes vars`,'One changed-value enumeration expected; exact new prefix and actual placeholder-paint ordering retained',async()=>{await go(page,ids.wide);const probe=await trace(page,cdp,'wide-live-changed-vars-'+width,async()=>{await advance(ids.wide,'touched');await settled(page,ids.wide);});const current=await call('/runs/'+ids.wide+'/thread');assert(Object.keys(current.vars).length===350000&&current.vars.key0==='CHANGED 🚀\u0000','changed API value wrong');assert(await variables(page).textContent()===JSON.stringify(current.vars,null,2).slice(0,4096)+'…','changed prefix stale');await writeFile(resolve(dir,'wide-api-after-'+width+'.json.txt'),JSON.stringify(current.vars,null,2));return {probe,keys:350000,exactPrefix:true};});
 await check(`${width}: cold earlier wait without a new diff component`,'Independent cold-selection paint trace for an existing Variables component, with no new diff layout',async()=>{await select(/run\.queued/,'queued before unseen earlier wait');const target=timeline(page,/run\.waiting ask\b/),click=await prepareClick(page,cdp,target);const probe=await trace(page,cdp,'wide-cold-wait-'+width,click);assert(await variables(page).textContent()===(await readFile(resolve(dir,'wide-api-before.json.txt'),'utf8')).slice(0,4096)+'…','old wait prefix differs');await save('wide-all-tasks-'+width+'.json',await page.evaluate(()=>window.__qaTasks));return {probe};});
 await check(`${width}: full incremental thread equals live API`,'Reconstructed values equal API after untouched and touched vars appends',async()=>{await go(page,ids.wide);return fullThread(ids.wide,'wide',width);});
 await page.setViewportSize({width,height:width===390?844:900});
 await check(`${width}: 6000-message re-selection and exact download`,'Latest 50; exact notice; bounded ordinary selections; API-byte-identical messages download',async()=>{await go(page,ids.many);assert((await metrics(page)).messages===50,'unbounded initial messages');const probes=[];for(let i=0;i<2;i++){probes.push(await select(/run\.queued/,'messages queued '+i));probes.push(await select(/run\.waiting ask\b/,'messages ask '+i));assert((await metrics(page)).messages===50,'re-selection unbounded');}assert(await page.getByText('Showing the last 50 of 6,000 messages.',{exact:true}).count()===1,'notice wrong');const promise=page.waitForEvent('download');await page.getByRole('button',{name:'Download messages as JSON',exact:true}).click();const raw=await readFile(await (await promise).path()),api=await readFile(resolve(dir,'messages-api.json.txt'));assert(raw.equals(api),'messages download differs');const artifact='messages-download-'+width+'.json.txt';await writeFile(resolve(dir,artifact),raw);return {probes,bytes:raw.length,sha256:sha(raw),artifact};});
 for(const all of [false,true])await check(`${width}: message append with ${all?'Show all':'window'} on`,'Stored message count grows; current window returns to 50; ordinary controls retain focus',async()=>{await go(page,ids.many);if(all)await page.getByRole('button',{name:'Show all messages',exact:true}).click();const before=await metrics(page);const probe=await measure(page,'messages live '+(all?'all':'window'),async()=>{await advance(ids.many,all?'touched':'notouch');await settled(page,ids.many);});const thread=await call('/runs/'+ids.many+'/thread');assert(probe.messages===50&&thread.messages.length===(all?6004:6002),'message context/window wrong');await save('messages-live-'+(all?'all':'window')+'-tasks-'+width+'.json',await page.evaluate(()=>window.__qaTasks));return {before,probe,storedCount:thread.messages.length,focus:await focus(page)};});
 await check(`${width}: full message-thread replay equals API`,'All 6004 messages and full replayed values are preserved',()=>fullThread(ids.many,'messages',width));
 await check(`${width}: original A/B/A and strict nested download`,'2322721-character expansion, bounded return, same exact stored data',async()=>{await go(page,fixtures.runs.nested);await select(/node\.finished replace\b/,'select original A');await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();assert((await variables(page).textContent()).length===2322721,'fixture size differs');const b=await select(/node\.finished seed\b/,'A to B'),a=await select(/node\.finished replace\b/,'B to A');assert((await variables(page).textContent()).length===4097,'old expansion returned');const expected=Buffer.from(JSON.stringify((await call('/runs/'+fixtures.runs.nested+'/thread')).vars,null,2));const promise=page.waitForEvent('download');await page.getByRole('button',{name:'Download Variables as JSON',exact:true}).click();const raw=await readFile(await (await promise).path());assert(raw.equals(expected),'nested bytes differ');const artifact='nested-download-'+width+'.json.txt';await writeFile(resolve(dir,artifact),raw);if(width===1440)await writeFile(resolve(dir,'nested-api.json.txt'),expected);return {b,a,bytes:raw.length,sha256:sha(raw),artifact};});
}
await save('performance-errors.json',s.errors);await browser.close();console.log(JSON.stringify({checks:results.length,failed:results.filter(x=>!x.pass).length,errors:s.errors.length}));process.exitCode=results.some(x=>!x.pass)?1:0;
