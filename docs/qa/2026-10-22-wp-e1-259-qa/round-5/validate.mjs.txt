import {readFile,writeFile,readdir,stat} from 'node:fs/promises';import {resolve} from 'node:path';import {createHash} from 'node:crypto';import {createRequire} from 'node:module';import {isDeepStrictEqual} from 'node:util';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-5'),prettier=createRequire(resolve('package.json'))('prettier'),files=await readdir(dir),read=n=>readFile(resolve(dir,n),'utf8');
const report=await read('README.md'),links=[...report.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(m=>m[1]),missing=[];for(const n of links){if(/^https?:|^#/.test(n))continue;try{await stat(resolve(dir,n.split('#')[0]));}catch{missing.push(n);}}
const unlinked=files.filter(n=>n!=='README.md'&&!links.includes(n)),localPaths=[];for(const n of files){if(/\.(png|jpg)$/.test(n))continue;const text=await read(n);if(/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/i.test(text))localPaths.push(n);}
const downloads=[];for(const width of [1440,390])for(const kind of ['messages','nested']){const raw=await readFile(resolve(dir,kind+'-download-'+width+'.json.txt')),api=await readFile(resolve(dir,kind+'-api.json.txt'));downloads.push({kind,width,bytes:raw.length,identical:raw.equals(api),sha256:createHash('sha256').update(raw).digest('hex')});}
const replay=JSON.parse(await read('incremental-replay-results.json'));for(const r of replay){const a=JSON.parse(await read(r.download)),b=JSON.parse(await read(r.api));if(!isDeepStrictEqual(a,b))throw Error('replay artifact value difference');}
const assessment=JSON.parse(await read('assessment.json')),focus=JSON.parse(await read('focus-results.json')),prior=JSON.parse(await read('focus-small-repro.json'));
const result={files:files.length,missingLinks:missing,unlinkedFiles:unlinked,localUserPathFiles:localPaths,coreCases:assessment.main.length,coreFailures:assessment.main.filter(x=>!x.pass).length,focusCases:focus.length,focusFailures:focus.filter(x=>!x.pass).length,oldFocusCases:prior.length,oldFocusFailures:prior.filter(x=>!x.pass).length,liveReplayCases:replay.length,liveReplayFailures:replay.filter(x=>!x.pass).length,knownDefectsReported:assessment.defects.every(x=>report.includes(x.id))&&report.includes('**FAIL**'),downloads};
await writeFile(resolve(dir,'validation.json'),await prettier.format(JSON.stringify(result),{parser:'json'}));console.log(JSON.stringify(result));process.exitCode=missing.length||unlinked.length||localPaths.length||result.coreFailures||result.oldFocusFailures||result.liveReplayFailures||!result.knownDefectsReported||downloads.some(x=>!x.identical)?1:0;
