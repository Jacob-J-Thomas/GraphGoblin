// Remove host prefixes from the retained source map, preserving its generated mapping table.
import {readFile,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';import {createHash} from 'node:crypto';import {createRequire} from 'node:module';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-5'),prettier=createRequire(resolve('package.json'))('prettier'),file=resolve(dir,'app-source-map.json.txt'),raw=await readFile(file,'utf8'),old=JSON.parse(raw),pattern=/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/gi,redactions=[...raw.matchAll(pattern)].length,clean=raw.replace(pattern,'<USER>'),current=JSON.parse(clean),sha=s=>createHash('sha256').update(s).digest('hex');
if(old.mappings!==current.mappings)throw Error('source map mappings changed');await writeFile(file,clean);
const provenanceFile=resolve(dir,'source-map-provenance.json'),provenance=JSON.parse(await readFile(provenanceFile,'utf8'));provenance.retainedSanitizedSha256=sha(clean);provenance.originalSha256=sha(raw);provenance.hostPrefixRedactions=redactions;provenance.mappingTableUnchanged=true;
await writeFile(provenanceFile,await prettier.format(JSON.stringify(provenance),{parser:'json'}));await writeFile(resolve(dir,'sanitization.json'),await prettier.format(JSON.stringify({artifact:'app-source-map.json.txt',redactions,originalSha256:sha(raw),retainedSha256:sha(clean),mappingTableUnchanged:true,downloadsAndProfilesUnchanged:true}),{parser:'json'}));console.log(JSON.stringify({redactions,mappingTableUnchanged:true}));
