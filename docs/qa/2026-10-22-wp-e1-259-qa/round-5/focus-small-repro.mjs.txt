// Causal browser diagnosis: compare an immediate small mutation with a fake inference delay.

const {chromium}=createRequire(resolve('apps/web/package.json'))('@playwright/test'),dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-5'),api='http://127.0.0.1:5720';
async function call(path,body,base=api){const r=await fetch(base+path,{method:body===undefined?'GET':'POST',headers:{'content-type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)}),v=await r.json();if(!r.ok)throw new Error(`${r.status} ${path}`);return v;}
const browser=await chromium.launch({channel:'msedge',headless:true}),page=await browser.newPage();page.setDefaultTimeout(12000);const results=[];
const set=(id,value)=>({id,kind:'mutate',label:id,config:{operations:[{op:'set',path:'/vars',value:{kind:'literal',value}}]}}),wait=id=>({id,kind:'wait',label:id,config:{mode:'input',prompt:'Focus reproduction',inputSchema:{type:'string'}}});
for(const width of [1440,390])for(const delayMs of [0,500]){
 const nodes=[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},set('seed',{marker:'A'.repeat(20000)}),wait('ask1'),...(delayMs?[{id:'work',kind:'inference',label:'Work',config:{prompt:{template:'Emit a short result while waiting.'}}}]:[]),set('small',{small:true}),wait('ask2'),{id:'done',kind:'exit',label:'Done',config:{}}].map((n,i)=>({...n,ui:{x:i*260,y:80}}));const definition={schemaVersion:4,name:'QA small focus '+width+' '+delayMs+' '+Date.now(),nodes,edges:nodes.slice(0,-1).map((n,i)=>({id:'e'+i,from:{node:n.id,port:'out'},to:{node:nodes[i+1].id}}))};
 if(delayMs)await call('/script',{turns:[{delayMs,finalText:'OK'}]},'http://127.0.0.1:5721');const {loop}=await call('/loops',{definition});await call('/loops/'+loop.id+'/publish',{});const {run}=await call('/loops/'+loop.id+'/runs',{input:{}});
 let record;for(let i=0;i<200;i++){record=await call('/runs/'+run.id);if(record.waiting?.nodeId==='ask1')break;await new Promise(r=>setTimeout(r,50));}
 await page.setViewportSize({width,height:width===390?844:900});await page.goto(api+'/app/runs/'+run.id);await page.getByRole('list',{name:'Timeline',exact:true}).getByRole('button',{name:/run\.waiting/}).waitFor();await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();await page.keyboard.press('Shift+Tab');
 const pre=page.getByLabel('Variables',{exact:true});if(!(await pre.evaluate(e=>document.activeElement===e)))throw new Error('expanded region not focused');
 await page.evaluate(()=>{window.__focusTrace=[];const info=()=>{const e=document.activeElement,p=document.querySelector('pre[aria-label="Variables"]');return {at:performance.now(),tag:e?.tagName,label:e?.getAttribute('aria-label'),buttonText:e?.tagName==='BUTTON'?e.textContent:undefined,chars:p?.textContent.length,tabindex:p?.getAttribute('tabindex'),region:p?.getAttribute('role'),first:p?.textContent.slice(0,45),showAll:!!document.querySelector('button[aria-label="Show all of Variables"]')};};window.__focusTrace.push({cause:'start',...info()});document.addEventListener('focusin',e=>window.__focusTrace.push({cause:'focusin',target:e.target.tagName,...info()}),true);document.addEventListener('focusout',e=>window.__focusTrace.push({cause:'focusout',target:e.target.tagName,...info()}),true);new MutationObserver(list=>window.__focusTrace.push({cause:'mutation',changes:list.map(x=>({type:x.type,attribute:x.attributeName,old:x.oldValue})),...info()})).observe(document.querySelector('pre[aria-label="Variables"]').parentElement,{subtree:true,childList:true,characterData:true,attributes:true,attributeOldValue:true});});
 await call('/runs/'+run.id+'/input',{waitId:record.waiting.waitId,input:'continue'});for(let i=0;i<200;i++){record=await call('/runs/'+run.id);if(record.waiting?.nodeId==='ask2')break;await new Promise(r=>setTimeout(r,50));}await page.waitForFunction(()=>document.querySelector('pre[aria-label="Variables"]')?.textContent.includes('"small": true'));await page.evaluate(()=>new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r))));
 const beforeTab=await page.evaluate(()=>{const e=document.activeElement,p=document.querySelector('pre[aria-label="Variables"]');return {tag:e.tagName,label:e.getAttribute('aria-label'),tabindex:p.getAttribute('tabindex'),preFocused:e===p};});await page.keyboard.press('Tab');const afterTab=await page.evaluate(()=>({tag:document.activeElement.tagName,text:document.activeElement.textContent.slice(0,80),tabindex:document.querySelector('pre[aria-label="Variables"]').getAttribute('tabindex')}));const trace=await page.evaluate(()=>window.__focusTrace),screenshot=`focus-small-${width}-${delayMs}.png`;await pre.scrollIntoViewIfNeeded();await page.screenshot({path:resolve(dir,screenshot)});
 const out={width,delayMs,runId:run.id,expected:'Small Variables pre holds focus at tabindex -1; Tab reaches Full thread JSON and removes tabindex',beforeTab,afterTab,pass:beforeTab.preFocused&&beforeTab.tabindex==='-1'&&afterTab.text==='Full thread JSON'&&afterTab.tabindex===null,trace,screenshot,events:await call('/runs/'+run.id+'/events')};results.push(out);await writeFile(resolve(dir,'focus-small-repro.json'),JSON.stringify(results,null,2));console.log(JSON.stringify({width,delayMs,pass:out.pass,beforeTab,afterTab,trace:trace.map(x=>({cause:x.cause,tag:x.tag,label:x.label,chars:x.chars,tabindex:x.tabindex,showAll:x.showAll}))}));
}
