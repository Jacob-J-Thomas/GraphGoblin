// Read-only DOM observers, Chrome tracing/screenshots and CPU sampling; no app internals patched.
import {createRequire} from 'node:module';import {resolve,basename} from 'node:path';import {readFile,writeFile} from 'node:fs/promises';
import {dir,api,call,assert,save,sha} from './gg-qa-e1-r5-support.mjs';
const req=createRequire(resolve('apps/web/package.json')),{chromium}=req('@playwright/test');
const {SourceMapConsumer}=req(req.resolve('source-map-js',{paths:[req.resolve('vite'),req.resolve('postcss',{paths:[req.resolve('vite')]})]}));
const consumers=new Map();
export async function launch(){const browser=await chromium.launch({channel:'msedge',headless:true}),context=await browser.newContext();await context.addInitScript(()=>{window.__qaTasks=[];window.__qaStates=[];new PerformanceObserver(l=>window.__qaTasks.push(...l.getEntries().map(x=>({start:x.startTime,duration:x.duration,name:x.name})))).observe({type:'longtask',buffered:true});const snapshot=()=>{const p=document.querySelector('pre[aria-label="Variables"]'),s=p?.textContent??'';const state={at:performance.now(),chars:s.length,text:s.slice(0,70),pending:s.startsWith('Preparing the preview'),selected:document.querySelector('[aria-label="Timeline"] [aria-current="true"]')?.textContent};const previous=window.__qaStates.at(-1);if(!previous||previous.text!==state.text||previous.selected!==state.selected){window.__qaStates.push(state);if(window.__qaProbe)performance.mark('qa-state-'+window.__qaProbe+'-'+window.__qaStates.length);}};new MutationObserver(snapshot).observe(document,{childList:true,subtree:true,characterData:true,attributes:true,attributeFilter:['aria-current']});});const page=await context.newPage(),errors=[];page.setDefaultTimeout(30000);page.on('pageerror',e=>errors.push(e.message));page.on('console',m=>{if(m.type()==='error')errors.push(m.text());});const cdp=await context.newCDPSession(page);await cdp.send('Performance.enable');return {browser,context,page,cdp,errors};}
export const variables=p=>p.getByLabel('Variables',{exact:true}),timeline=(p,re)=>p.getByRole('list',{name:'Timeline',exact:true}).getByRole('button',{name:re});
export async function frame(p){await p.evaluate(()=>new Promise(ok=>requestAnimationFrame(()=>requestAnimationFrame(ok))));}
export async function settled(p,id){const run=await call('/runs/'+id);await p.waitForFunction(seq=>[...document.querySelectorAll('[aria-label="Timeline"] button')].some(b=>b.firstElementChild?.textContent==='#'+seq),run.lastEventSeq);await frame(p);}
export async function go(p,id){await p.goto(api+'/app/runs/'+id);await settled(p,id);await variables(p).waitFor();await p.waitForFunction(()=>!document.querySelector('pre[aria-label="Variables"]')?.textContent.startsWith('Preparing the preview'));}
export async function focus(p){return p.evaluate(()=>{const e=document.activeElement;return {tag:e.tagName,label:e.getAttribute('aria-label'),role:e.getAttribute('role'),tabindex:e.getAttribute('tabindex'),text:e.tagName==='BUTTON'?e.textContent:undefined};});}
export async function metrics(p){return p.evaluate(()=>({domElements:document.querySelectorAll('*').length,messages:document.querySelectorAll('ol[aria-label="Messages"] > li').length,preChars:document.querySelector('pre[aria-label="Variables"]')?.textContent.length,maxPre:Math.max(0,...[...document.querySelectorAll('pre')].map(e=>e.textContent.length)),width:document.documentElement.scrollWidth,viewport:innerWidth}));}
export async function measure(p,name,fn){const since=await p.evaluate(()=>performance.now()),start=performance.now();await fn();await frame(p);const wallMs=performance.now()-start,end=await p.evaluate(()=>performance.now());await p.waitForTimeout(80);return {name,wallMs,since,end,longTasks:await p.evaluate(({since,end})=>window.__qaTasks.filter(t=>t.start>=since&&t.start<end),{since,end}),...await metrics(p)};}
export async function prepareClick(p,cdp,locator){await locator.scrollIntoViewIfNeeded();await frame(p);const box=await locator.boundingBox();assert(box,'target box absent');return ()=>cdp.send('Input.dispatchMouseEvent',{type:'mousePressed',x:box.x+box.width/2,y:box.y+box.height/2,button:'left',clickCount:1}).then(()=>cdp.send('Input.dispatchMouseEvent',{type:'mouseReleased',x:box.x+box.width/2,y:box.y+box.height/2,button:'left',clickCount:1}));}
async function original(frame){if(!frame.url?.startsWith(api)||frame.lineNumber<0)return null;const file=basename(new URL(frame.url).pathname);try{let map=consumers.get(file);if(!map){const raw=await readFile(resolve('apps/web/dist/assets',file+'.map'),'utf8');map=new SourceMapConsumer(JSON.parse(raw));consumers.set(file,map);await writeFile(resolve(dir,'app-source-map.json.txt'),raw);await save('source-map-provenance.json',{asset:file,sourceMapSha256:sha(raw)});}return map.originalPositionFor({line:frame.lineNumber+1,column:frame.columnNumber});}catch{return null;}}
export async function trace(p,cdp,id,fn,fullResolution=false){const events=[],collector=e=>events.push(...e.value),sc=[];const timeOrigin=await p.evaluate(()=>performance.timeOrigin);const collectFrame=e=>{sc.push(e);void cdp.send('Page.screencastFrameAck',{sessionId:e.sessionId}).catch(()=>{});};cdp.on('Tracing.dataCollected',collector);if(fullResolution){cdp.on('Page.screencastFrame',collectFrame);const size=p.viewportSize();await cdp.send('Page.startScreencast',{format:'png',everyNthFrame:1,maxWidth:size.width,maxHeight:size.height});}await cdp.send('Profiler.enable');await cdp.send('Profiler.setSamplingInterval',{interval:1000});await cdp.send('Tracing.start',{categories:'devtools.timeline,blink.user_timing'+(fullResolution?'':',disabled-by-default-devtools.screenshot'),options:'record-as-much-as-possible',transferMode:'ReportEvents'});await cdp.send('Profiler.start');const startMark='qa-start-'+id;const start=await p.evaluate(({id,startMark})=>{window.__qaProbe=id;window.__qaStates=[];const mark=performance.mark(startMark);return mark.startTime;},{id,startMark});
 const result=await measure(p,id,async()=>{await fn();await p.waitForFunction(()=>!document.querySelector('pre[aria-label="Variables"]')?.textContent.startsWith('Preparing the preview'));});await p.waitForTimeout(150);const states=await p.evaluate(()=>{window.__qaProbe=undefined;return window.__qaStates;});const {profile}=await cdp.send('Profiler.stop');await cdp.send('Profiler.disable');const complete=new Promise(ok=>cdp.once('Tracing.tracingComplete',ok));await cdp.send('Tracing.end');await complete;cdp.off('Tracing.dataCollected',collector);
 if(fullResolution){await cdp.send('Page.stopScreencast');cdp.off('Page.screencastFrame',collectFrame);await writeFile(resolve(dir,id+'.screencast.json.txt'),JSON.stringify(sc));}const marker=events.find(e=>e.name===startMark),origin=marker?marker.ts/1000-start:undefined;const nodeById=new Map(profile.nodes.map(n=>[n.id,n])),parents=new Map();for(const n of profile.nodes){n.original=await original(n.callFrame);for(const ch of n.children??[])parents.set(ch,n.id);n.prefixFrame=!!n.original?.source?.endsWith('/lib/utils.ts')&&n.original.line>=41&&n.original.line<=174;}
 let clock=profile.startTime,firstPrefixAt;const prefixSamples=[],serializerSamples=[];for(let i=0;i<(profile.samples??[]).length;i++){clock+=(profile.timeDeltas?.[i]??0);let node=nodeById.get(profile.samples[i]),prefix=false,serializer=false;while(node){if(node.prefixFrame)prefix=true;if(node.original?.source?.endsWith('/lib/utils.ts')&&node.original.line>=41&&node.original.line<=147)serializer=true;node=nodeById.get(parents.get(node.id));}const sample={at:clock/1000-(origin??0),deltaMs:(profile.timeDeltas?.[i]??0)/1000};if(prefix){firstPrefixAt??=sample.at;prefixSamples.push(sample);}if(serializer)serializerSamples.push(sample);}
 const screenshots=fullResolution?sc.map(e=>({at:e.metadata.timestamp*1000-timeOrigin,data:e.data})):events.filter(e=>e.name==='Screenshot'&&e.args?.snapshot).map(e=>({at:e.ts/1000-(origin??0),data:e.args.snapshot}));const pending=states.find(s=>s.pending),ready=pending?states.find(s=>!s.pending&&s.at>pending.at):undefined;const chosen=new Set();
 if(screenshots.length){chosen.add(0);chosen.add(screenshots.length-1);if(firstPrefixAt!==undefined){const before=screenshots.findLastIndex(s=>s.at<firstPrefixAt);if(before>=0)chosen.add(before);const after=screenshots.findIndex(s=>s.at>=firstPrefixAt);if(after>=0)chosen.add(after);}if(pending){const during=screenshots.map((s,i)=>({s,i})).filter(x=>x.s.at>=pending.at&&(!ready||x.s.at<ready.at));for(const x of during.slice(0,3))chosen.add(x.i);}}
 const frames=[];for(const i of [...chosen].sort((a,b)=>a-b)){const shot=screenshots[i],bytes=Buffer.from(shot.data,'base64'),extension=bytes[0]===255?'jpg':'png',name=id+'-frame-'+i+'.'+extension;await writeFile(resolve(dir,name),bytes);frames.push({at:shot.at,name,duringPendingDom:!!pending&&shot.at>=pending.at&&(!ready||shot.at<ready.at)});}
 const sanitized=s=>s.replace(/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/gi,'<USER>');await writeFile(resolve(dir,id+'.trace.json.txt'),sanitized(JSON.stringify({traceEvents:events})));await writeFile(resolve(dir,id+'.cpuprofile.json.txt'),sanitized(JSON.stringify(profile)));
 const out={...result,trace:id+'.trace.json.txt',cpuProfile:id+'.cpuprofile.json.txt',fullResolution,screencast:fullResolution?id+'.screencast.json.txt':undefined,states,traceOriginMs:origin,firstPrefixSampleAt:firstPrefixAt,prefixSamples:prefixSamples.length,prefixSampledMs:prefixSamples.reduce((s,x)=>s+x.deltaMs,0),serializerSamples:serializerSamples.length,serializerSampledMs:serializerSamples.reduce((s,x)=>s+x.deltaMs,0),firstSerializerSampleAt:serializerSamples[0]?.at,pendingDomAt:pending?.at,settledDomAt:ready?.at,paintEvents:events.filter(e=>e.name==='Paint').map(e=>({at:e.ts/1000-(origin??0),duration:(e.dur??0)/1000,clip:e.args?.data?.clip})),frames,allFrameTimes:screenshots.map(s=>s.at)};await save(id+'.json',out);return out;
}
