import {readFile,writeFile,mkdir,stat} from 'node:fs/promises';import {resolve,relative} from 'node:path';import {execFileSync} from 'node:child_process';import {createHash} from 'node:crypto';
const base=resolve('docs/qa/2026-10-22-wp-e1-259-qa'),dir=resolve(base,'round-5');await mkdir(dir,{recursive:true});
for(const name of ['run-command.mjs.txt','server.mjs.txt','playwright.config.mjs.txt','focus-small-repro.mjs.txt'])await writeFile(resolve(dir,name),(await readFile(resolve(base,'round-4',name),'utf8')).replaceAll('round-4','round-5').replaceAll('r4-playwright','r5-playwright'));
const tracked=new Set(execFileSync('git',['ls-files'],{encoding:'utf8'}).split(/\r?\n/)),records=[];
for(const round of ['', 'round-2','round-3','round-4']){const folder=resolve(base,round),report=await readFile(resolve(folder,'README.md'),'utf8');const refs=[...new Set([...report.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(x=>x[1]).filter(x=>!/^https?:|^#/.test(x)))];for(const ref of refs){const file=resolve(folder,ref);try{const s=await stat(file);records.push({round:round||'round-1',ref,exists:true,tracked:tracked.has(relative(process.cwd(),file).replaceAll('\\','/')),bytes:s.size,sha256:createHash('sha256').update(await readFile(file)).digest('hex')});}catch{records.push({round:round||'round-1',ref,exists:false});}}}
await writeFile(resolve(dir,'prior-evidence-audit.json'),JSON.stringify({references:records.length,missing:records.filter(x=>!x.exists).length,untracked:records.filter(x=>!x.tracked).length,records},null,2));
await writeFile(resolve(dir,'environment.json'),JSON.stringify({head:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),node:process.version,status:execFileSync('git',['status','--short'],{encoding:'utf8'}).trim(),apiPort:5720,controlPort:5721},null,2));
await writeFile(resolve(dir,'change.diff.txt'),execFileSync('git',['diff','8984d7f44b1e3f88e9f7ada8388da2f616bad6f2','HEAD']));
console.log(JSON.stringify({node:process.version,priorReferences:records.length,missing:records.filter(x=>!x.exists).length,untracked:records.filter(x=>!x.tracked).length}));
