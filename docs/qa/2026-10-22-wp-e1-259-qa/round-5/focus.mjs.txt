import {resolve} from 'node:path';import {dir,call,createRun,op,mut,waitNode,messages,advance,assert,save} from './gg-qa-e1-r5-support.mjs';
import {launch,variables,timeline,frame,settled,go,focus,metrics} from './gg-qa-e1-r5-browser-support.mjs';
const s=await launch(),{browser,page}=s,base=await messages(),results=[];
async function check(id,expected,fn){try{const actual=await fn();results.push({id,expected,actual,pass:true});console.log('PASS '+id);}catch(e){results.push({id,expected,pass:false,error:e.message,focus:await focus(page)});console.log('FAIL '+id+' '+e.message.slice(0,200));}await save('focus-results.json',results);}
for(const width of [1440,390]){
 await page.setViewportSize({width,height:width===390?844:900});
 for(const mode of ['window control','all region exact R4','all control','outside Full thread JSON','outside timeline','all region to no controls'])await check(`${width}: ${mode}`,'Removed focused message hands to Show all or held list; outside focus unchanged; Tab cleanup correct',async()=>{const index=mode.startsWith('window')?5950:0,ms=base.map((m,i)=>i===index?{...m,content:'F'.repeat(20000)}:m);const clear=mode.includes('no controls');const runId=await createRun(mode+' '+width,[mut('seed',[op('/messages',ms)]),waitNode('ask'),mut('append',clear?[op('/messages',[])]:[{op:'append-message',role:'note',content:'live appended message'}]),waitNode('notouch')]);await go(page,runId);
   if(!mode.startsWith('window'))await page.getByRole('button',{name:'Show all messages',exact:true}).click();const label='Message m'+index+' content';const control=page.getByRole('button',{name:'Show all of '+label,exact:true});
   if(mode.includes('region')||mode.startsWith('outside')){await control.click();await page.keyboard.press('Shift+Tab');assert((await focus(page)).label===label,'region not focused before replacement');}else await control.focus();
   let outside;if(mode==='outside Full thread JSON')outside=page.getByRole('button',{name:'Full thread JSON',exact:true});if(mode==='outside timeline')outside=timeline(page,/node\.finished seed\b/);if(outside)await outside.focus();
   const before=await focus(page);await advance(runId,'notouch');await settled(page,runId);const after=await focus(page),list=page.getByRole('list',{name:'Messages',exact:true}),listTabindex=await list.getAttribute('tabindex');
   const record={mode,runId,before,after,listTabindex,metrics:await metrics(page),storedCount:(await call('/runs/'+runId+'/thread')).messages.length};
   if(outside)assert(await outside.evaluate(e=>document.activeElement===e),'outside keyboard moved');else if(clear)assert(after.tag==='OL'&&after.label==='Messages'&&listTabindex==='-1','list did not hold focus without controls');else assert(after.tag==='BUTTON'&&after.label==='Show all messages','message hand-off lost: '+JSON.stringify(after));
   await page.keyboard.press('Tab');record.afterTab=await focus(page);record.listTabindexAfterTab=await list.getAttribute('tabindex');if(!outside){assert(record.listTabindexAfterTab===null,'list tabindex left behind');assert(clear?record.afterTab.text==='Full thread JSON':record.afterTab.label==='Download messages as JSON','wrong next Tab control');}
   const screenshot='focus-'+width+'-'+mode.replaceAll(' ','-')+'.png';await list.scrollIntoViewIfNeeded();await page.screenshot({path:resolve(dir,screenshot)});record.screenshot=screenshot;return record;
 });
 await check(`${width}: value hand-off followed by outside A/B/A navigation`,'After a live value hand-off, returning to that cached value must leave the keyboard on the timeline',async()=>{const runId=await createRun('outside return '+width,[mut('seed',[op('/vars',{marker:'A'.repeat(20000)})]),waitNode('ask'),mut('replace',[op('/vars',{marker:'B'.repeat(21000)})]),waitNode('notouch')]);await go(page,runId);await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();await page.keyboard.press('Shift+Tab');assert((await focus(page)).label==='Variables','Variables region not focused');await advance(runId,'notouch');await settled(page,runId);await page.waitForFunction(()=>!document.querySelector('pre[aria-label="Variables"]').textContent.startsWith('Preparing'));const handed=await focus(page);assert(handed.label==='Show all of Variables','initial large-to-large hand-off failed');const a=timeline(page,/node\.finished seed\b/),b=timeline(page,/run\.waiting notouch\b/);await a.focus();await page.keyboard.press('Enter');await frame(page);await page.waitForFunction(()=>document.querySelector('pre[aria-label="Variables"]').textContent.includes('AAA'));const onA=await focus(page);await b.focus();await page.keyboard.press('Enter');await frame(page);await page.waitForFunction(()=>document.querySelector('pre[aria-label="Variables"]').textContent.includes('BBB'));await frame(page);const returned=await focus(page),timelineFocused=await b.evaluate(e=>document.activeElement===e);const evidence={runId,handed,onA,returned,timelineFocused};await save('outside-return-'+width+'.json',evidence);assert(timelineFocused,'returning cached value stole keyboard: '+JSON.stringify(returned));return evidence;});
}
await save('focus-errors.json',s.errors);await browser.close();console.log(JSON.stringify({checks:results.length,failed:results.filter(x=>!x.pass).length,errors:s.errors.length}));process.exitCode=results.some(x=>!x.pass)?1:0;
