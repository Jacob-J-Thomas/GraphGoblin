// Compare API values with a browser reconstruction that has just received a live completion;
// no reload between the warm remembered thread and the download.
import {readFile,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';import {isDeepStrictEqual} from 'node:util';
import {dir,call,assert,save,sha} from './gg-qa-e1-r5-support.mjs';import {launch,go,frame} from './gg-qa-e1-r5-browser-support.mjs';
const fixtures=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8')),s=await launch(),{browser,page}=s,results=[];
for(const width of [1440,390])for(const kind of ['wide','many']){
 const runId=fixtures.runs[width][kind];await page.setViewportSize({width,height:width===390?844:900});await go(page,runId);const before=await call('/runs/'+runId);await call('/runs/'+runId+'/input',{waitId:before.waiting.waitId,input:{finished:true}});for(let i=0;i<300;i++){if((await call('/runs/'+runId)).status==='succeeded')break;await new Promise(ok=>setTimeout(ok,50));}
 await page.getByRole('list',{name:'Timeline',exact:true}).getByText('run.finished',{exact:true}).waitFor();await frame(page);const expected=await call('/runs/'+runId+'/thread');await page.getByRole('button',{name:'Full thread JSON',exact:true}).click();const pending=page.waitForEvent('download');await page.getByRole('button',{name:'Download Full thread JSON as JSON',exact:true}).click();const raw=await readFile(await (await pending).path()),download='live-completion-'+kind+'-download-'+width+'.json.txt',api='live-completion-'+kind+'-api-'+width+'.json.txt';await writeFile(resolve(dir,download),raw);await writeFile(resolve(dir,api),JSON.stringify(expected,null,2));const equal=isDeepStrictEqual(JSON.parse(raw),expected);results.push({width,kind,runId,expected:'Incremental reconstruction equals full live API values after completion without reload',semanticEqual:equal,pass:equal,download,api,bytes:raw.length,sha256:sha(raw),storedMessages:expected.messages.length,keys:Object.keys(expected.vars).length});await save('incremental-replay-results.json',results);console.log(JSON.stringify(results.at(-1)));
}
await save('incremental-replay-errors.json',s.errors);await browser.close();process.exitCode=results.some(x=>!x.pass)?1:0;
