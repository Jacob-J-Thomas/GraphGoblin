// Compile the unmodified exported utility with TypeScript, then measure only prefix calls.
import {pathToFileURL} from 'node:url';import {resolve} from 'node:path';import {createRequire} from 'node:module';import {readFile,writeFile} from 'node:fs/promises';
const parent=pathToFileURL(resolve('apps/web/package.json')).href,req=createRequire(parent),ts=req('typescript');
const src=await readFile('apps/web/src/lib/utils.ts','utf8');const emitted=ts.transpileModule(src,{compilerOptions:{target:ts.ScriptTarget.ES2022,module:ts.ModuleKind.ESNext}}).outputText;
const js=emitted.replace(/from (['"])(.*?)\1/g,(_,q,n)=>'from '+JSON.stringify(import.meta.resolve(n,parent)));
const {prettyJsonPrefix}=await import('data:text/javascript;base64,'+Buffer.from(js).toString('base64'));
