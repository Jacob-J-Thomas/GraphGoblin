import {createRequire} from 'node:module';import {resolve} from 'node:path';import {readFile,writeFile} from 'node:fs/promises';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa'),f=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8')),req=createRequire(resolve('apps/web/package.json')),{chromium}=req('@playwright/test');
const browser=await chromium.launch({channel:'msedge',headless:true}),page=await browser.newPage();const results=[];
for(const width of [1440,390]){await page.setViewportSize({width,height:width===390?844:900});await page.goto('http://127.0.0.1:5720/app/runs/'+f.runs.inference);const toggle=page.getByRole('button',{name:/^Node progress/});await page.getByRole('list',{name:'Timeline',exact:true}).getByRole('button',{name:/node\.finished work\b/}).waitFor();const before=await page.evaluate(()=>document.documentElement.scrollWidth);await toggle.click();const drawer=toggle.locator('..');await drawer.screenshot({path:resolve(dir,`progress-drawer-${width}.png`)});const after=await page.evaluate(()=>document.documentElement.scrollWidth);results.push({width,before,after,pass:before===width&&after===width});}
await writeFile(resolve(dir,'progress-visual.json'),JSON.stringify(results,null,2));await browser.close();console.log(JSON.stringify(results));
