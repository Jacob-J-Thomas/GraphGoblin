import {readFile,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';import {execFileSync} from 'node:child_process';import {createConnection} from 'node:net';import {dir,save,sha} from './gg-qa-e1-r10-support.mjs';
const before=await(await fetch('http://127.0.0.1:5721/info')).json(),stopped=await(await fetch('http://127.0.0.1:5721/stop',{method:'POST'})).json();
const listening=port=>new Promise(ok=>{const s=createConnection({host:'127.0.0.1',port});s.once('connect',()=>{s.destroy();ok(true)});s.once('error',()=>ok(false));s.setTimeout(500,()=>{s.destroy();ok(false)});});
for(let i=0;i<200;i++){if(!(await listening(5720))&&!(await listening(5721)))break;await new Promise(ok=>setTimeout(ok,50));}
const shutdown={before,stopped,apiListening:await listening(5720),controlListening:await listening(5721)};await save('shutdown.json',shutdown);
const audit=JSON.parse(await readFile(resolve(dir,'prior-evidence-audit.json'),'utf8')),base=resolve(dir,'..'),changes=[];for(const x of audit.records){if(!x.exists)continue;try{const now=await readFile(resolve(base,x.round==='round-1'?'':x.round,x.ref));if(sha(now)!==x.sha256)changes.push({round:x.round,ref:x.ref});}catch{changes.push({round:x.round,ref:x.ref,missing:true});}}
const state={head:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),branch:'detached',trackedDiff:execFileSync('git',['diff','--name-only'],{encoding:'utf8'}).trim(),stagedDiff:execFileSync('git',['diff','--cached','--name-only'],{encoding:'utf8'}).trim(),status:execFileSync('git',['status','--short'],{encoding:'utf8'}).trim(),priorEvidenceChanges:changes,shutdown};await save('final-state.json',state);console.log(JSON.stringify({head:state.head,trackedDiff:state.trackedDiff,stagedDiff:state.stagedDiff,priorEvidenceChanges:changes.length,apiListening:shutdown.apiListening,controlListening:shutdown.controlListening}));
