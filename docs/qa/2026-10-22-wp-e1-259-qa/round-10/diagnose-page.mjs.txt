import {readFile} from 'node:fs/promises';import {resolve} from 'node:path';import {dir,api,call,events,save} from './gg-qa-e1-r10-support.mjs';import {launch,frame} from './gg-qa-e1-r10-browser-support.mjs';
const fixtures=JSON.parse(await readFile(resolve(dir,'result-fixtures.json'),'utf8')),s=await launch(),{browser,page}=s,records=[];for(const f of fixtures){await page.goto(api+'/app/runs/'+f.id);await page.getByRole('heading',{name:/^Run /}).waitFor();await page.waitForTimeout(750);await frame(page);const row=await call('/runs/'+f.id),es=await events(f.id),loop=await call('/loops/'+row.loopId+'/versions/'+row.versionId);const screenshot='diagnose-'+f.kind+'.png';await page.screenshot({path:resolve(dir,screenshot)});const actual=await page.getByTestId('run-result').textContent().catch(()=>null),expectedValue=f.kind==='fallback'?es.findLast(e=>e.type==='run.finished').result:row.result,expected=expectedValue===undefined?null:JSON.stringify(expectedValue,null,2).slice(0,4096)+'…';records.push({id:f.id,kind:f.kind,apiLast:row.lastEventSeq,events:es.map(e=>({seq:e.seq,type:e.type})),exitConfig:loop.definition?.nodes.find(n=>n.kind==='exit')?.config,body:(await page.locator('body').textContent()).slice(0,11000),actual,expected,summary:await page.getByTestId('exit-returns').allTextContents(),timeline:await page.getByRole('list',{name:'Timeline',exact:true}).textContent(),screenshot});await save('page-diagnosis.json',records);console.log(JSON.stringify({kind:f.kind,apiLast:row.lastEventSeq,actualChars:actual?.length,summary:records.at(-1).summary}));}await save('diagnosis-errors.json',s.errors);await browser.close();
