import {call,op,mut,assert,save,capture,events,sha} from './gg-qa-e1-r10-support.mjs';
const payload=kind=>({kind,text:'L'.repeat(20000),escaped:'line\n"quote"\\slash\u0000',unicode:'雪 🧪 \ud800',nested:[null,false,0]});
async function finish(name,value,returns=true){const nodes=[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},mut('seed',[op('/vars/value',value)]),{id:'done',kind:'exit',label:'Done',config:returns?{returns:'vars.value'}:{}}].map((n,i)=>({...n,ui:{x:i*260,y:80}}));const definition={schemaVersion:4,name:'QA R10 result '+name+' '+Date.now(),nodes,edges:nodes.slice(1).map((n,i)=>({id:'e'+i,from:{node:nodes[i].id,port:'out'},to:{node:n.id}}))};const {loop}=await call('/loops',{definition});await call('/loops/'+loop.id+'/publish',{});const {run}=await call('/loops/'+loop.id+'/runs',{input:{}});for(let i=0;i<600;i++){const row=await call('/runs/'+run.id);if(row.status==='succeeded')return run.id;assert(!['failed','cancelled','exhausted'].includes(row.status),'fixture failed '+JSON.stringify(row.failure));await new Promise(ok=>setTimeout(ok,50));}throw Error('fixture did not finish');}
async function rowFixture(id,mode,value){const r=await fetch('http://127.0.0.1:5721/row-result',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({id,mode,...(mode==='set'?{value}:{})})});const v=await r.json();assert(r.ok,'row fixture rejected '+JSON.stringify(v));return v;}
const records=[];
for(const kind of['native','fallback','null','absent','row-null-precedence']){const value=kind==='null'||kind==='absent'?null:payload(kind),id=await finish(kind,value,kind!=='absent');let alteration;if(kind==='fallback')alteration=await rowFixture(id,'omit');if(kind==='row-null-precedence')alteration=await rowFixture(id,'set',null);const row=await call('/runs/'+id),es=await events(id),finished=es.findLast(e=>e.type==='run.finished');assert(finished,'finish event missing');if(kind==='fallback')assert(!Object.hasOwn(row,'result')&&Object.hasOwn(finished,'result'),'fallback state not admitted');if(kind==='null')assert(Object.hasOwn(row,'result')&&row.result===null&&finished.result===null,'native null was omitted');if(kind==='absent')assert(!Object.hasOwn(row,'result')&&!Object.hasOwn(finished,'result'),'none unexpectedly returned');if(kind==='row-null-precedence')assert(row.result===null&&typeof finished.result==='object'&&finished.result!==null,'precedence fixture wrong');records.push({kind,id,alteration,rowHasResult:Object.hasOwn(row,'result'),eventHasResult:Object.hasOwn(finished,'result'),finishSeq:finished.seq,rowResult:row.result,eventResult:finished.result,runCapture:await capture('result-'+kind+'-run-api.json.txt',JSON.stringify(row,null,2)),eventsCapture:await capture('result-'+kind+'-events-api.json.txt',JSON.stringify(es,null,2)),before:{run:sha(JSON.stringify(row)),events:sha(JSON.stringify(es)),thread:sha(JSON.stringify(await call('/runs/'+id+'/thread')))}});await save('result-fixtures.json',records);console.log('CREATED '+kind+' '+id);}
