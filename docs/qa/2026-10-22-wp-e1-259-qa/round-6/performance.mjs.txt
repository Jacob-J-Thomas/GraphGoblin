import {readFile} from 'node:fs/promises';import {resolve} from 'node:path';import {dir,api,call,createRun,op,mut,waitNode,advance,events,historySize,assert,save,sha,download} from './gg-qa-e1-r6-support.mjs';import {launch,variables,timeline,reconstruct,frame,settled,go,focus,previewSettled,metrics,measure,prepareClick,trace} from './gg-qa-e1-r6-browser-support.mjs';
const fixtures=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8')),s=await launch(),{browser,page,cdp}=s,results=[];
async function check(id,expected,fn){try{const actual=await fn();results.push({id,expected,actual,pass:true});console.log('PASS '+id);}catch(e){results.push({id,expected,pass:false,error:e.message,focus:await focus(page)});console.log('FAIL '+id+' '+e.message.slice(0,180));}await save('performance-results.json',results);}
await save('browser-environment.json',{browser:browser.version(),node:process.version,server:await(await fetch('http://127.0.0.1:5721/info')).json(),viewports:[{width:1440,height:900},{width:390,height:844}],profilerSampleIntervalUs:1000,standardLongTaskThresholdMs:50});
for(const width of [1440,390]){
 const ids=fixtures.runs[width],adm=fixtures.admission.filter(x=>x.width===width);await page.setViewportSize({width,height:width===390?844:900});
 await check(width+': large cold card','Cold selection shows the request card and no automatic reconstruction',async()=>{await go(page,ids.wide);await reconstruct(page).waitFor();const queued=timeline(page,/run\.queued\b/);await queued.click();await reconstruct(page).waitFor();const seed=timeline(page,/node\.finished seed\b/),click=await prepareClick(page,cdp,seed);const timing=await trace(page,cdp,'large-cold-card-'+width,click,()=>reconstruct(page).waitFor());assert((await reconstruct(page).textContent()).endsWith(String(adm.find(x=>x.name==='wide').seed)),'wrong event');assert(await variables(page).count()===0,'thread reconstructed without asking');const screenshot='large-card-'+width+'.png';await reconstruct(page).scrollIntoViewIfNeeded();await page.screenshot({path:resolve(dir,screenshot)});return {timing,notice:await page.getByText(/This run.s history holds/).textContent(),screenshot};});
 await check(width+': requested reconstruction','Replay and preview run only after Reconstruct; exact prefix and Variables download',async()=>{const click=await prepareClick(page,cdp,reconstruct(page));const timing=await trace(page,cdp,'large-request-'+width,click,()=>previewSettled(page));const expected=(await call('/runs/'+ids.wide+'/thread')).vars;assert(await variables(page).textContent()===JSON.stringify(expected,null,2).slice(0,4096)+'…','preview differs');const dl=await download(page,'Download Variables as JSON',expected,'wide-download-'+width+'.json.txt');const controls=await measure(page,'large-controls-'+width,async()=>{await page.getByRole('button',{name:'Full thread JSON',exact:true}).focus();await page.keyboard.press('Enter');await page.getByTestId('thread-json').waitFor();await page.keyboard.press('Enter');});assert(await page.getByRole('button',{name:'Pause',exact:true}).isEnabled(),'Pause unavailable');return {timing,download:dl,controls};});
 await check(width+': prepared return','Returning to a requested event shows its bounded thread with no request',async()=>{await timeline(page,/run\.queued\b/).click();const seed=timeline(page,/node\.finished seed\b/),click=await prepareClick(page,cdp,seed);const timing=await measure(page,'large-prepared-return-'+width,async()=>{await click();await previewSettled(page);});assert(await reconstruct(page).count()===0,'forgot requested event');return {timing};});
 await check(width+': live selection asks again','A new live selected event returns the card; retained reconstructed value is unchanged',async()=>{await go(page,ids.wide);await reconstruct(page).click();await previewSettled(page);await page.getByRole('button',{name:'Show all of Variables',exact:true}).click();await page.keyboard.press('Shift+Tab');const before=await focus(page);const timing=await measure(page,'large-live-'+width,async()=>{await advance(ids.wide,'notouch');await settled(page,ids.wide);await reconstruct(page).waitFor();});const after=await focus(page),cardText=await reconstruct(page).textContent();await save('large-live-focus-'+width+'.json',{before,after,cardText});await reconstruct(page).click();await previewSettled(page);const expected=(await call('/runs/'+ids.wide+'/thread')).vars;assert(await variables(page).textContent()===JSON.stringify(expected,null,2).slice(0,4096)+'…','live exact prefix differs');return {timing,before,after,cardText};});
 await check(width+': reload and other-run resume','Large log omitted from session storage and replays from zero; ordinary log resumes its cursor',async()=>{const small=await createRun('resume '+width,[mut('seed',[op('/vars',{value:'r'.repeat(10000)})]),waitNode('ask')]);await go(page,small);await previewSettled(page);const smallSeq=(await call('/runs/'+small)).lastEventSeq;await go(page,ids.wide);await reconstruct(page).waitFor();const stored=await page.evaluate(()=>Object.fromEntries(Object.keys(sessionStorage).filter(k=>k.startsWith('graphgoblin-run-events')).map(k=>[k,JSON.parse(sessionStorage.getItem(k))])));const state=Object.values(stored)[0];assert(!state.state.runs[ids.wide],'large run persisted');assert(state.state.runs[small]?.lastSeq===smallSeq,'ordinary run lost');const index=s.requests.length;const timing=await measure(page,'large-reload-'+width,async()=>{await page.reload();await settled(page,ids.wide);await reconstruct(page).waitFor();});const largeRequests=s.requests.slice(index);assert(largeRequests.some(r=>new URL(r.url).searchParams.get('after')==='0'),'large reload did not replay from zero');const cardBefore=await reconstruct(page).textContent();await reconstruct(page).click();await previewSettled(page);const anotherIndex=s.requests.length;await go(page,small);await previewSettled(page);const smallRequests=s.requests.slice(anotherIndex);assert(smallRequests.some(r=>new URL(r.url).searchParams.get('after')===String(smallSeq)),'small run did not resume');return {small,smallSeq,storedRuns:Object.keys(state.state.runs),storedVersion:state.version,largeRequests,smallRequests,cardBefore,timing};});
 for(const kind of ['medium','string'])await check(width+': below threshold '+kind,'Selection paints its note before deferred replay; bounded exact preview, no request card',async()=>{await go(page,ids[kind]);await previewSettled(page);await timeline(page,/run\.queued\b/).click();await previewSettled(page);const target=timeline(page,/node\.finished seed\b/),click=await prepareClick(page,cdp,target),timing=await trace(page,cdp,'below-'+kind+'-'+width,click,()=>previewSettled(page));assert(await reconstruct(page).count()===0,'unexpected large mode');const expected=(await call('/runs/'+ids[kind]+'/thread')).vars;assert(await variables(page).textContent()===JSON.stringify(expected,null,2).slice(0,4096)+'…','below-threshold prefix wrong');return {history:adm.find(x=>x.name===kind),timing};});
 await check(width+': messages window download and live focus','50 latest; Show all 6000; exact download; live hides focused old message with hand-off',async()=>{await go(page,ids.many);await previewSettled(page);const list=page.getByRole('list',{name:'Messages',exact:true});assert(await list.locator(':scope > li').count()===50,'not 50');const initial=await metrics(page),notice=await page.getByText('Showing the last 50 of 6,000 messages.',{exact:true}).textContent();const showAll=await measure(page,'messages-all-'+width,async()=>{await page.getByRole('button',{name:'Show all messages',exact:true}).click();});assert(await list.locator(':scope > li').count()===6000,'not 6000');const dl=await download(page,'Download messages as JSON',(await call('/runs/'+ids.many+'/thread')).messages,'messages-download-'+width+'.json.txt');await page.getByRole('button',{name:'Show all of Message m0 content',exact:true}).click();await page.keyboard.press('Shift+Tab');const before=await focus(page);const live=await measure(page,'messages-live-'+width,async()=>{await advance(ids.many,'notouch');await settled(page,ids.many);await previewSettled(page);});const after=await focus(page);assert(after.label==='Show all messages','old message lost keyboard');assert(await list.locator(':scope > li').count()===50,'did not reset window');await page.keyboard.press('Tab');const afterTab=await focus(page);assert(afterTab.label==='Download messages as JSON','next control wrong');assert(await list.getAttribute('tabindex')===null,'held tabindex left');return {initial,notice,showAll,download:dl,before,live,after,afterTab};});
}
await save('performance-errors.json',s.errors);await browser.close();console.log(JSON.stringify({checks:results.length,failed:results.filter(x=>!x.pass).length,errors:s.errors}));process.exitCode=results.some(x=>!x.pass)?1:0;
