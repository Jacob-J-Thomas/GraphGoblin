import {readFile} from 'node:fs/promises';import {resolve} from 'node:path';import {dir,assert,save} from './gg-qa-e1-r6-support.mjs';import {launch,variables,timeline,reconstruct,go,previewSettled,prepareClick,trace} from './gg-qa-e1-r6-browser-support.mjs';
const fixtures=JSON.parse(await readFile(resolve(dir,'fixtures.json'),'utf8')),s=await launch(),{browser,page,cdp}=s,results=[];await page.setViewportSize({width:390,height:3000});
for(const kind of ['wide','medium']){const id=fixtures.runs[390][kind];await go(page,id);if(kind==='wide')await reconstruct(page).waitFor();else await previewSettled(page);await timeline(page,/run\.queued\b/).click();if(kind==='wide')await reconstruct(page).waitFor();else await previewSettled(page);const click=await prepareClick(page,cdp,timeline(page,/node\.finished seed\b/));const timing=await trace(page,cdp,'visible-'+kind+'-390',click,()=>kind==='wide'?reconstruct(page).waitFor():previewSettled(page));results.push({kind,id,viewport:{width:390,height:3000},timing});await save('visible-paint-results.json',results);console.log('Captured '+kind);}
await save('visible-paint-errors.json',s.errors);await browser.close();
