import {resolve} from 'node:path';import {isDeepStrictEqual} from 'node:util';import {dir,api,call,createRun,op,mut,waitNode,advance,events,assert,save,sha,capture} from './gg-qa-e1-r6-support.mjs';import {launch,variables,timeline,go,settled,frame,previewSettled,focus} from './gg-qa-e1-r6-browser-support.mjs';
const s=await launch(),{browser,context,page}=s,results=[];
async function check(id,expected,fn){try{const actual=await fn();results.push({id,expected,actual,pass:true});console.log('PASS '+id);}catch(e){results.push({id,expected,pass:false,error:e.message,focus:await focus(page)});console.log('FAIL '+id+' '+e.message.slice(0,200));}await save('identifier-results.json',results);}
async function armDownload(oldSeq,newSeq){await page.evaluate(({oldSeq,newSeq})=>{window.__qaIdentifier=undefined;const observer=new MutationObserver(()=>{const box=document.querySelector('[data-reconstructing="true"]');if(!box)return;const section=box.closest('section'),title=section.querySelector('h2')?.textContent,note=section.querySelector('[aria-live]')?.textContent;if(!note?.includes('event '+newSeq))return;observer.disconnect();window.__qaIdentifier={at:performance.now(),title,note,oldSeq,newSeq,selected:document.querySelector('[aria-label="Timeline"] [aria-current]')?.textContent};for(const label of ['Download Variables as JSON','Download Full thread JSON as JSON']){const button=[...section.querySelectorAll('button')].find(b=>b.getAttribute('aria-label')===label);if(!button)throw Error('Download unavailable in pending render: '+label);button.click();}});observer.observe(document,{subtree:true,childList:true,characterData:true,attributes:true,attributeFilter:['data-reconstructing']});},{oldSeq,newSeq});}
for(const width of [1440,390]){
 await page.setViewportSize({width,height:width===390?844:900});
 let fixture;
 await check(width+': pending download identifiers','Shown title and both download filenames/content name A; note names selected uncached B',async()=>{const id=await createRun('identifiers '+width,[mut('seed',[op('/vars',{marker:'A'.repeat(20000)})]),waitNode('ask'),mut('replace',[op('/vars',{marker:'B'.repeat(1000000)})]),waitNode('notouch')]);const apiA=await call('/runs/'+id+'/thread'),oldSeq=(await call('/runs/'+id)).lastEventSeq;await advance(id,'notouch');const es=await events(id),newSeq=es.find(e=>e.type==='node.finished'&&e.nodeId==='replace').seq;fixture={id,oldSeq,newSeq,apiA,es};await go(page,id);await previewSettled(page);await timeline(page,/run\.waiting ask\b/).click();await previewSettled(page);await page.getByRole('button',{name:'Full thread JSON',exact:true}).click();await page.getByRole('button',{name:'Download Full thread JSON as JSON',exact:true}).waitFor();
  const pending=[];const listener=d=>pending.push(d);page.on('download',listener);await armDownload(oldSeq,newSeq);await timeline(page,/node\.finished replace\b/).click();await previewSettled(page);const moment=await page.evaluate(()=>window.__qaIdentifier);assert(moment,'did not capture pending render');assert(moment.title==='Thread at event '+oldSeq,'pending title identifies selected instead of shown event');for(let i=0;i<100&&pending.length<2;i++)await page.waitForTimeout(20);page.off('download',listener);assert(pending.length===2,'expected two downloads');const downloads=[];
  for(const d of pending){const chunks=[];for await(const c of await d.createReadStream())chunks.push(c);const raw=Buffer.concat(chunks),isVars=d.suggestedFilename().includes('-variables-'),expected=isVars?apiA.vars:apiA,filename=`run-${id}-${isVars?'variables':'thread'}-at-${oldSeq}.json`;assert(d.suggestedFilename()===filename,'wrong filename '+d.suggestedFilename());assert(isDeepStrictEqual(JSON.parse(raw),expected),'download does not describe shown event');const name='pending-'+(isVars?'variables':'thread')+'-download-'+width+'.json.txt';downloads.push({suggestedFilename:d.suggestedFilename(),capture:await capture(name,raw),api:await capture(name.replace('download','api'),JSON.stringify(expected,null,2)),semanticEqual:true,byteEqual:raw.equals(Buffer.from(JSON.stringify(expected,null,2)))});}
  return {id,oldSeq,newSeq,moment,downloads,interaction:'MutationObserver clicks the visible public download buttons in the committed pending DOM; no application state is patched',settledTitle:await page.getByRole('heading',{name:'Thread at event '+newSeq,exact:true}).textContent()};
 });
 if(fixture)await check(width+': controlled failed replay identifiers','Failure alert names B and timeline can still select valid A; stored history unchanged',async()=>{const {id,oldSeq,newSeq,apiA,es}=fixture,before={thread:sha(JSON.stringify(await call('/runs/'+id+'/thread'))),events:sha(JSON.stringify(await events(id)))};const bad=structuredClone(es),event=bad.find(e=>e.seq===newSeq);event.patch=[{op:'remove',path:'/vars/absent-qa'}];const altered={seq:newSeq,original:es.find(e=>e.seq===newSeq).patch.map(p=>({op:p.op,path:p.path,valueChars:p.value===undefined?undefined:JSON.stringify(p.value).length})),injectedPatch:event.patch};await save('failure-injection-'+width+'.json',altered);
  const p=await context.newPage();await p.route('**/runs/'+id+'/events**',async route=>{if(!(route.request().headers().accept??'').includes('text/event-stream'))return route.continue();const after=Number(new URL(route.request().url()).searchParams.get('after')??0),body=bad.filter(e=>e.seq>after).map(e=>`id: ${e.seq}\nevent: ${e.type}\ndata: ${JSON.stringify(e)}\n\n`).join('');await route.fulfill({status:200,contentType:'text/event-stream',body:body||': connected\n\n'});});await go(p,id);await timeline(p,/run\.waiting ask\b/).click();await previewSettled(p);await timeline(p,/node\.finished replace\b/).click();const alert=p.getByRole('alert').filter({hasText:'The thread cannot be reconstructed at event '+newSeq});await alert.waitFor();const text=await alert.textContent();assert(text.includes('event '+newSeq),'wrong failed event');await timeline(p,/run\.waiting ask\b/).click();await previewSettled(p);assert(await variables(p).textContent()===JSON.stringify(apiA.vars,null,2).slice(0,4096)+'…','cannot recover A');const screenshot='failure-alert-'+width+'.png';await timeline(p,/node\.finished replace\b/).click();await alert.scrollIntoViewIfNeeded();await p.screenshot({path:resolve(dir,screenshot)});await p.close();const after={thread:sha(JSON.stringify(await call('/runs/'+id+'/thread'))),events:sha(JSON.stringify(await events(id)))};assert(isDeepStrictEqual(before,after),'stored data changed');return {id,oldSeq,newSeq,altered,text,before,after,screenshot,controlledResponseOnly:true};
 });
}
await save('identifier-errors.json',s.errors);await browser.close();console.log(JSON.stringify({checks:results.length,failed:results.filter(x=>!x.pass).length,errors:s.errors}));process.exitCode=results.some(x=>!x.pass)?1:0;
