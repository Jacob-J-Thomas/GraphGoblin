import {readFile,writeFile,readdir,stat} from 'node:fs/promises';import {resolve} from 'node:path';import {createRequire} from 'node:module';import {createHash} from 'node:crypto';
const dir=resolve('docs/qa/2026-10-22-wp-e1-259-qa/round-6'),req=createRequire(resolve('package.json')),prettier=req('prettier'),report=await readFile(resolve(dir,'README.md'),'utf8'),names=await readdir(dir),missing=[],unlinked=[],paths=[],tooLarge=[];
const refs=[...new Set([...report.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)].map(x=>x[1]).filter(x=>!/^https?:|^#/.test(x)))];for(const ref of refs){try{await stat(resolve(dir,ref));}catch{missing.push(ref);}}
for(const name of names){if(name!=='README.md'&&!refs.includes(name))unlinked.push(name);const file=resolve(dir,name),s=await stat(file);if(s.size>1000000)tooLarge.push({name,bytes:s.size});if(!/\.(?:png|jpg|jpeg)$/.test(name)){const raw=await readFile(file,'utf8');if(/(?:[A-Z]:)?[\\/]+Users[\\/]+[A-Za-z0-9._-]+/i.test(raw))paths.push(name);}}
const load=async n=>JSON.parse(await readFile(resolve(dir,n),'utf8')),all=[];for(const n of ['focus-results.json','performance-results.json','identifier-results.json','followup-results.json'])for(const c of await load(n))all.push({source:n,id:c.id,pass:c.pass,error:c.error});const expectedFailures=all.filter(c=>!c.pass),state=await load('final-state.json'),prior=await load('prior-evidence-audit.json'),errors={};for(const n of ['focus-errors.json','performance-errors.json','identifier-errors.json','followup-errors.json','visible-paint-errors.json'])errors[n]=await load(n);
const downloads=[];for(const c of await load('performance-results.json'))if(c.actual?.download)downloads.push({case:c.id,...c.actual.download});for(const c of await load('identifier-results.json'))if(c.actual?.downloads)for(const d of c.actual.downloads)downloads.push({case:c.id,...d});
const mismatches=[];for(const d of downloads){const a=d.download??d.capture,b=d.expected??d.api;if(d.byteEqual===false&&!d.semanticEqual)mismatches.push({case:d.case,reason:'unequal download'});if(a.kept&&b.kept){const x=await readFile(resolve(dir,a.name)),y=await readFile(resolve(dir,b.name));const hash=b=>createHash('sha256').update(b).digest('hex');if(hash(x)!==a.sha256||hash(y)!==b.sha256)mismatches.push({case:d.case,reason:'hash drift'});if(d.byteEqual&&!x.equals(y))mismatches.push({case:d.case,reason:'byte drift'});}}
const manifest=await load('not-kept.json');for(const m of manifest){if(m.bytes<=1000000||!report.includes('NOT-KEPT.md'))mismatches.push({case:m.name,reason:'bad capture policy'});}
const knownFailuresCorrect=expectedFailures.length===2&&expectedFailures.every(c=>c.id.includes('large-history focus loss')&&c.error.includes('BODY'))&&report.includes('QA-259-R6-01')&&report.includes('**FAIL**');
const out={files:names.length,reportReferences:refs.length,missingLinks:missing,unlinkedFiles:unlinked,localUserPathFiles:paths,oversizeKeptFiles:tooLarge,checks:all.length,expectedFailures,knownFailuresCorrect,downloadComparisons:downloads.length,downloadMismatches:mismatches,browserErrors:errors,head:state.head,headMatches:state.head==='689b885a3cdb30a5fbb9bea6b6fa694f053b0646',trackedDiffEmpty:state.trackedDiff===''&&state.stagedDiff==='',priorEvidenceUnchanged:state.priorEvidenceChanges.length===0,priorReferences:prior.references,unaccountedPriorMissing:prior.missingUnmanifested,shutdownClean:!state.shutdown.apiListening&&!state.shutdown.controlListening};
out.integrityPass=!missing.length&&!unlinked.length&&!paths.length&&!tooLarge.length&&knownFailuresCorrect&&!mismatches.length&&Object.values(errors).every(x=>x.length===0)&&out.headMatches&&out.trackedDiffEmpty&&out.priorEvidenceUnchanged&&out.unaccountedPriorMissing===0&&out.shutdownClean;
await writeFile(resolve(dir,'validation.json'),await prettier.format(JSON.stringify(out),{parser:'json',printWidth:100}));console.log(JSON.stringify(out));process.exitCode=out.integrityPass?0:1;
