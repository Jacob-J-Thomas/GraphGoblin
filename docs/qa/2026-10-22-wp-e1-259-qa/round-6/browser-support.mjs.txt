const {SourceMapConsumer}=req(req.resolve('source-map-js',{paths:[req.resolve('vite'),req.resolve('postcss',{paths:[req.resolve('vite')]})]}));const maps=new Map();
export async function launch(){const browser=await chromium.launch({channel:'msedge',headless:true}),context=await browser.newContext();await context.addInitScript(()=>{window.__qaTasks=[];window.__qaStates=[];new PerformanceObserver(l=>window.__qaTasks.push(...l.getEntries().map(t=>({start:t.startTime,duration:t.duration})))).observe({type:'longtask',buffered:true});const snapshot=()=>{const pre=document.querySelector('pre[aria-label="Variables"]'),state={at:performance.now(),selected:document.querySelector('[aria-label="Timeline"] [aria-current="true"]')?.textContent,title:[...document.querySelectorAll('h2')].find(e=>e.textContent.startsWith('Thread at event'))?.textContent,card:[...document.querySelectorAll('button')].find(e=>e.textContent.startsWith('Reconstruct the thread at event'))?.textContent,note:[...document.querySelectorAll('[aria-live="polite"]')].find(e=>e.textContent.startsWith('Reconstructing'))?.textContent,preview:pre?.textContent.slice(0,65)};const old=window.__qaStates.at(-1);if(!old||Object.keys(state).some(k=>k!=='at'&&state[k]!==old[k])){window.__qaStates.push(state);if(window.__qaProbe)performance.mark('qa-state-'+window.__qaStates.length);}};new MutationObserver(snapshot).observe(document,{subtree:true,childList:true,characterData:true,attributes:true,attributeFilter:['aria-current']});});const page=await context.newPage(),errors=[],requests=[];page.setDefaultTimeout(30000);page.on('pageerror',e=>errors.push(e.message));page.on('console',m=>{if(m.type()==='error')errors.push(m.text());});page.on('request',r=>{if(r.url().includes('/events'))requests.push({url:r.url(),lastEventId:r.headers()['last-event-id']});});const cdp=await context.newCDPSession(page);return {browser,context,page,cdp,errors,requests};}

export async function frame(p){await p.evaluate(()=>new Promise(ok=>requestAnimationFrame(()=>requestAnimationFrame(ok))));}
export async function settled(p,id){const run=await call('/runs/'+id);await p.waitForFunction(seq=>[...document.querySelectorAll('[aria-label="Timeline"] button')].some(b=>b.firstElementChild?.textContent==='#'+seq),run.lastEventSeq);await frame(p);}
export async function previewSettled(p){await variables(p).waitFor();await p.waitForFunction(()=>!document.querySelector('pre[aria-label="Variables"]')?.textContent.startsWith('Preparing'));await frame(p);}

export async function focus(p){return p.evaluate(()=>{const e=document.activeElement;return {tag:e.tagName,label:e.getAttribute('aria-label'),role:e.getAttribute('role'),tabindex:e.getAttribute('tabindex'),text:e.tagName==='BUTTON'?e.textContent:undefined};});}
export async function metrics(p){return p.evaluate(()=>({domElements:document.querySelectorAll('*').length,messages:document.querySelectorAll('ol[aria-label="Messages"] > li').length,preChars:document.querySelector('pre[aria-label="Variables"]')?.textContent.length,width:document.documentElement.scrollWidth,viewport:innerWidth}));}
export async function measure(p,name,fn){const since=await p.evaluate(()=>performance.now()),start=performance.now();await fn();await frame(p);const wallMs=performance.now()-start,end=await p.evaluate(()=>performance.now());await p.waitForTimeout(100);return {name,wallMs,since,end,longTasks:await p.evaluate(({since,end})=>window.__qaTasks.filter(t=>t.start>=since&&t.start<end),{since,end}),...await metrics(p)};}
export async function prepareClick(p,cdp,target){await target.scrollIntoViewIfNeeded();await frame(p);const b=await target.boundingBox();assert(b,'No target box');return async()=>{await cdp.send('Input.dispatchMouseEvent',{type:'mousePressed',x:b.x+b.width/2,y:b.y+b.height/2,button:'left',clickCount:1});await cdp.send('Input.dispatchMouseEvent',{type:'mouseReleased',x:b.x+b.width/2,y:b.y+b.height/2,button:'left',clickCount:1});};}
async function original(f){if(!f.url?.startsWith(api)||f.lineNumber<0)return null;const file=basename(new URL(f.url).pathname);let map=maps.get(file);if(!map){const raw=await readFile(resolve('apps/web/dist/assets',file+'.map'),'utf8');map=new SourceMapConsumer(JSON.parse(raw));maps.set(file,map);await save('source-map-provenance.json',{asset:file,sha256:sha(raw),sources:map.sources.filter(s=>s.includes('/lib/utils')||s.includes('ThreadViewer')||s.includes('projections')||s.includes('/domain/'))});}return map.originalPositionFor({line:f.lineNumber+1,column:f.columnNumber});}
export async function trace(p,cdp,id,action,done){const events=[],shots=[],collect=e=>events.push(...e.value),frameCollector=e=>{shots.push(e);void cdp.send('Page.screencastFrameAck',{sessionId:e.sessionId}).catch(()=>{});};cdp.on('Tracing.dataCollected',collect);cdp.on('Page.screencastFrame',frameCollector);const size=p.viewportSize(),timeOrigin=await p.evaluate(()=>performance.timeOrigin);await cdp.send('Page.startScreencast',{format:'png',everyNthFrame:1,maxWidth:size.width,maxHeight:size.height});await cdp.send('Profiler.enable');await cdp.send('Profiler.setSamplingInterval',{interval:1000});await cdp.send('Tracing.start',{categories:'devtools.timeline,blink.user_timing',transferMode:'ReportEvents'});await cdp.send('Profiler.start');const mark='qa-start-'+id,start=await p.evaluate(mark=>{window.__qaProbe=true;window.__qaStates=[];return performance.mark(mark).startTime;},mark);const measured=await measure(p,id,async()=>{await action();if(done)await done();});await p.waitForTimeout(100);const states=await p.evaluate(()=>{window.__qaProbe=false;return window.__qaStates;});const {profile}=await cdp.send('Profiler.stop');await cdp.send('Profiler.disable');const stopped=new Promise(ok=>cdp.once('Tracing.tracingComplete',ok));await cdp.send('Tracing.end');await stopped;cdp.off('Tracing.dataCollected',collect);await cdp.send('Page.stopScreencast');cdp.off('Page.screencastFrame',frameCollector);const marker=events.find(e=>e.name===mark),origin=marker.ts/1000-start,nodes=new Map(profile.nodes.map(n=>[n.id,n])),parents=new Map();for(const n of profile.nodes){n.original=await original(n.callFrame);for(const c of n.children??[])parents.set(c,n.id);}
 let clock=profile.startTime;const serializer=[],replay=[];for(let i=0;i<(profile.samples??[]).length;i++){clock+=(profile.timeDeltas?.[i]??0);let n=nodes.get(profile.samples[i]),ser=false,rep=false;while(n){if(n.original?.source?.endsWith('/lib/utils.ts')&&n.original.line>=41&&n.original.line<=147)ser=true;if(n.original?.source?.includes('/domain/')&&/replay|patch/.test(n.original.source))rep=true;n=nodes.get(parents.get(n.id));}const s={at:clock/1000-origin,ms:(profile.timeDeltas?.[i]??0)/1000};if(ser)serializer.push(s);if(rep)replay.push(s);}
 const paints=events.filter(e=>e.name==='Paint').map(e=>({at:e.ts/1000-origin,ms:(e.dur??0)/1000}));const frameTimes=shots.map(s=>s.metadata.timestamp*1000-timeOrigin),chosen=new Set([0,shots.length-1]);for(const target of [states.find(s=>s.card)?.at,states.find(s=>s.note)?.at,serializer[0]?.at]){if(target===undefined)continue;const before=frameTimes.findLastIndex(t=>t<target),after=frameTimes.findIndex(t=>t>=target);if(before>=0)chosen.add(before);if(after>=0)chosen.add(after);}
 const frames=[];for(const i of [...chosen].filter(i=>i>=0).sort((a,b)=>a-b)){const name=id+'-frame-'+i+'.png';await capture(name,Buffer.from(shots[i].data,'base64'));frames.push({at:frameTimes[i],name});}
 const rawProfile=await capture(id+'.cpuprofile.json.txt',sanitize(JSON.stringify(profile)));const traceEvidence=events.filter(e=>e.name==='Paint'||e.name.startsWith('qa-')||e.name==='EventDispatch'||e.name==='RunTask'&&e.dur>=50000);await capture(id+'.trace-evidence.json.txt',sanitize(JSON.stringify({traceEvents:traceEvidence})));const out={...measured,states,serializer:{samples:serializer.length,sampledMs:serializer.reduce((n,s)=>n+s.ms,0),firstAt:serializer[0]?.at,observations:serializer},replay:{samples:replay.length,sampledMs:replay.reduce((n,s)=>n+s.ms,0),firstAt:replay[0]?.at},paints,frames,frameTimes,profile:rawProfile,traceEvidence:id+'.trace-evidence.json.txt'};await save(id+'.json',out);return out;}
