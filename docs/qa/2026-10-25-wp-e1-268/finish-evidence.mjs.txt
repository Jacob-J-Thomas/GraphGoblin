import { readFile, writeFile, readdir, stat, copyFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { join } from 'node:path';
const root = process.cwd(), qa = join(root, 'docs/qa/2026-10-25-wp-e1-268');
const records = JSON.parse(await readFile(join(qa, 'measurements.json'), 'utf8'));
// Component summaries are portable derivatives; originals remain hashed in NOT-KEPT.md.
for (const variant of ['before-1', 'after-1', 'after-2', 'before-2']) {
  const file = join(qa, `guard-${variant}-summary.json`);
  const entry = JSON.parse(await readFile(file, 'utf8'));
  entry.command[0] = '.tmp/gg-wp-e1-268-guard.mjs';
  await writeFile(file, JSON.stringify(entry, null, 2) + '\n');
}
const matches = [];
for (const trial of [1, 2]) {
  const before = records.find((entry) => entry.name === `guard-before-${trial}`);
  const after = records.find((entry) => entry.name === `guard-after-${trial}`);
  before.cases.forEach((entry, i) => matches.push({ trial, passes: entry.specification.passes, before: entry.equivalence, after: after.cases[i].equivalence,
    equal: entry.equivalence.eventsSha256 === after.cases[i].equivalence.eventsSha256 && entry.equivalence.stateSha256 === after.cases[i].equivalence.stateSha256 }));
}
if (matches.some((entry) => !entry.equal)) throw new Error('Engine-control state mismatch');
await writeFile(join(qa, 'guard-equivalence.json'), JSON.stringify(matches, null, 2) + '\n');
async function sha(path) { return createHash('sha256').update(await readFile(join(root, path))).digest('hex'); }
const active = await sha('packages/engine/dist/run-manager.js'), snapshot = await sha('.tmp/gg-wp-e1-268-final-run-manager.js');
if (active !== snapshot) throw new Error('Final runtime was not restored');
await writeFile(join(qa, 'runtime-restored.json'), JSON.stringify({ active, snapshot, equal: true }, null, 2) + '\n');

// Range inventory is regenerated after final documentation edits; source snapshots stay byte exact.
const ranges = [];
let file;
for (const line of execFileSync('git', ['diff', '--unified=0'], { encoding: 'utf8' }).split('\n')) {
  if (line.startsWith('+++ b/')) file = line.slice(6).trim();
  const match = /^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@/.exec(line);
  if (match) ranges.push({ file, beforeStart: Number(match[1]), beforeCount: Number(match[2] ?? 1), afterStart: Number(match[3]), afterCount: Number(match[4] ?? 1) });
}
for (const file of execFileSync('git', ['ls-files', '--others', '--exclude-standard'], { encoding: 'utf8' }).trim().split('\n').filter((path) => path && !path.startsWith('docs/qa/'))) {
  const text = await readFile(join(root, file), 'utf8');
  ranges.push({ file, added: true, afterStart: 1, afterCount: text.split('\n').length - Number(text.endsWith('\n')) });
}
await writeFile(join(qa, 'touched-ranges.json'), JSON.stringify(ranges, null, 2) + '\n');
await copyFile(new URL(import.meta.url), join(qa, 'finish-evidence.mjs.txt'));
const files = [];
for (const name of await readdir(qa)) {
  if (name === 'files.json') continue;
  const path = join(qa, name), info = await stat(path);
  if (!info.isFile()) throw new Error('Unexpected directory in retained evidence');
  if (info.size > 1_000_000) throw new Error('Oversized retained artifact: ' + name);
  const bytes = await readFile(path);
  if (name.endsWith('.json')) JSON.parse(bytes.toString('utf8'));
  files.push({ name, bytes: bytes.length, sha256: createHash('sha256').update(bytes).digest('hex') });
}
await writeFile(join(qa, 'files.json'), JSON.stringify(files, null, 2) + '\n');
process.stdout.write(JSON.stringify({ guardPairs: matches.length, guardPairsEqual: true, runtimeRestored: true, retainedFiles: files.length, oversized: 0 }) + '\n');
