async function open(p,name='Force stop'){await keyboardTo(p,p.getByRole('button',{name,exact:true}));await p.keyboard.press('Enter');await p.getByRole('dialog').waitFor();}
async function confirm(p,name='Confirm force stop'){
 const response=p.waitForResponse(r=>r.url().endsWith('/system/shutdown')&&r.request().method()==='POST',{timeout:45000});
 await keyboardTo(p,p.getByRole('button',{name,exact:true}));await p.keyboard.press('Enter');
 const r=await response;return{status:r.status(),body:await r.json(),request:JSON.parse(r.request().postData()),controlToken:r.request().headers()['x-graphgoblin-shutdown-token']};
}
