const inherited=Object.fromEntries(['PATH','SystemRoot','WINDIR','ComSpec','PATHEXT','TEMP','TMP','APPDATA','LOCALAPPDATA','USERPROFILE','ProgramFiles','ProgramFiles(x86)','ProgramData','PROCESSOR_ARCHITECTURE'].filter(k=>process.env[k]).map(k=>[k,process.env[k]]));
export const gateways=[];
export async function boot(name,{directory,port=6070,env={},script,runId}={}){
 const dir=directory??await mkdtemp(join(tmpdir(),'gg-qa-c7-319-'+name+'-'));
 const base={...inherited,GG_DATA_DIR:dir,GG_HOST:'127.0.0.1',GG_PORT:String(port),GG_WEB_DIST:resolve('apps/web/dist'),GG_REQUIRE_API_KEY:'true',GG_SHUTDOWN_GRACE_MS:'800',GG_TIMER_POLL_MS:'50',GG_MAX_CONCURRENT_RUNS:'1',...env};
 let key;
 if(!directory){
  const c=spawn(process.execPath,[resolve('apps/api/dist/main.js'),'--create-api-key','Independent QA'],{cwd:root,env:base,windowsHide:true,stdio:['ignore','pipe','pipe']});let text='';c.stdout.on('data',b=>text+=b);c.stderr.on('data',b=>text+=b);await new Promise((r,j)=>{c.on('error',j);c.on('exit',code=>code===0?r():j(new Error(text)));});key=text.split(/\r?\n/).find(s=>s.startsWith('  '))?.trim();ok(key,'CLI did not mint a key');await writeFile(join(dir,'qa-key'),key);
 }else key=(await readFile(join(dir,'qa-key'),'utf8')).trim();
 const args=script?['--input-type=module']:[resolve('apps/api/dist/main.js')];
 const child=spawn(process.execPath,args,{cwd:root,env:base,windowsHide:true,stdio:script?['pipe','pipe','pipe','ipc']:['ignore','pipe','pipe']});
 let logs='';child.stdout.on('data',b=>logs+=b);child.stderr.on('data',b=>logs+=b);
 if(script)child.stdin.end(script);
 const g={name,dir,base,key,child,origin:'http://127.0.0.1:'+port,trace:[],exit:null,closed:false,script};gateways.push(g);
 child.on('exit',(code,signal)=>{g.exit={code,signal};g.closed=true;});
 await until(()=>g.closed?Promise.reject(new Error(logs)):logs.includes(script?'QA_PHASE_READY':'GraphGoblin API listening'),60000);
 g.call=async(path,method='GET',body,token,apiKey=key)=>{
  const r=await fetch(g.origin+path,{method,headers:{authorization:'Bearer '+apiKey,...(body===undefined?{}:{'content-type':'application/json'}),...(token===undefined?{}:{'x-graphgoblin-shutdown-token':token})},...(body===undefined?{}:{body:JSON.stringify(body)}),signal:AbortSignal.timeout(15000)});
  const text=await r.text();let value;try{value=JSON.parse(text);}catch{value=text;}
  const entry={at:new Date().toISOString(),path,method,...(body?{request:body}:{}),...(token?{controlToken:token}:{}),status:r.status,body:value};
  if(path!=='/api-keys')g.trace.push(entry);
  return entry;
 };
 g.flush=async()=>{await save(name+'-gateway.log.txt',logs);await save(name+'-http.json',g.trace);};
 g.close=async()=>{
  if(!g.closed){
   for(const f of ['release','fake-release'])await writeFile(join(dir,f),'release');
   if(script)child.send('close');
   else{await g.call('/system/shutdown','POST',{action:'drain'}).catch(()=>{});const s=await g.call('/system/shutdown').catch(()=>null);if(s?.body.phase==='force-failed')await g.call('/system/shutdown','POST',{action:'force'});}
   await until(async()=>{if(g.closed)return true;if(!script){const s=await g.call('/system/shutdown').catch(()=>null);if(s?.body.phase==='force-failed')await g.call('/system/shutdown','POST',{action:'force'});}return g.closed;},30000);
  }
  await g.flush();
 };
 return g;
}
export function definition(name,work=[]){const nodes=[{id:'start',kind:'trigger',label:'Start',config:{type:'manual'}},...work,{id:'done',kind:'exit',label:'Done',config:{}}];return{schemaVersion:4,name,nodes,edges:nodes.slice(1).map((n,i)=>({id:'edge'+i,from:{node:nodes[i].id,port:'out'},to:{node:n.id}}))};}
export async function publish(g,def){const r=await g.call('/loops','POST',{definition:def});ok(r.status===201,JSON.stringify(r));const id=r.body.loop.id;const p=await g.call('/loops/'+id+'/publish','POST');ok(p.status===200,JSON.stringify(p));return id;}
