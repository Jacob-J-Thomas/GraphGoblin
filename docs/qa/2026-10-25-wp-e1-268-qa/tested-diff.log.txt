+import type { RunEvent } from '@graphgoblin/contracts';
+import type { EventStorePort } from './ports.js';
+
+/** Keep finalization's complete evidence, but bound each synchronous SQLite decode span. */
+export async function finalizationHistory(
+  store: EventStorePort,
+  runId: string,
+): Promise<RunEvent[]> {
+  // Engineering quantum, not a byte bound or latency SLA. 256 keeps the measured small-event
+  // workload's decode spans short without changing event validation or return/recovery evidence.
+  const pageSize = 256;
+  const history: RunEvent[] = [];
+  let after = 0;
+  for (;;) {
+    const page = await store.read(runId, after, pageSize);
+    history.push(...page);
+    if (page.length < pageSize) return history;
+    after = page.at(-1)!.seq;
+    // Promise-only fake/native ports can settle entirely in microtasks; explicitly let timers
+    // and API requests run between pages, just as the executor does between nodes.
+    await new Promise<void>((resolve) => setTimeout(resolve, 0));
+  }
+}

+import type { RunEvent } from '@graphgoblin/contracts';
+
+/** Fresh starts from this run's durable log, independent of replay-seed/display counters. */
+export class FreshEntries {
+  private readonly counts = new Map<string, number>();
+
+  constructor(history: readonly RunEvent[]) {
+    this.record(history);
+  }
+
+  /** Called only after append resolves; a rejected/uncertain append is reconciled next execution. */
+  record(committed: readonly RunEvent[]): void {
+    for (const event of committed) {
+      if (event.type === 'node.started' && event.attempt === 1) {
+        this.counts.set(event.nodeId, this.count(event.nodeId) + 1);
+      }
+    }
+  }
+
+  count(nodeId: string): number {
+    return this.counts.get(nodeId) ?? 0;
+  }
+}

+import { Timings } from './common.mjs';

+/** Supply an observed client to a separate repository, never patch the production repository. */
+export function observeAppendClient(client) {
+  const spans = new Map();
+  const record = (key, ms) => {
+    if (!spans.has(key)) spans.set(key, new Timings());
+    spans.get(key).add(ms);
+  };
+  function decorate(target, transaction = false) {
+    return new Proxy(target, {
+      get(object, key) {
+        const value = Reflect.get(object, key);
+        if (typeof value !== 'function') return value;
+        if (!['transaction', 'execute', 'commit', 'rollback'].includes(key))
+          return value.bind(object);
+        return async (...args) => {
+          const sql = typeof args[0] === 'string' ? args[0] : (args[0]?.sql ?? '');
+          const label =
+            key === 'execute'
+              ? /coalesce\(max\(/i.test(sql)
+                ? 'sequenceLookup'
+                : /^insert/i.test(sql)
+                  ? 'insert'
+                  : /^update/i.test(sql)
+                    ? 'runSequenceUpdate'
+                    : 'otherStatement'
+              : key === 'transaction'
+                ? 'begin'
+                : key;
+          const at = performance.now();
+          try {
+            const result = await value.apply(object, args);
+            return key === 'transaction' ? decorate(result, true) : result;
+          } finally {
+            record((transaction ? 'transaction.' : 'client.') + label, performance.now() - at);
+          }
+        };
+      },
+    });
+  }
+  return {
+    client: decorate(client),
+    reset() {
+      spans.clear();
+    },
+    summary() {
+      return Object.fromEntries([...spans].map(([key, value]) => [key, value.json()]));

+import { createHash } from 'node:crypto';
+
+// Compare controlled runs, retaining identity relationships but excluding clock measurements.
+// Raw event hashes separately check the HTTP replay against the exact stored bytes.
+export function equivalence() {
+  const identities = new Map();
+  const hash = createHash('sha256');
+  function normalize(value, key = '') {
+    if (
+      [
+        'ts',
+        'at',
+        'createdAt',
+        'startedAt',
+        'finishedAt',
+        'receivedAt',
+        'durationMs',
+        'elapsedSeconds',
+      ].includes(key)
+    )
+      return '<clock>';
+    if (typeof value === 'string')
+      return value.replace(/[0-9A-HJKMNP-TV-Z]{26}/g, (id) => {
+        if (!identities.has(id)) identities.set(id, `<id:${identities.size}>`);
+        return identities.get(id);
+      });
+    if (Array.isArray(value)) return value.map((item) => normalize(item));
+    if (value && typeof value === 'object')
+      return Object.fromEntries(
+        Object.keys(value)
+          .sort()
+          .map((name) => [name, normalize(value[name], name)]),
+      );
+    return value;
+  }
+  return {
+    event(event) {
+      const normalized = normalize(event);
+      hash.update(JSON.stringify(normalized) + '\n');
+      return normalized;
+    },
+    finish(thread, run) {
+      return {
+        eventsSha256: hash.digest('hex'),
+        stateSha256: createHash('sha256')
+          .update(
+            JSON.stringify(
+              normalize({ thread, status: run.status, outcome: run.outcome, result: run.result }),
+            ),
+          )
+          .digest('hex'),
+        normalization:
+          'Sorted keys; generated ULIDs retain identity relationships; ts/at/createdAt/startedAt/finishedAt/receivedAt/durationMs/elapsedSeconds are clock observations.',
+      };
+    },
+  };
