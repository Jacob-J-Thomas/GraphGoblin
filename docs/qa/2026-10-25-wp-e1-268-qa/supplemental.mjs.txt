import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { mkdtemp, readFile, writeFile } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
const root=process.cwd(),out=join(root,'docs/qa/2026-10-25-wp-e1-268-qa');
const dep=name=>import(import.meta.resolve(name,pathToFileURL(join(root,'apps/api/package.json')).href));
const native=JSON.parse(await readFile(join(out,'native-restart.json'),'utf8'));
const dir=native.dataDir.replace('%USERPROFILE%',process.env.USERPROFILE);
const {openReadOnlyDatabaseClient,databaseView,SqliteRunRepository}=await dep('@graphgoblin/infrastructure/sqlite');
const client=openReadOnlyDatabaseClient(join(dir,'graphgoblin.db'));
const statements=[];
const observed=new Proxy(client,{get(target,key){const value=Reflect.get(target,key);if(typeof value!=='function')return value;return (...args)=>{if(key==='execute')statements.push(typeof args[0]==='string'?args[0]:args[0].sql);return value.apply(target,args);};}});
const runs=new SqliteRunRepository(databaseView(observed));
const run=await runs.get(native.crash.runId),metadataSql=statements.splice(0);
assert.equal(run.status,'succeeded');assert(metadataSql.length>0);assert(metadataSql.every(sql=>!sql.includes('initial_thread')&&!sql.includes('thread_snapshot')));
const initial=await runs.getInitialThread(run.id),initialSql=statements.splice(0);assert(initial);assert(initialSql.some(sql=>sql.includes('initial_thread')));
const checkpoint=await runs.getThreadCheckpoint(run.id),checkpointSql=statements.splice(0);assert(checkpoint);assert(checkpointSql.some(sql=>sql.includes('thread_snapshot')));
client.close();
await writeFile(join(out,'metadata-control.json'),JSON.stringify({status:'pass',evidence:'Real read-only SQLite client injected into public repository; explicit thread reads preserved',metadataSql,initialSql,checkpointSql,result:{status:run.status,result:run.result},causeReproduced:false},null,2)+'\n');
const temp=await mkdtemp(join(tmpdir(),'gg-qa-e1-268-supplemental-'));
const folders={};
const loader=pathToFileURL(join(tmpdir(),'gg-qa-e1-268-loader.mjs')).href;
const sanitize=x=>x.replaceAll(process.env.USERPROFILE,'%USERPROFILE%').replaceAll(process.env.USERPROFILE.replaceAll('\\','/'),'%USERPROFILE%');
async function runTask(name,args,variant='after'){
 console.log('START',name);const env={...process.env,QA_ROOT:root,QA_VARIANT:variant};if(variant==='baseline')env.NODE_OPTIONS='--loader='+loader;else delete env.NODE_OPTIONS;
 const child=spawn(process.execPath,args,{cwd:root,env,windowsHide:true});let log='';child.stdout.on('data',x=>log+=x);child.stderr.on('data',x=>log+=x);const code=await new Promise((resolve,reject)=>{child.on('error',reject);child.on('close',resolve);});await writeFile(join(out,name+'.log.txt'),sanitize(JSON.stringify({args,variant})+'\n'+log+'\nEXIT='+code+'\n'));const expected=name.startsWith('append-paced-')&&!name.endsWith('-pair')?1:0;assert.equal(code,expected,name);console.log('END',name,code);
}
for(const workload of ['context','append-paced'])for(const [side,variant]of [['before','baseline'],['after','after']]){
 const name=workload+'-'+side,output=join(temp,name);folders[name]=output;
 const args=['--experimental-import-meta-resolve','tooling/perf/measure.mjs',workload==='context'?'context':'append','--passes='+(workload==='context'?'100':'10'),'--items='+(workload==='context'?'2':'100'),'--profile=inspector','--port=6190','--store-prefix=gg-qa-e1-268-','--out='+output];if(workload!=='context')args.push('--item-pause-ms=10');
 await runTask(name,args,variant);const raw=JSON.parse(await readFile(join(output,'raw-summary.json'),'utf8'));assert(raw.cases[0].correctness.ok);if(workload!=='context'){assert(raw.cases[0].correctness.replayOk);assert.equal(raw.cases[0].recovery.beforeHash,raw.cases[0].recovery.afterHash);}await writeFile(join(out,name+'-summary.json'),sanitize(JSON.stringify(raw,null,2)));
}
for(const name of ['context','append-paced'])await runTask(name+'-pair',['--experimental-import-meta-resolve','tooling/perf/compare.mjs',join(folders[name+'-before'],'raw-summary.json'),join(folders[name+'-after'],'raw-summary.json'),join(out,name+'-equivalence.json')]);
await writeFile(join(out,'supplemental-location.json'),sanitize(JSON.stringify({temp,folders},null,2)+'\n'));
console.log('SUPPLEMENTAL_COMPLETE');
