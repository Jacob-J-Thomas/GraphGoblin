async function poll(read, ok, label, timeout = 10000) {
  const stop = Date.now() + timeout;
  let value;
  do {
    value = await read();
    if (ok(value)) return value;
    await delay(100);
  } while (Date.now() < stop);
  throw new Error(`${label}: last observation ${JSON.stringify(value)}`);
}

async function api(path, method = 'GET', body) {
  const res = await fetch(origin + path, {
    method,
    headers: body === undefined ? {} : { 'content-type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body),
    signal: AbortSignal.timeout(5000),
  });
  return { status: res.status, data: await res.json() };
}

function loopDef(name, { variables, description, inference } = {}) {
  return {
    schemaVersion: 4,
    name,
    ...(description ? { description } : {}),
    ...(variables ? { variables } : {}),
    nodes: [
      { id: 'start', kind: 'trigger', label: 'Start', config: { type: 'manual' }, ui: { x: 0, y: 80 } },
      inference
        ? { id: 'approve', kind: 'inference', label: 'Check', config: inference, ui: { x: 290, y: 80 } }
        : { id: 'approve', kind: 'wait', label: 'Approve', config: { mode: 'input', prompt: 'Approve?' }, ui: { x: 290, y: 80 } },
      { id: 'done', kind: 'exit', label: 'Done', config: {}, ui: { x: 570, y: 80 } },
    ],
    edges: [
      { id: 'e1', from: { node: 'start', port: 'out' }, to: { node: 'approve' } },
      { id: 'e2', from: { node: 'approve', port: 'out' }, to: { node: 'done' } },
    ],
  };
}

let seq = 0;
async function seed(name, options) {
  seq += 1;
  const created = await api('/loops', 'POST', { definition: loopDef(`${name} ${seq}`, options) });
  assert.equal(created.status, 201, JSON.stringify(created.data));
  return { id: created.data.loop.id, name: `${name} ${seq}` };
}

/** The draft as the API returns it and as the libsql row holds it; they must agree. */
async function stored(loop) {
  const res = await api('/loops/' + loop.id);
  assert.equal(res.status, 200);
  assert.ok(res.data.draft, 'a draft exists');
  const db = await fixture.container.handle.client.execute({
    sql: 'SELECT definition FROM loop_versions WHERE id = ?',
    args: [res.data.draft.id],
  });
  const sql = JSON.parse(db.rows[0].definition);
  assert.deepEqual(sql, res.data.draft.definition, 'API GET and SQL row agree');
  return { api: res.data.draft.definition, sql, current: res.data.current ?? null };
}

function observePuts(page, loop) {
  const seen = [];
  page.on('requestfinished', async (req) => {
    if (req.method() !== 'PUT' || new URL(req.url()).pathname !== `/loops/${loop.id}/draft`) return;
    const res = await req.response();
    seen.push({ at: new Date().toISOString(), status: res?.status(), sent: req.postDataJSON() });
  });
  return seen;
}

const code = (scope, label) => scope.locator('.cm-content').and(scope.getByLabel(label, { exact: true }));

/** Replace the editor's whole text in one input (select all, then insert over the selection). */
async function typeCode(page, target, text) {
  await target.click();
  await page.keyboard.press('ControlOrMeta+A');
  if (text === '') await page.keyboard.press('Backspace');
  else await page.keyboard.insertText(text);
}

/** CodeMirror virtualizes long documents: read the whole text through a synthetic copy event. */
async function readCode(page, target) {
  await target.focus();
  await page.keyboard.press('ControlOrMeta+A');
  const text = await target.evaluate((el) => {
    const clipboardData = new DataTransfer();
    el.dispatchEvent(new ClipboardEvent('copy', { clipboardData, bubbles: true, cancelable: true }));
    return clipboardData.getData('text/plain');
  });
  await page.keyboard.press('ArrowRight');
  return text;
}

async function fieldState(target) {
  return target.evaluate((el) => ({
    invalid: el.getAttribute('aria-invalid'),
    label: el.getAttribute('aria-label'),
    description: (el.getAttribute('aria-describedby') ?? '')
      .split(/\s+/)
      .map((id) => document.getElementById(id)?.textContent ?? '')
      .filter(Boolean)
      .join(' | '),
    focused: el === document.activeElement || el.contains(document.activeElement),
  }));
}

const activeLabel = (page) =>
  page.evaluate(() => {
    const el = document.activeElement;
    return el ? { tag: el.tagName, label: el.getAttribute('aria-label'), text: el.textContent?.slice(0, 80) } : null;
  });

async function saveStatus(page) {
  const draft = page.getByTestId('draft-state');
  const held = page.getByTestId('held-input-state');
  return {
    whole: (await page.getByTestId('save-state').innerText()).replace(/\s+/g, ' ').trim(),
    draft: (await draft.count()) ? (await draft.innerText()).trim() : null,
    held: (await held.count()) ? (await held.innerText()).replace(/\s+/g, ' ').trim() : null,
  };
}

/** One case: named checks, the first failed check stops it; everything observed is kept. */
function makeCase(id, issue, title, expected) {
  const record = { id, issue, title, expected, steps: [], checks: [], screenshots: [], verdict: 'RUNNING' };
  return {
    record,
    step(text) {
      record.steps.push(text);
    },
    check(label, actual, pass) {
      const ok = typeof pass === 'function' ? Boolean(pass(actual)) : Boolean(pass);
      record.checks.push({ label, actual, pass: ok });
      if (!ok) throw new Error(`CHECK FAILED: ${label}: ${JSON.stringify(actual)}`);
    },
    equal(label, actual, expectedValue) {
      let ok = true;
      try {
        assert.deepStrictEqual(actual, expectedValue);
      } catch {
        ok = false;
      }
      record.checks.push({ label, actual, expected: expectedValue, pass: ok });
      if (!ok) throw new Error(`CHECK FAILED: ${label}`);
    },
    note(label, value) {
      record.checks.push({ label, actual: value, observation: true });
    },
    async shot(page, name) {
      const file = `${id}-${name}.png`;
      await page.screenshot({ path: join(out, file) });
      record.screenshots.push(file);
    },
  };
}

async function case527Positive(page, c) {
  const loop = await seed('QA 527 rich schema', { variables: { topic: TOPIC, other: OTHER }, description: 'kept' });
  const puts = observePuts(page, loop);
  c.step('Seed loop with Variables {topic, other} and description; open the built editor; show loop panel.');
  await openEditor(page, loop);
  await showLoopPanel(page);
  const first = code(page, 'Variables value 1');
  const text = JSON.stringify(RICH, null, 2);
  c.step('Type a rich, admitted JSON Schema (constructor, prototype, __proto, proto__, toString, empty and numeric member names, dotted, unicode, deep arrays, a __proto__ mention in a string) into Variables value 1.');
  await typeCode(page, first, text);
  const state = await fieldState(first);
  c.check('Value not marked invalid', state.invalid, (v) => v !== 'true');
  const saved = await poll(() => stored(loop), (row) => row.api.variables.topic.title === 'Rich', 'rich schema saved', 10000);
  c.equal('SQL row variables.topic equals the entered JSON (whole object)', saved.sql.variables.topic, RICH);
  c.equal('neighbouring Variables member unchanged', saved.sql.variables.other, OTHER);
  c.check('description unchanged', saved.sql.description, (v) => v === 'kept');
  const last = puts.at(-1);
  c.check('last PUT status 200', last?.status, (v) => v === 200);
  c.equal('PUT payload variables equal the entered JSON plus neighbour', last.sent.definition.variables, { topic: RICH, other: OTHER });
  c.note('PUT count', puts.length);
  c.check('no held-input segment for an accepted value', (await saveStatus(page)).held, (v) => v === null);
  await c.shot(page, 'saved');
  c.step('Reload; read the whole Value text through the editor copy event.');
  await page.reload();
  await showLoopPanel(page);
  const reloaded = JSON.parse(await readCode(page, code(page, 'Variables value 1')));
  c.equal('reloaded Value 1 equals the entered JSON', reloaded, RICH);
  c.equal('reloaded Value 2 equals neighbour', JSON.parse(await readCode(page, code(page, 'Variables value 2'))), OTHER);
  c.equal('own member names preserved exactly (properties)', Object.keys(reloaded.properties).sort(), Object.keys(RICH.properties).sort());
  return { puts };
}

async function case527Refusal(page, c) {
  const loop = await seed('QA 527 own proto refusal', { variables: { topic: TOPIC, other: OTHER }, description: 'kept' });
  const puts = observePuts(page, loop);
  const before = await stored(loop);
  await openEditor(page, loop);
  await showLoopPanel(page);
  const first = code(page, 'Variables value 1');
  const heldText = '{"type":"object","description":"held","properties":{"__proto__":{"type":"number"},"name":{"type":"string"}}}';
  c.step('Type a schema with an own "__proto__" member under properties into Variables value 1.');
  await typeCode(page, first, heldText);
  await page.getByText(/This member name cannot be saved/).first().waitFor();
  const held = await fieldState(first);
  c.check('Value aria-invalid=true', held.invalid, (v) => v === 'true');
  c.check('accessible description names properties.__proto__ and the remedy', held.description, (d) => d.includes(`properties.__proto__: ${MEMBER_MSG}`));
  c.check('Value keeps the exact typed text', await readCode(page, first), (t) => t === heldText);
  c.check('Discard action offered beside the message', await page.getByRole('button', { name: 'Discard the unparsed text' }).count(), (n) => n === 1);
  await delay(1500);
  c.check('no draft PUT for the refused value (1.5 s > 600 ms debounce)', puts.length, (n) => n === 0);
  c.equal('server+SQL draft unchanged while held', (await stored(loop)).sql, before.sql);
  const status = await saveStatus(page);
  c.note('status while held', status);
  c.check('held segment shown', status.held, (v) => v?.includes(HELD));
  c.check('draft segment says Draft saved (not All changes saved)', status.draft, (v) => v === 'Draft saved');
  await c.shot(page, 'held');

  c.step('Variants: own __proto__ at top level, inside an array item, and deep inside items.');
  const variants = [
    ['{"__proto__":{"type":"string"},"type":"object"}', '__proto__'],
    ['{"anyOf":[{"type":"string"},{"__proto__":{}}]}', 'anyOf.1.__proto__'],
    ['{"properties":{"a":{"items":[{"properties":{"__proto__":{"const":1}}}]}}}', 'properties.a.items.0.properties.__proto__'],
  ];
  for (const [text, path] of variants) {
    await typeCode(page, first, text);
    const s = await poll(() => fieldState(first), (v) => v.description.includes(`${path}: ${MEMBER_MSG}`), `variant ${path}`, 4000);
    c.check(`variant ${path}: invalid and named`, s, (v) => v.invalid === 'true');
  }
  await delay(1000);
  c.check('still no PUT after variants', puts.length, (n) => n === 0);

  c.step('Neighbour edit (Variables value 2) while Value 1 is held.');
  await typeCode(page, first, heldText);
  const neighbour = { type: 'integer', description: 'neighbour saved independently' };
  await typeCode(page, code(page, 'Variables value 2'), JSON.stringify(neighbour));
  const afterNeighbour = await poll(() => stored(loop), (row) => row.api.variables.other.type === 'integer', 'neighbour saved', 10000);
  c.equal('SQL variables: last complete topic + new neighbour', afterNeighbour.sql.variables, { topic: TOPIC, other: neighbour });
  const sent = puts.at(-1);
  c.check('neighbour PUT 200', sent?.status, (v) => v === 200);
  c.equal('PUT payload variables', sent.sent.definition.variables, { topic: TOPIC, other: neighbour });
  c.check('PUT payload carries no __proto__ text and no held description', JSON.stringify(sent.sent), (t) => !t.includes('__proto__') && !t.includes('"held"'));
  c.check('Value 1 still holds the exact text', await readCode(page, first), (t) => t === heldText);

  c.step('Publish while held.');
  await page.getByRole('button', { name: 'Publish', exact: true }).click();
  const alert = page.getByRole('alert').filter({ hasText: 'Publish failed' });
  await alert.waitFor();
  c.note('publish refusal text', (await alert.innerText()).replace(/\s+/g, ' '));
  c.check('nothing published', (await stored(loop)).current, (v) => v === null);
  await c.shot(page, 'publish-refused');

  c.step('Reload: window-only text is gone, the last complete value and the neighbour remain.');
  await page.reload();
  await showLoopPanel(page);
  c.equal('reloaded Value 1 = last complete topic', JSON.parse(await readCode(page, code(page, 'Variables value 1'))), TOPIC);
  c.equal('reloaded Value 2 = neighbour', JSON.parse(await readCode(page, code(page, 'Variables value 2'))), neighbour);
  c.check('no held segment after reload', (await saveStatus(page)).held, (v) => v === null);
  return { puts };
}

async function case527Correct(page, c) {
  const loop = await seed('QA 527 correct and discard', { variables: { topic: TOPIC, other: OTHER } });
  const puts = observePuts(page, loop);
  await openEditor(page, loop);
  await showLoopPanel(page);
  const first = code(page, 'Variables value 1');
  const heldText = '{"type":"object","properties":{"__proto__":{"type":"number"},"name":{"type":"string"}}}';
  c.step('Hold a value with own __proto__, then Discard.');
  await typeCode(page, first, heldText);
  await page.getByText(/This member name cannot be saved/).first().waitFor();
  await page.getByRole('button', { name: 'Discard the unparsed text' }).click();
  await delay(300);
  const afterDiscard = await fieldState(first);
  c.check('focus returns to the Value editor after Discard', afterDiscard.focused, (v) => v === true);
  c.note('active element after discard', await activeLabel(page));
  c.check('Value no longer invalid', afterDiscard.invalid, (v) => v !== 'true');
  c.equal('Value text restored to the last complete value', JSON.parse(await readCode(page, first)), TOPIC);
  const status = await saveStatus(page);
  c.check('held segment gone after discard', status.held, (v) => v === null);
  c.check('draft label back to All changes saved', status.draft, (v) => v === 'All changes saved');
  await delay(1000);
  c.check('no PUT from hold+discard', puts.length, (n) => n === 0);

  c.step('Hold again, then correct by renaming the member; the whole corrected object saves.');
  await typeCode(page, first, heldText);
  await page.getByText(/This member name cannot be saved/).first().waitFor();
  const corrected = { type: 'object', properties: { proto_name: { type: 'number' }, name: { type: 'string' } } };
  await typeCode(page, first, JSON.stringify(corrected));
  const saved = await poll(() => stored(loop), (row) => row.api.variables.topic.type === 'object', 'corrected saved', 10000);
  c.equal('SQL variables after correction', saved.sql.variables, { topic: corrected, other: OTHER });
  c.equal('last PUT payload variables', puts.at(-1)?.sent.definition.variables, { topic: corrected, other: OTHER });
  await poll(() => saveStatus(page), (s) => s.held === null && s.draft === 'All changes saved', 'status settles after correction', 6000);
  await page.reload();
  await showLoopPanel(page);
  c.equal('reloaded corrected Value', JSON.parse(await readCode(page, code(page, 'Variables value 1'))), corrected);
  return { puts };
}

async function inspectLive(page, selector) {
  const session = await page.context().newCDPSession(page);
  try {
    const { root } = await session.send('DOM.getDocument', { depth: -1 });
    const { nodeId } = await session.send('DOM.querySelector', { nodeId: root.nodeId, selector });
    if (!nodeId) return { found: false };
    const { node } = await session.send('DOM.describeNode', { nodeId });
    const { nodes } = await session.send('Accessibility.getPartialAXTree', { nodeId, fetchRelatives: true });
    const match = nodes.find((entry) => entry.backendDOMNodeId === node.backendNodeId);
    return {
      found: true,
      ignored: match?.ignored ?? null,
      ignoredReasons: match?.ignoredReasons ?? null,
      live: match?.properties?.find((p) => p.name === 'live')?.value?.value ?? null,
      atomic: match?.properties?.find((p) => p.name === 'atomic')?.value?.value ?? null,
      name: match?.name?.value ?? null,
      role: match?.role?.value ?? null,
    };
  } finally {
    await session.detach();
  }
}

async function case526BlockedReload(page, c, context) {
  const old = await context.newPage();
  await old.goto(origin + '/app/loops');
  await old.evaluate(() => new Promise((ok, fail) => {
    const req = indexedDB.open('graphgoblin', 1);
    req.onupgradeneeded = () => req.result.createObjectStore('drafts');
    req.onsuccess = () => { window.__olderTab = req.result; ok(); };
    req.onerror = () => fail(req.error);
  }));
  c.step('Device storage blocked by an older tab; hold unfinished JSON; save a valid neighbour; reload while still blocked.');
  const loop = await seed('QA 526 blocked reload', { variables: { topic: TOPIC }, description: 'before' });
  const puts = observePuts(page, loop);
  await page.bringToFront();
  await openEditor(page, loop);
  const notice = page.getByText('Changes are not kept on this device');
  await notice.first().waitFor();
  await showLoopPanel(page);
  await typeCode(page, code(page, 'Variables value 1'), '{"type":');
  await page.getByTestId('held-input-state').waitFor();
  await page.getByLabel('Description', { exact: true }).fill('neighbour before blocked reload');
  await poll(() => stored(loop), (row) => row.api.description === 'neighbour before blocked reload', 'neighbour saved', 10000);
  c.check('before reload: Draft saved + held segment', await poll(() => saveStatus(page), (v) => v.draft === 'Draft saved', 'draft saved', 6000), (v) => v.held?.includes(HELD));
  await page.reload();
  await page.getByRole('heading', { name: loop.name, exact: true }).waitFor();
  await notice.first().waitFor();
  c.check('device notice still shown after reload (no device copy available)', await notice.first().isVisible(), (v) => v === true);
  await showLoopPanel(page);
  c.equal('Value 1 = last complete server value (held text gone)', JSON.parse(await readCode(page, code(page, 'Variables value 1'))), TOPIC);
  c.check('description from server', await page.getByLabel('Description', { exact: true }).inputValue(), (v) => v === 'neighbour before blocked reload');
  const after = await poll(() => saveStatus(page), (v) => v.draft !== null, 'status present', 6000);
  c.note('status after blocked reload', after);
  c.check('no held segment and no saved-to-server overclaim beyond the draft', after, (v) => v.held === null);
  await c.shot(page, 'blocked-reloaded');
  c.step('Observation: 480 px wide, held input again.');
  await page.setViewportSize({ width: 480, height: 800 });
  await typeCode(page, code(page, 'Variables value 1'), '{"type":').catch(async () => { await showLoopPanel(page); await typeCode(page, code(page, 'Variables value 1'), '{"type":'); });
  await page.getByTestId('held-input-state').waitFor();
  const narrow = await page.getByTestId('save-state').evaluate((el) => {
    const r = el.getBoundingClientRect();
    return { viewport: window.innerWidth, docScroll: document.documentElement.scrollWidth, left: Math.round(r.left), right: Math.round(r.right), clientWidth: el.clientWidth, scrollWidth: el.scrollWidth,
      parts: [...el.querySelectorAll('[data-testid]')].map((p) => { const b = p.getBoundingClientRect(); return { id: p.dataset.testid, top: Math.round(b.top), left: Math.round(b.left), right: Math.round(b.right) }; }) };
  });
  c.note('480 px layout', narrow);
  await c.shot(page, 'narrow-480');
  await old.close();
  return { puts };
}
