import {spawn} from 'node:child_process';
import {open,writeFile} from 'node:fs/promises';
import {join,resolve} from 'node:path';
import assert from 'node:assert/strict';
const out=resolve('docs/qa/2026-10-24-wp-g1-255-qa/round-2');
assert.match(process.version,/^v22\./);
const commands=[['typecheck','typecheck'],['lint','lint'],['layers','check:layers'],['deps','check:deps'],['tokens','check:tokens'],['format','format:check'],['web-coverage','--filter @graphgoblin/web test:coverage --maxWorkers=1']];
const save=(name,value)=>writeFile(join(out,name),JSON.stringify(value,null,2)+'\n');
for(const [name,args] of commands){
 const command='pnpm.cmd '+args,started=new Date().toISOString();
 await save('active-gate.json',{name,command,status:'running',started,qaProcesses:'Edge closed; gateway process exited; sequential gates only'});
 const log=await open(join(out,'gate-'+name+'.log.txt'),'w');
 const child=spawn(process.env.ComSpec,['/d','/s','/c',command],{env:process.env,stdio:['ignore',log.fd,log.fd],windowsHide:true});
 const result=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('exit',(exitCode,signal)=>resolve({exitCode,signal}));});await log.close();
 await save('gate-'+name+'-result.json',{name,command,started,completed:new Date().toISOString(),...result,qaProcesses:'Edge closed; gateway process exited; sequential gates only'});
 console.log(name+' '+result.exitCode);
}
await save('active-gate.json',{status:'complete',completed:new Date().toISOString()});
