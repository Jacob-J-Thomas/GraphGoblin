import {readFile,writeFile}from'node:fs/promises';
const ctl=async(path,body={})=>(await fetch('http://127.0.0.1:6151'+path,{method:'POST',body:JSON.stringify(body)})).json();
const c=await ctl('/connection'),r=JSON.parse(await readFile('.tmp/qa186/run.json','utf8'));
const oracle=await ctl('/oracle',{runId:r.runId,at:[0,128,129,r.total+100]});
await writeFile('.tmp/qa186/restart-connection.json',JSON.stringify(c));
await writeFile('docs/qa/2026-10-24-wp-e1-186-r2-qa/restart-before.json',JSON.stringify({pid:c.pid,runId:r.runId,oracle},null,2));
console.log(await ctl('/stop'));
