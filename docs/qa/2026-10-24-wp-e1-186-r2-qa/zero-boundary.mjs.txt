import{readFile,writeFile}from'node:fs/promises';
const ctl=async(path,body={})=>(await fetch('http://127.0.0.1:6151'+path,{method:'POST',body:JSON.stringify(body)})).json();
const c=await ctl('/connection'),r=JSON.parse(await readFile('.tmp/qa186/run.json','utf8'));
