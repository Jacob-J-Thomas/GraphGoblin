+import type { ContextThread } from '@graphgoblin/contracts';
+import { replayThread, replayStateAt, type ReplayState } from '@graphgoblin/domain';
+import { z } from 'zod';
+import type { Container } from './container.js';
+
+const Cursor = z.strictObject({
+  runId: z.string(),
+  seq: z.number().int().nonnegative().safe(),
+  direction: z.enum(['older', 'newer']),
+});
+type Cursor = z.infer<typeof Cursor>;
+
+export function historyCursor(value: Cursor): string {
+  return Buffer.from(JSON.stringify(value)).toString('base64url');
+}
+
+export function readHistoryCursor(cursor: string, runId: string): Cursor {
+  try {
+    if (cursor.length > 256 || !/^[A-Za-z0-9_-]+$/.test(cursor)) throw new Error('invalid');
+    const value = Cursor.parse(JSON.parse(Buffer.from(cursor, 'base64url').toString('utf8')));
+    if (value.runId !== runId || historyCursor(value) !== cursor) throw new Error('invalid');
+    return value;
+  } catch {
+    throw Object.assign(new Error('Choose a cursor returned by this run history.'), {
+      code: 'INVALID_CURSOR',
+      statusCode: 400,
+    });
+  }
+}
+
+/** Historical reads carry the domain replay state across 128-row pages, never the whole log. */
+export async function historicalThread(
+  container: Container,
+  runId: string,
+  at: number,
+): Promise<ContextThread | undefined> {
+  const { repos } = container;
+  let thread = await repos.runs.getInitialThread(runId);
+  if (!thread) return undefined;
+  let after = 0;
+  let state: ReplayState = { strict: false };
+  const checkpoint = await repos.runs.getThreadCheckpoint(runId);
+  if (checkpoint?.replayState && checkpoint.seq <= at) {
+    const [boundary] = await repos.events.read(runId, checkpoint.seq - 1, 1);
+    if (boundary?.seq === checkpoint.seq) {
+      thread = checkpoint.thread;
+      after = checkpoint.seq;
+      state = checkpoint.replayState;
+    }
+  }
+  while (after < at) {
+    const page = await repos.events.read(runId, after, 128);
+    if (!page.length) break;
+    thread = replayThread(thread, page, at, state);
+    state = replayStateAt(page, at, state);
+    after = page.at(-1)!.seq;
+    // Give cancellation and other requests a turn between historical pages.
+    await new Promise<void>((resolve) => setImmediate(resolve));
+  }
+  return thread;
+}

+## Round 2: bounded browser history (2026-10-24)
+
+The browser half retains one window per run, at most 512 rows and 1,048,576 estimated UTF-16 row-JSON bytes, persisted with its independent SSE acknowledgement, mode and opaque paging boundaries. Count protects tiny events; bytes protect payload-heavy runs. A 512-row window covers about 40 passes of the measured cyclic workload; 128-event history reads align with the server's bounded replay pages. Oversized events retain a small sequence/type/description marker (ordinary accessible names are preserved, with the existing 4,096-character preview budget for pathological summaries); one exact explicitly requested payload is transient, never cached. Storage failures are nonfatal. Device cache version 2 discards the old cache; server history and stored shape are unchanged.
+
+History mode freezes the visible window while live acknowledgement continues. Older/newer pages join by sequence and trim the opposite side, invalidating any cursor no longer at the retained edge. Numeric seeks acquire fresh positions; subsequent paging passes run-bound opaque cursors unchanged. Returning live catches the acknowledged suffix before switching modes, bounded to eight additional pages per explicit action, with a visible retry reason if the history keeps outrunning that action.
