async function exactDownload(p,event,label){
 const disclosure=p.getByRole('button',{name:'Exact recorded event #'+event.seq,exact:true});if(await disclosure.getAttribute('aria-expanded')!=='true')await disclosure.click();
 const exact=p.getByTestId('recorded-event-json');await exact.waitFor();
 const more=p.getByRole('button',{name:'Show all of Exact recorded event',exact:true});if(await more.isVisible())await more.click();
 assert.deepEqual(JSON.parse(await exact.innerText()),event);
 const waiting=p.waitForEvent('download');await p.getByRole('button',{name:'Download Exact recorded event as JSON',exact:true}).click();const download=await waiting;
 const path='.tmp/qa186r2/'+label+'-download.json';await download.saveAs(path);const raw=await readFile(path,'utf8');assert.deepEqual(JSON.parse(raw),event);
 const result={seq:event.seq,bytes:Buffer.byteLength(raw),sha256:createHash('sha256').update(raw).digest('hex'),filename:download.suggestedFilename()};
 if(Buffer.byteLength(raw)<=1000000)await writeFile(out+'/'+label+'-download.json',raw);else await writeFile(out+'/'+label+'-omitted.json',result);
 return result;
}
