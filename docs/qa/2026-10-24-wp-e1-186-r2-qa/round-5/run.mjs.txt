async function run(name, cmd, cwd = root) {
  const start = new Date();
  let log = `Node ${process.version}\nCommand: ${cmd}\nCwd: ${cwd}\nStarted: ${start.toISOString()}\n`;
  console.log(`START ${name}: ${cmd}`);
  await writeFile(join(out, name + '.log.txt'), clean(log));
  const child = spawn('cmd.exe', ['/d', '/s', '/c', cmd], { cwd, env: { ...env, PLAYWRIGHT_JSON_OUTPUT_NAME: join(out, name + '.json'), GG_THREAD_EVIDENCE: join(out, name + '-thread.json') }, windowsHide: true, windowsVerbatimArguments: true, stdio: ['ignore', 'pipe', 'pipe'] });
  for (const stream of [child.stdout, child.stderr]) { stream.setEncoding('utf8'); stream.on('data', (s) => { log += s; appendFileSync(join(out, name + '.log.txt'), clean(s)); }); }
  const code = await new Promise((done) => { child.once('error', (e) => { log += String(e); done(-1); }); child.once('exit', done); });
  const elapsedSeconds = (Date.now() - start.getTime()) / 1000;
  log += `\nExit code: ${code}\nElapsed seconds: ${elapsedSeconds}\n`;
  await writeFile(join(out, name + '.log.txt'), clean(log));
  const result = { name, command: cmd, cwd: clean(cwd), code, elapsedSeconds, started: start.toISOString(), storeLock: log.includes('ERR_PNPM_STORE_DIR_OPEN_OPERATION_LOCK') };
  await appendFile(join(out, 'commands.jsonl'), JSON.stringify(result) + '\n');
  console.log(JSON.stringify(result));
  return result;
}
