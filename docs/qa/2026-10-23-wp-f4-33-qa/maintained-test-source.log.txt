function cycle(worker: SessionPolicy, reviewer: SessionPolicy = { policy: 'fresh' }, visits = 2) {
  const def = singleNodeLoop('session cycle', {
    id: 'worker',
    kind: 'inference',
    label: 'Worker',
    config: {
      session: worker,
      prompt: { template: 'worker {{ run.id }}: {{ messages.last.content }}' },
    },
  });
  def.nodes.push(
    {
      id: 'reviewer',
      kind: 'inference',
      label: 'Reviewer',
      config: { session: reviewer, prompt: { template: 'reviewer' } },
    },
    {
      id: 'again',
      kind: 'decision',
      label: 'Again',
      config: {
        kind: 'expression',
        jsonata: '"repeat"',
        routes: [
          { id: 'repeat', label: 'Repeat', repeat: true },
          { id: 'finish', label: 'Finish' },
        ],
        maxPasses: visits,
      },
    },
  );
  def.edges = def.edges.filter((e) => e.from.node !== 'worker');
  def.edges.push(
    { id: 'review', from: { node: 'worker', port: 'out' }, to: { node: 'reviewer' } },
    { id: 'gate', from: { node: 'reviewer', port: 'out' }, to: { node: 'again' } },
    { id: 'repeat', from: { node: 'again', port: 'repeat' }, to: { node: 'worker' } },
    { id: 'limit', from: { node: 'again', port: 'limit-reached' }, to: { node: 'done' } },
    { id: 'finish', from: { node: 'again', port: 'finish' }, to: { node: 'done' } },
  );
  return def;
}

function handles(e: TestEngine, runId: string) {
  return e
    .events(runId)
    .filter((event) => event.type === 'harness.session')
    .map(({ nodeId, sessionId, mode }) => ({ nodeId, sessionId, mode }));
}

  async function setup(visits = 2, resetAfterVisits?: number) {
    const e = await createTestEngine();
    const fake = harness === 'codex' ? e.ports.harness : new FakeHarness([], 'claude');
    e.ports.harnesses[harness] = fake;
    if (harness === 'claude')
      e.ports.modelCatalog.entries.push({
        harness: 'claude',
        model: 'claude-opus-5-5',
        source: 'harness',
        displayName: 'Claude',
        efforts: ['low'],
        defaultEffort: 'low',
        enabled: true,
      });
    const def = cycle(
      { policy: 'resume-own', ...(resetAfterVisits === undefined ? {} : { resetAfterVisits }) },
      { policy: 'resume-own' },
      visits,
    );
    def.settings = { maxNodeVisits: null };
    for (const node of def.nodes)
      if (node.kind === 'inference')
        node.config = {
          ...node.config,
          harness,
          model: harness === 'codex' ? 'gpt-6-luna' : 'claude-opus-5-5',
          effort: 'low',
          harnessOptions: { sandbox: 'read-only', approval: 'never' },
        };
    return { e, fake, v: e.publish(def) };
  }

  it('keeps two parallel runs and both nodes separate', async () => {
    const { e, v } = await setup();
    const runs = await Promise.all([e.start(v.loopId), e.start(v.loopId)]);
    const firstIds: string[] = [];
    for (const run of runs) {
      expect((await e.settle(run.id)).status).toBe('succeeded');
      const events = handles(e, run.id);
      for (const nodeId of ['worker', 'reviewer']) {
        const visits = events.filter((x) => x.nodeId === nodeId);
        expect(visits.map((x) => x.mode)).toEqual(['fresh', 'resumed']);
        expect(visits[1]?.sessionId).toBe(visits[0]?.sessionId);
        firstIds.push(visits[0]!.sessionId);
      }
    }
    expect(new Set(firstIds).size).toBe(4);
    e.manager.stop();
  });

  it('does not inherit the private handle or reset count in a replay fork or child run', async () => {
    const { e, v } = await setup();
    const source = await e.runToIdle(v.loopId);
    const fork = await e.manager.replay({ runId: source.id, nodeId: 'worker' });
    expect((await e.settle(fork.id)).status).toBe('succeeded');

      singleNodeLoop('own-session parent', {
        id: 'sub',

    expect(parentRun.status).toBe('succeeded');
    const child = (await e.ports.runs.listChildren(parentRun.id))[0]!;
    const firstIds = [source.id, fork.id, child.id].map((id) => {
      const first = handles(e, id)[0]!;
      expect(first).toMatchObject({ nodeId: 'worker', mode: 'fresh' });
      const turns = e.events(id).filter((x) => x.type === 'harness.turn');
      expect(turns[0]?.resumeSessionId).toBeUndefined();

describe.each(['sqlite', 'memory'] as const)('%s session family selection', (backend) => {
  it('filters before sorting, preserving older same-family rows and existing keys', async () => {
    const handle = backend === 'sqlite' ? await openMemoryDatabase() : undefined;
    const repo: HarnessSessionRepository = handle
      ? new SqliteSessionRepository(handle.db)
      : new InMemorySessionRepository();
    try {
      const base = {
        runId: 'run',
        attempt: 1,
        scopeKey: 'loop:shared',
        status: 'finished' as const,
      };
      for (const [nodeId, harness, sessionId, time] of [
        ['c-old', 'codex', 'codex-old', '01'],
        ['a-old', 'claude', 'claude-old', '02'],
        ['c-new', 'codex', 'codex-new', '03'],
        ['a-empty', 'claude', undefined, '04'],
      ] as const)
        await repo.upsert({
          ...base,
          nodeId,
          harness,
          ...(sessionId ? { sessionId } : {}),
          updatedAt: '2026-10-07T12:00:' + time + '.000Z',
        });
      for (const [harness, expected] of [
        ['codex', 'codex-new'],
        ['claude', 'claude-old'],
      ] as const) {
        expect((await repo.latestWithSession('run', harness))?.sessionId).toBe(expected);
        expect((await repo.byScopeKey('loop:shared', harness))?.sessionId).toBe(expected);
        expect(await repo.latestWithSession('other', harness)).toBeUndefined();
        expect(await repo.byScopeKey('other', harness)).toBeUndefined();
      }
      await repo.upsert({
        ...base,
        nodeId: 'a-new',
        harness: 'claude',
        sessionId: 'claude-new',
        updatedAt: '2026-10-07T12:00:05.000Z',
      });
      expect((await repo.latestWithSession('run', 'codex'))?.sessionId).toBe('codex-new');
      expect((await repo.byScopeKey('loop:shared', 'claude'))?.sessionId).toBe('claude-new');
      expect((await repo.forNode('run', 'c-new'))?.harness).toBe('codex');
    } finally {
      handle?.close();
    }
  });
});

  const reset = within(session).getByLabelText('Reset after visits');
  expect(reset).toHaveValue(null);
  expect(within(session).getByText(/Blank means no automatic reset/)).toBeVisible();
  await user.type(reset, '2');
  expect(screen.getByTestId('saved')).toHaveTextContent(
    '"session":{"policy":"resume-own","resetAfterVisits":2}',
  );
  await user.clear(reset);
  expect(screen.getByTestId('saved')).toHaveTextContent('"session":{"policy":"resume-own"}');
  await user.type(reset, '0');
  expect(reset).toHaveAttribute('aria-invalid', 'true');
  expect(screen.getByTestId('saved')).toHaveTextContent('"session":{"policy":"resume-own"}');
});
